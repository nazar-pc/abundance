13/// Error of a read with [`SharedAlignedBufferReader`]
14#[derive(Debug, Copy, Clone, Eq, PartialEq)]
15pub enum ReadError {
16    /// The contents end before the read does
17    NotEnoughBytes {
18        /// How many more bytes the read needs.
19        ///
20        /// Counted from the position of the reader, so it is not larger than the number of bytes
21        /// missing for the rest of the data, whose size the reader doesn't know. Saturates at
22        /// `u32::MAX`, a read that needs more can't succeed with any buffer.
23        missing: NonZeroU32,
24    },
25    /// Bytes at the position of the reader are not aligned in memory as the read requires
26    Misaligned {
27        /// Required alignment in bytes
28        alignment: u32,
29    },
30}
31
32impl fmt::Display for ReadError {
33    #[inline]
34    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
35        match self {
36            Self::NotEnoughBytes { missing } => write!(f, "Need {missing} more bytes"),
37            Self::Misaligned { alignment } => {
38                write!(f, "Bytes are not aligned to {alignment} bytes")
39            }
40        }
41    }
42}
43
44impl core::error::Error for ReadError {}

46/// Reads the contents of a [`SharedAlignedBuffer`] front to back, see
47/// [`SharedAlignedBuffer::reader()`].
48///
49/// Bytes are borrowed from the buffer for as long as the buffer is borrowed, so any number of them
50/// can be held at once, while [`Self::read_buffer()`] creates buffers that reference a part of the
51/// contents without copying them.
52///
53/// A read that fails leaves the reader unchanged.
54#[derive(Debug, Clone)]
55pub struct SharedAlignedBufferReader<'a, A>
56where
57    A: Alignment,
58{
59    buffer: &'a SharedAlignedBuffer<A>,
60    /// Never larger than the length of the buffer
61    position: u32,
62}
63
64impl<'a, A> SharedAlignedBufferReader<'a, A>
65where
66    A: Alignment,
67{
68    #[inline(always)]
69    pub(crate) const fn new(buffer: &'a SharedAlignedBuffer<A>) -> Self {
70        Self {
71            buffer,
72            position: 0,
73        }
74    }
75
76    /// Number of bytes read so far, including padding
77    #[inline(always)]
78    pub const fn position(&self) -> u32 {
79        self.position
80    }
81
82    /// Bytes that were not read yet
83    #[inline(always)]
84    pub fn remaining(&self) -> &'a [u8] {
85        let contents: &'a [u8] = self.buffer.as_slice();
86        contents
87            .get(self.position as usize..)
88            .expect("Position is not larger than the length of the buffer; qed")
89    }
90
91    /// Read padding up to the next multiple of `B` bytes from the beginning of the contents.
92    ///
93    /// Returns padding bytes, which are empty if the position is already a multiple of `B`, so
94    /// that formats that require zero padding can check them.
95    #[inline]
96    pub fn read_to_alignment<B>(&mut self) -> Result<&'a [u8], ReadError>
97    where
98        B: Alignment,
99    {
100        let misalignment = self.position % B::ALIGN;
101        let padding = if misalignment == 0 {
102            0
103        } else {
104            B::ALIGN - misalignment
105        };
106        self.read_bytes(padding)
107    }
108
109    /// Read the next `len` bytes
110    #[inline]
111    pub fn read_bytes(&mut self, len: u32) -> Result<&'a [u8], ReadError> {
112        let start = self.advance(u64::from(len))?;
113        let contents: &'a [u8] = self.buffer.as_slice();
114        Ok(contents
115            .get(start as usize..)
116            .and_then(|bytes| bytes.get(..len as usize))
117            .expect("Within the contents as checked above; qed"))
118    }
119
120    /// Read the next `N` bytes
121    #[inline]
122    pub fn read_bytes_array<const N: usize>(&mut self) -> Result<&'a [u8; N], ReadError> {
123        const {
124            assert!(
125                N <= u32::MAX as usize,
126                "Buffers are never longer than `u32::MAX` bytes"
127            );
128        }
129        let bytes = self.read_bytes(u32::try_from(N).expect("Checked above; qed"))?;
130        Ok(bytes.try_into().expect("Exactly `N` bytes; qed"))
131    }
132
133    /// Create a buffer that references the next `len` bytes without copying them.
134    ///
135    /// The bytes must be aligned to `B` in memory, [`Self::read_to_alignment()`] reads padding in
136    /// front of them if necessary. The new buffer keeps the whole allocation alive unless it is
137    /// empty, see [`SharedAlignedBuffer::into_standalone()`].
138    #[inline]
139    pub fn read_buffer<B>(&mut self, len: u32) -> Result<SharedAlignedBuffer<B>, ReadError>
140    where
141        B: Alignment,
142    {
143        self.check_alignment(B::ALIGN)?;
144        let start = self.advance(u64::from(len))?;
145        if len == 0 {
146            return Ok(SharedAlignedBuffer::EMPTY);
147        }
148
149        Ok(SharedAlignedBuffer::from_inner(
150            self.buffer.inner.slice(start, len),
151        ))
152    }
153
154    /// Read a value of a trivial type.
155    ///
156    /// The bytes must be aligned for `T` in memory, [`Self::read_to_alignment()`] reads padding in
157    /// front of them if necessary. [`Unaligned`] reads values that are not aligned.
158    ///
159    /// [`Unaligned`]: ab_io_type::unaligned::Unaligned
160    #[cfg(feature = "io-type")]
161    #[inline]
162    pub fn read_trivial<T>(&mut self) -> Result<&'a T, ReadError>
163    where
164        T: AnyBitPattern,
165    {
166        self.check_alignment(u32::from(T::ALIGNMENT.get()))?;
167        let bytes = self.read_bytes(T::SIZE)?;
168        Ok(T::from_bytes_exact(bytes).expect("Aligned bytes of the size of `T`; qed"))
169    }
170
171    /// Read `N` values of a trivial type, like [`Self::read_trivial()`]
172    #[cfg(feature = "io-type")]
173    #[inline]
174    pub fn read_trivial_array<T, const N: usize>(&mut self) -> Result<&'a [T; N], ReadError>
175    where
176        [T; N]: AnyBitPattern,
177    {
178        self.read_trivial()
179    }
180
181    /// Read `count` values of a trivial type, like [`Self::read_trivial()`]
182    #[cfg(feature = "io-type")]
183    #[inline]
184    pub fn read_trivial_slice<T>(&mut self, count: u32) -> Result<&'a [T], ReadError>
185    where
186        T: AnyBitPattern,
187    {
188        self.check_alignment(u32::from(T::ALIGNMENT.get()))?;
189        let start = self.advance(u64::from(T::SIZE) * u64::from(count))?;
190        let contents: &'a [u8] = self.buffer.as_slice();
191        let (values, _remainder) = contents
192            .get(start as usize..)
193            .and_then(|bytes| T::slice_from_bytes(bytes, count as usize))
194            .expect("Aligned bytes of the size of `count` values of `T`; qed");
195        Ok(values)
196    }
197
198    /// Check that bytes at the position of the reader are aligned to `alignment` in memory
199    #[inline(always)]
200    fn check_alignment(&self, alignment: u32) -> Result<(), ReadError> {
201        if self
202            .remaining()
203            .as_ptr()
204            .addr()
205            .is_multiple_of(alignment as usize)
206        {
207            Ok(())
208        } else {
209            Err(ReadError::Misaligned { alignment })
210        }
211    }
212
213    /// Advance by `len` bytes, returns the position before advancing
214    #[inline(always)]
215    fn advance(&mut self, len: u64) -> Result<u32, ReadError> {
216        let remaining = self.buffer.len() - self.position;
217        let Some(len) = u32::try_from(len).ok().filter(|&len| len <= remaining) else {
218            let missing = NonZeroU64::new(len - u64::from(remaining))
219                .expect("More bytes are needed than remain as checked above; qed");
220            return Err(ReadError::NotEnoughBytes {
221                missing: NonZeroU32::try_from(missing).unwrap_or(NonZeroU32::MAX),
222            });
223        };
224
225        let start = self.position;
226        // Within the contents, whose length is a `u32`, as checked above
227        self.position += len;
228        Ok(start)
229    }
