1use crate::chiapos::constants::{PARAM_B, PARAM_C, PARAM_M};
2use crate::chiapos::table::types::R;
3use core::array;
4use core::ops::{Add, Sub};
5use core::simd::prelude::*;
6use core::simd::{Mask, Simd, SimdElement};
7
8/// Number of left targets calculated at a time
9const SIMD_FACTOR: usize = 16;
10
11/// Calculates left targets of the left bucket with a given parity.
12///
13/// Two `y`s match when `r` of the right entry is one of the [`PARAM_M`] targets calculated here
14/// from `r` of the left entry.
15#[derive(Debug)]
16pub(super) struct LeftTargets {
17    /// [`Self::SQUARES`] of the parity this instance was created for
18    squares: &'static [u16; const { usize::from(PARAM_M) }],
19}
20
21impl LeftTargets {
22    /// `(2 * m + parity)^2 % PARAM_C` for each `m` of each parity.
23    ///
24    /// This is the only part of a target that doesn't depend on `r`, the rest is a handful of
25    /// cheap SIMD instructions, which is why targets are calculated on the fly instead of being
26    /// read from a large precomputed table (the GPU implementation calculates them too).
27    const SQUARES: [[u16; const { usize::from(PARAM_M) }]; 2] =
28        [Self::squares(0), Self::squares(1)];
29
30    /// A single row of [`Self::SQUARES`].
31    ///
32    /// `array::from_fn()` would have been nicer, but it needs a `const` closure to be usable here.
33    const fn squares(parity: u16) -> [u16; const { usize::from(PARAM_M) }] {
34        let mut squares = [0; _];
35
36        let mut m = 0;
37        while m < PARAM_M {
38            let value = 2 * m + parity;
39            squares[usize::from(m)] = (value * value) % PARAM_C;
40            m += 1;
41        }
42
43        squares
44    }
45
46    #[inline(always)]
47    pub(super) fn new(odd_parity: bool) -> Self {
48        Self {
49            squares: &Self::SQUARES[usize::from(odd_parity)],
50        }
51    }
52
53    /// `a + b` modulo `modulus`.
54    ///
55    /// Both `a` and `b` must be below `modulus`, which puts the sum below `modulus * 2` and turns
56    /// the modulo into a single conditional subtraction.
57    ///
58    /// Writing `%` instead is not an option. LLVM derives the necessary bound for scalars only:
59    /// the range of a splat is whatever the scalar's range was, but `LazyValueInfo` has no rule
60    /// for `shufflevector`, and `InstCombine` canonicalizes a masked splat into exactly that
61    /// shape, so no way of writing this expression reaches the conditional subtraction.
62    /// `std::hint::assert_unchecked()` doesn't help either. What is emitted instead is the generic
63    /// reciprocal multiply, which for `u16x16` is 7 instructions with two multiplications against
64    /// 4 here on x86-64, and 16 with six against 7 on aarch64.
65    // TODO: Use `%` once LLVM looks through splats when computing ranges:
66    //  https://github.com/llvm/llvm-project/issues/223468
67    #[inline(always)]
68    fn add_mod<T, const N: usize>(a: Simd<T, N>, b: Simd<T, N>, modulus: Simd<T, N>) -> Simd<T, N>
69    where
70        T: SimdElement + Default,
71        Simd<T, N>: Add<Output = Simd<T, N>>
72            + Sub<Output = Simd<T, N>>
73            + SimdPartialOrd<Mask = Mask<T::Mask, N>>,
74    {
75        let sum = a + b;
76
77        sum - modulus.simd_le(sum).select(modulus, Simd::default())
78    }
79
80    /// Calculate all [`PARAM_M`] targets of `r` at once
81    #[inline(always)]
82    pub(super) fn calculate(&self, r: R) -> [R; const { usize::from(PARAM_M) }] {
83        let r = u16::from(r);
84        let c = Simd::splat(r / PARAM_C);
85        let d = Simd::splat(r % PARAM_C);
86        let param_b = Simd::splat(PARAM_B);
87        let param_c = Simd::splat(PARAM_C);
88
89        let mut targets = [0u16; _];
90
91        for (index, (targets, squares)) in targets
92            .as_chunks_mut::<SIMD_FACTOR>()
93            .0
94            .iter_mut()
95            .zip(self.squares.as_chunks::<SIMD_FACTOR>().0)
96            .enumerate()
97        {
98            let ms = Simd::splat((index * SIMD_FACTOR) as u16)
99                + Simd::from_array(array::from_fn(|offset| offset as u16));
100
101            let target_c = Self::add_mod(c, ms, param_b);
102            let target_d = Self::add_mod(d, Simd::from_array(*squares), param_c);
103
104            *targets = (target_c * param_c + target_d).to_array();
105        }
106
107        R::array_from_repr(targets)
108    }
109}
