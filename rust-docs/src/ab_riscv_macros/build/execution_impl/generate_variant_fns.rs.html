13/// A single named+typed parameter, used both for variant-specific fields and for arguments
14/// shared by all generated functions (derived from `execute()`'s own signature)
15struct Param<'a> {
16    /// Local identifier used both as the parameter name and as the name of the value passed at
17    /// the call site
18    ident: &'a Ident,
19    ty: Type,
20}
21
22/// Name of the function generated for a single instruction variant's execution logic.
23///
24/// Shared with the threaded emitter, which calls those functions rather than generating its own.
25pub(super) fn variant_fn_name(enum_name: &Ident, variant: &Ident) -> Ident {
26    format_ident!(
27        "execute_{}_{}",
28        enum_name.to_string().to_snake_case(),
29        variant.to_string().to_snake_case(),
30    )
31}

33/// This rewrites every `Self` into a fully qualified concrete instruction enum type instead.
34///
35/// `execute()`'s own body is written against `Self`, but the functions generated here are
36/// standalone functions without `Self`, which needs correction.
37struct SelfRewriter<'a> {
38    self_ty: &'a Type,
39}
40
41impl SelfRewriter<'_> {
42    /// If `path` starts with `Self::`, returns the `<self_ty as Instruction>` qualifier and the
43    /// remaining path (`Instruction` followed by whatever came after `Self::`)
44    fn qualify_self_item(&self, path: &Path) -> Option<(QSelf, Path)> {
45        let first = path.segments.first()?;
46        if first.ident != "Self" || !first.arguments.is_none() || path.segments.len() < 2 {
47            return None;
48        }
49
50        let self_ty = &self.self_ty;
51        let rest = path.segments.iter().skip(1);
52        let TypePath {
53            attrs: _,
54            qself,
55            path,
56        } = parse_quote! {
57            <#self_ty as Instruction>::#(#rest)::*
58        };
59
60        Some((qself.expect("Statically known to be present; qed"), path))
61    }
62}
63
64impl VisitMut for SelfRewriter<'_> {
65    fn visit_type_mut(&mut self, i: &mut Type) {
66        if let Type::Path(TypePath {
67            qself: None,
68            path,
69            attrs: _,
70        }) = i
71        {
72            if path.segments.len() == 1 && path.segments[0].ident == "Self" {
73                *i = self.self_ty.clone();
74            } else if let Some((qself, path)) = self.qualify_self_item(path) {
75                *i = Type::Path(TypePath {
76                    attrs: Vec::new(),
77                    qself: Some(qself),
78                    path,
79                });
80            }
81        }
82        visit_mut::visit_type_mut(self, i);
83    }
84
85    fn visit_expr_mut(&mut self, i: &mut Expr) {
86        if let Expr::Path(ExprPath {
87            qself: None,
88            path,
89            attrs,
90        }) = i
91            && let Some((qself, path)) = self.qualify_self_item(path)
92        {
93            *i = Expr::Path(ExprPath {
94                attrs: mem::take(attrs),
95                qself: Some(qself),
96                path,
97            });
98        }
99        visit_mut::visit_expr_mut(self, i);
100    }
101}

103/// Extracts the single generic type argument out of a type like `Foo<Bar>`, returning `Bar`.
104///
105/// This is used to recover field types of destructured arguments of `execute()` (such as
106/// `Rs1Rs2OperandValues<T>`) without having access to the original struct definition.
107fn extract_single_generic_type(ty: &Type) -> anyhow::Result<&Type> {
108    let Type::Path(type_path) = ty else {
109        return Err(anyhow::anyhow!(
110            "Expected a generic path type for a destructured `execute()` argument, found: {}",
111            ty.to_token_stream()
112        ));
113    };
114    let last_segment = type_path
115        .path
116        .segments
117        .last()
118        .expect("Path is never empty; qed");
119    let PathArguments::AngleBracketed(generic_args) = &last_segment.arguments else {
120        return Err(anyhow::anyhow!(
121            "Expected a single generic argument in destructured `execute()` argument type `{}`",
122            ty.to_token_stream()
123        ));
124    };
125    let mut args = generic_args.args.iter();
126    let (Some(GenericArgument::Type(inner_ty)), None) = (args.next(), args.next()) else {
127        return Err(anyhow::anyhow!(
128            "Expected exactly one generic type argument in destructured `execute()` argument \
129            type `{}`",
130            ty.to_token_stream()
131        ));
132    };
133
134    Ok(inner_ty)
135}

137/// Extracts parameters shared by all generated per-variant functions from `execute()`'s own
138/// signature (excluding `self`)
139fn extract_shared_params(sig: &Signature) -> anyhow::Result<Vec<Param<'_>>> {
140    let mut shared_params = Vec::new();
141
142    for input in &sig.inputs {
143        let FnArg::Typed(PatType {
144            pat,
145            ty,
146            attrs: _,
147            colon_token: _,
148        }) = input
149        else {
150            // `self` receiver is ignored (matched on by `match` and unused after that)
151            continue;
152        };
153
154        match pat.as_ref() {
155            Pat::Ident(PatIdent {
156                ident,
157                attrs: _,
158                by_ref: _,
159                mutability: _,
160                subpat: _,
161            }) => {
162                shared_params.push(Param {
163                    ident,
164                    ty: ty.as_ref().clone(),
165                });
166            }
167            Pat::Struct(pat_struct) => {
168                let field_ty = extract_single_generic_type(ty)?;
169                for field in &pat_struct.fields {
170                    let Pat::Ident(PatIdent {
171                        ident,
172                        attrs: _,
173                        by_ref: _,
174                        mutability: _,
175                        subpat: _,
176                    }) = field.pat.as_ref()
177                    else {
178                        return Err(anyhow::anyhow!(
179                            "Destructured `execute()` argument field `{}` must be bound to an \
180                            identifier",
181                            field.to_token_stream()
182                        ));
183                    };
184                    shared_params.push(Param {
185                        ident,
186                        ty: field_ty.clone(),
187                    });
188                }
189            }
190            _ => {
191                return Err(anyhow::anyhow!(
192                    "Unsupported `execute()` argument pattern `{}`",
193                    pat.to_token_stream()
194                ));
195            }
196        }
197    }
198
199    Ok(shared_params)
200}

202/// Extracts variant-specific parameters (fields bound by the match arm's pattern that are not `_`)
203/// together with their types looked up from the enum variant's definition
204fn extract_variant_params<'a>(
205    arm_pat: &'a Pat,
206    variant: &'a Variant,
207) -> anyhow::Result<Vec<Param<'a>>> {
208    let Pat::Struct(pat_struct) = arm_pat else {
209        return Err(anyhow::anyhow!(
210            "Expected a struct pattern for instruction variant `{}`, found: {}",
211            variant.ident,
212            arm_pat.to_token_stream()
213        ));
214    };
215    let Fields::Named(fields_named) = &variant.fields else {
216        return Err(anyhow::anyhow!(
217            "Instruction variant `{}` must have named fields",
218            variant.ident
219        ));
220    };
221
222    let mut params = Vec::new();
223    for field_pat in &pat_struct.fields {
224        let Pat::Ident(PatIdent {
225            ident: local_ident,
226            attrs: _,
227            by_ref: _,
228            mutability: _,
229            subpat: _,
230        }) = field_pat.pat.as_ref()
231        else {
232            // `_` or another unsupported nested pattern, treated as unused
233            continue;
234        };
235
236        let Member::Named(field_name) = &field_pat.member else {
237            return Err(anyhow::anyhow!(
238                "Instruction variant `{}` fields must be named",
239                variant.ident
240            ));
241        };
242
243        let field = fields_named
244            .named
245            .iter()
246            .find(|field| field.ident.as_ref() == Some(field_name))
247            .ok_or_else(|| {
248                anyhow::anyhow!(
249                    "Field `{field_name}` not found in instruction variant `{}`",
250                    variant.ident
251                )
252            })?;
253
254        params.push(Param {
255            ident: local_ident,
256            ty: field.ty.clone(),
257        });
258    }
259
260    Ok(params)
261}

263/// Splits a composed `execute()` method into a `#[inline(always)]` function generated for each
264/// enum variant, replacing its match arm bodies with calls to those functions.
265///
266/// `execute()`'s generic parameters are kept in full on every generated function, and call sites
267/// always specify them explicitly. This is much simpler than tracking their usage. Same with
268/// arguments.
269///
270/// `variants` and `arms` must have the same length and be in the same order (typically
271/// `enum_definition.instructions` and the correspondingly ordered arms extracted from `execute()`).
272#[expect(clippy::allow_attributes, reason = "Attribute below")]
273#[allow(
274    clippy::too_many_arguments,
275    reason = "Each parameter is a distinct piece of already-parsed `execute()` context; bundling \
276    them would not make this any clearer"
277)]
278pub(super) fn generate_variant_fns(
279    enum_name: &Ident,
280    self_ty: &Type,
281    generics: &Generics,
282    constness: Option<Token![const]>,
283    no_panic_attr: Option<&Attribute>,
284    execute_sig: &Signature,
285    variants: &[Rc<Variant>],
286    match_arms: &mut [Arm],
287) -> anyhow::Result<Vec<ItemFn>> {
288    let mut self_rewriter = SelfRewriter { self_ty };
289
290    let mut shared_params = extract_shared_params(execute_sig)?;
291    let mut output = execute_sig.output.clone();
292    for param in &mut shared_params {
293        self_rewriter.visit_type_mut(&mut param.ty);
294    }
295    self_rewriter.visit_return_type_mut(&mut output);
296
297    let mut generated_fns = Vec::with_capacity(variants.len());
298    let generic_params = &generics.params;
299    let where_clause = &generics.where_clause;
300
301    for (variant, arm) in variants.iter().zip(match_arms) {
302        let mut variant_params = extract_variant_params(&arm.pat, variant)
303            .with_context(|| format!("Instruction variant `{}`", variant.ident))?;
304        for param in &mut variant_params {
305            self_rewriter.visit_type_mut(&mut param.ty);
306        }
307
308        let fn_name = variant_fn_name(enum_name, &variant.ident);
309
310        let call_args = variant_params
311            .iter()
312            .chain(&shared_params)
313            .map(|param| &param.ident);
314        let mut original_arm_body = mem::replace(
315            arm.body.as_mut(),
316            parse_quote! { #fn_name::<#generic_params>(#( #call_args ),*) },
317        );
318        arm.comma = Some(<Token![,]>::default());
319
320        self_rewriter.visit_expr_mut(&mut original_arm_body);
321
322        let call_args = variant_params
323            .iter()
324            .chain(&shared_params)
325            .map(|param| &param.ident);
326        let args = variant_params
327            .iter()
328            .chain(&shared_params)
329            .map(|Param { ident, ty }| {
330                parse_quote! { #ident: #ty }
331            })
332            .collect::<Vec<FnArg>>();
333
334        generated_fns.push(parse_quote! {
335            #[expect(clippy::allow_attributes, reason = "Attributes below")]
336            #[allow(
337                clippy::extra_unused_type_parameters,
338                reason = "Much easier in generated code than properly tracking used generics"
339            )]
340            #[allow(clippy::too_many_arguments, reason = "Generated and force-inlined")]
341            // Comments will be stripped, this will suppress some of the lints that are caused by it
342            #[allow(
343                clippy::undocumented_unsafe_blocks,
344                reason = "Comments will be stripped, this will suppress some of the lints \
345                that are caused by it"
346            )]
347            #no_panic_attr
348            #[inline(always)]
349            #constness fn #fn_name<#generic_params>(#( #args ),*) #output
350                #where_clause
351            {
352                // This prevents warnings about unused arguments in a way that avoids suppression of
353                // lints for the implementation itself
354                #[expect(clippy::let_underscore_untyped, reason = "Generated code")]
355                {
356                    #( let _ = #call_args; )*
357                }
358                #[allow(unused_braces, reason = "Generated code")]
359                #original_arm_body
360            }
361        });
362    }
363
364    Ok(generated_fns)
365}
