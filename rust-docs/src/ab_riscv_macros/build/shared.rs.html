11/// Whether an attribute has a given name.
12///
13/// Only the last path segment is compared, so path-qualified attributes like
14/// `#[ab_riscv_macros::instruction]` are recognized too.
15pub(super) fn is_named_attribute(attribute: &Attribute, name: &str) -> bool {
16    attribute
17        .path()
18        .segments
19        .last()
20        .is_some_and(|segment| segment.ident == name)
21}

23/// Arguments of an attribute, if there are any.
24///
25/// Empty parentheses like `#[instruction()]` are treated as no arguments, the same way attribute
26/// macros see them.
27pub(super) fn attribute_arguments(attribute: &Attribute) -> Option<String> {
28    match &attribute.meta {
29        Meta::Path(_) => None,
30        Meta::List(meta_list) => {
31            (!meta_list.tokens.is_empty()).then(|| meta_list.tokens.to_string())
32        }
33        Meta::NameValue(meta_name_value) => {
34            Some(meta_name_value.value.to_token_stream().to_string())
35        }
36    }
37}

39pub(super) fn collect_all_dependencies<InitialDependencies>(
40    state: &State,
41    initial_dependencies: InitialDependencies,
42) -> Result<Vec<(Ident, &KnownEnumDefinition)>, Ident>
43where
44    InitialDependencies: Iterator<Item = Ident>,
45{
46    let mut already_inserted = HashSet::new();
47    let mut all_dependencies = Vec::new();
48    let mut new_dependencies = VecDeque::from_iter(initial_dependencies);
49
50    while let Some(dependency_enum_name) = new_dependencies.pop_front() {
51        let Some(dependency_enum_definition) =
52            state.get_known_enum_definition(&dependency_enum_name)
53        else {
54            return Err(dependency_enum_name);
55        };
56
57        if !already_inserted.insert(dependency_enum_name.clone()) {
58            continue;
59        }
60
61        all_dependencies.push((dependency_enum_name, dependency_enum_definition));
62        new_dependencies.extend(
63            dependency_enum_definition
64                .direct_dependencies
65                .iter()
66                .cloned(),
67        );
68    }
69
70    Ok(all_dependencies)
71}

73/// Whether an instruction enum is implemented by an instruction set, with `is_present` checking
74/// whether an individual instruction is present in it.
75///
76/// An enum is implemented when at least one of its own instructions is present (otherwise it was
77/// ignored in one way or another) and the same is true for all enums it inherits. Enums without
78/// own instructions (like `B`) are only implemented when all enums they inherit are.
79pub(super) fn is_enum_implemented<IsPresent>(
80    state: &State,
81    enum_name: &Ident,
82    is_present: IsPresent,
83) -> bool
84where
85    IsPresent: Fn(&Ident) -> bool,
86{
87    collect_all_dependencies(state, iter::once(enum_name.clone())).is_ok_and(|all_enums| {
88        all_enums.into_iter().all(|(_enum_name, enum_definition)| {
89            enum_definition.own_instructions.is_empty()
90                || enum_definition
91                    .own_instructions
92                    .iter()
93                    .any(|instruction| is_present(&instruction.ident))
94        })
95    })
96}

98/// Strips `[const]` from `where` predicates.
99///
100/// Used both for a non-`const` implementation that inherits arms from `const` ones and for the
101/// threaded implementation, which is never `const`.
102pub(super) fn strip_const_where_predicates(predicates: &mut Punctuated<WherePredicate, Token![,]>) {
103    for predicate in predicates {
104        if let WherePredicate::Type(predicate_type) = predicate {
105            // TODO: `BRCONST` is a hack that allows `syn` to parse unstable Rust syntax
106            //  around const traits and such. It will change to a proper modifier once
107            //  stabilized
108            if predicate_type.bounds.first() == Some(&parse_quote! { BRCONST }) {
109                predicate_type.bounds = mem::take(&mut predicate_type.bounds)
110                    .into_iter()
111                    .skip(1)
112                    .collect();
113            }
114        }
115    }
