1use syn::visit::{Visit, visit_expr};
2use syn::{Block, Expr, ExprCall, ExprPath, ExprReturn, ExprStruct, Ident, Path, PathArguments};
3
4struct ForbiddenChecker<'a> {
5    found: bool,
6    enum_name: &'a Ident,
7}
8
9impl<'ast> Visit<'ast> for ForbiddenChecker<'ast> {
10    fn visit_expr(&mut self, i: &'ast Expr) {
11        if self.found {
12            return;
13        }
14
15        match i {
16            Expr::Return(ExprReturn {
17                attrs: _,
18                return_token: _,
19                expr: _,
20            }) => {
21                self.found = true;
22            }
23
24            // Unit variant: `Enum::Foo` (qself must be None to avoid <T as Trait>::Assoc false
25            // positives)
26            Expr::Path(ExprPath {
27                attrs: _,
28                qself: None,
29                path,
30            }) if is_forbidden_variant_path(path, self.enum_name) => {
31                self.found = true;
32            }
33
34            // Tuple variant: `Enum::Foo(...)`
35            Expr::Call(ExprCall {
36                attrs: _,
37                func,
38                paren_token: _,
39                args: _,
40            }) => {
41                if let Expr::Path(ExprPath {
42                    attrs: _,
43                    qself: None,
44                    path,
45                }) = func.as_ref()
46                    && is_forbidden_variant_path(path, self.enum_name)
47                {
48                    self.found = true;
49                }
50            }
51
52            // Struct variant: `Enum::Foo { .. }`
53            Expr::Struct(ExprStruct {
54                attrs: _,
55                qself: _,
56                path,
57                brace_token: _,
58                fields: _,
59                dot2_token: _,
60                rest: _,
61            }) if is_forbidden_variant_path(path, self.enum_name) => {
62                self.found = true;
63            }
64
65            _ => {}
66        }
67
68        // Recurse
69        visit_expr(self, i);
70    }
71}
72
73fn is_forbidden_variant_path(path: &Path, enum_name: &Ident) -> bool {
74    let mut segments = path.segments.iter().rev();
75
76    let Some(variant_segment) = segments.next() else {
77        return false;
78    };
79    if !matches!(variant_segment.arguments, PathArguments::None) {
80        return false;
81    }
82
83    let Some(enum_segment) = segments.next() else {
84        return false;
85    };
86    &enum_segment.ident == enum_name
87}

89/// Returns `true` if the block contains either an explicit `return` expression or a direct
90/// construction of any variant of the given enum (e.g. `Enum::Foo`, `Enum::Foo(arg)`,
91/// `Enum::Foo { ... }`, including with generics like `MyError::<T>::Foo`).
92pub(super) fn block_contains_forbidden_syntax(block: &Block, enum_name: &Ident) -> bool {
93    let mut checker = ForbiddenChecker {
94        found: false,
95        enum_name,
96    };
97    checker.visit_block(block);
98    checker.found
99}
