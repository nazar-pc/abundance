17/// Result of polling the entropy source behind the `Zkr` extension's `seed` CSR, corresponding to
18/// one of the specification's `OPST` states
19#[derive(Debug, Clone, Copy, PartialEq, Eq)]
20pub enum ZkrSeedPoll {
21    /// Built-in self-test is being performed (`OPST=BIST`).
22    ///
23    /// If this is returned after some other state was previously observed, it signals a
24    /// non-fatal, but noteworthy, self-test alarm.
25    Bist,
26    /// A sufficient amount of entropy is not yet available (`OPST=WAIT`).
27    ///
28    /// This is not an error condition and may be observed more often than [`Self::Es16`], since
29    /// physical entropy sources often have low bandwidth.
30    Wait,
31    /// 16 bits of randomness are available (`OPST=ES16`), guaranteed to meet the specification's
32    /// minimum entropy requirements regardless of implementation.
33    Es16(u16),
34    /// An unrecoverable self-test error has occurred (`OPST=DEAD`).
35    ///
36    /// Once returned, all future polls are expected to also return [`Self::Dead`].
37    Dead,
38}
39
40/// Entropy source behind the `Zkr` extension's `seed` CSR.
41///
42/// Every access to `seed` polls the entropy source once.
43pub const trait ZkrSeedSource {
44    /// Poll the entropy source once, advancing its internal state as necessary
45    fn poll_seed(&mut self) -> ZkrSeedPoll;
46}
47
48// Convenience for threaded execution
49const impl<T> ZkrSeedSource for &mut T
50where
51    T: [const] ZkrSeedSource,
52{
53    #[inline(always)]
54    fn poll_seed(&mut self) -> ZkrSeedPoll {
55        T::poll_seed(self)
56    }
57}

62#[instruction_execution]
63const impl<Reg, ExtState, CustomError> ExecutableInstructionCsr<ExtState, CustomError>
64    for ZkrInstruction<Reg>
65where
66    Reg: [const] Register,
67    ExtState: [const] ZkrSeedSource,
68    CustomError: [const] Destruct,
69{
70    /// Reads of `seed` are pass-through: the raw stored value already reflects the outcome of the
71    /// most recent poll (see [`Self::prepare_csr_write()`]).
72    ///
73    /// Per specification, only genuine read-write accesses (not e.g. `csrrs`/`csrrc` with
74    /// `rs1 = x0`) are legal against `seed`; a pure read is rejected as an illegal access.
75    #[inline(always)]
76    #[cfg_attr(feature = "no-panic", no_panic_const::no_panic(const))]
77    fn prepare_csr_read(
78        _ext_state: &ExtState,
79        csr_index: u16,
80        will_write: bool,
81        raw_value: Reg::Type,
82        output_value: &mut Reg::Type,
83    ) -> Result<bool, CsrError<CustomError>> {
84        if csr_index == SEED_CSR_INDEX {
85            if will_write {
86                *output_value = raw_value;
87                Ok(true)
88            } else {
89                Err(CsrError::IllegalRead { csr_index })
90            }
91        } else {
92            Ok(false)
93        }
94    }
95
96    /// Writes to `seed` ignore the write value entirely (per specification) and instead poll the
97    /// entropy source once, storing the newly encoded [`ZkrSeedPoll`] to be observed by the next
98    /// read
99    #[inline(always)]
100    #[cfg_attr(feature = "no-panic", no_panic_const::no_panic(const))]
101    fn prepare_csr_write(
102        ext_state: &mut ExtState,
103        csr_index: u16,
104        write_value: Reg::Type,
105        output_value: &mut Reg::Type,
106    ) -> Result<bool, CsrError<CustomError>> {
107        // Necessary to avoid unused variable depending on instructions composition
108        let _: Reg::Type = write_value;
109
110        if csr_index == SEED_CSR_INDEX {
111            *output_value = zkr_helpers::encode_seed_poll::<Reg>(ext_state.poll_seed());
112            Ok(true)
113        } else {
114            Ok(false)
115        }
116    }
117}
