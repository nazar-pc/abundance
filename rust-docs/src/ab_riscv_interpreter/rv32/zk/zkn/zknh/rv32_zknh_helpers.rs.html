59// SHA-512 sigma0: ROR64(x,1) ^ ROR64(x,8) ^ SHR64(x,7)
60
61/// High 32 bits of SHA-512 sigma0. rs1 = HIGH word, rs2 = LOW word.
62///
63/// ```text
64/// ROR64(x,1).hi  = (rs1>>1)  ^ (rs2<<31)
65/// ROR64(x,8).hi  = (rs1>>8)  ^ (rs2<<24)
66/// SHR64(x,7).hi  =  rs1>>7              <- shift: no rs2 contribution
67/// ```
68#[inline(always)]
69#[doc(hidden)]
70#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
71pub fn sha512sig0h(rs1: u32, rs2: u32) -> u32 {
72    // TODO: Miri is excluded because corresponding intrinsic is not implemented there
73    cfg_select! {
74        all(not(miri), target_arch = "riscv32", target_feature = "zknh") => {
75            // SAFETY: Compile-time checked for supported feature
76            unsafe { core::arch::riscv32::sha512sig0h(rs1, rs2) }
77        }
78        _ => (rs1 >> 1) ^ (rs2 << 31) ^ (rs1 >> 8) ^ (rs2 << 24) ^ (rs1 >> 7),
79    }
80}

147// SHA-512 Sum0: ROR64(x,28) ^ ROR64(x,34) ^ ROR64(x,39)
148//
149// Sail: let x = X(rs2) @ X(rs1)  =>  x[63:32] = rs2 (HIGH), x[31:0] = rs1 (LOW)
150// sum0r produces the LOW half of the result.
151//
152// ROR64({hi=rs2, lo=rs1}, 28).lo  = (rs1>>28) ^ (rs2<<4)   [n=28 < 32]
153// ROR64({hi=rs2, lo=rs1}, 34).lo  = (rs2>>2)  ^ (rs1<<30)  [n=34 = 32+2]
154// ROR64({hi=rs2, lo=rs1}, 39).lo  = (rs2>>7)  ^ (rs1<<25)  [n=39 = 32+7]
155
156/// Low 32 bits of SHA-512 Sum0. rs1 = LOW word, rs2 = HIGH word.
157///
158/// All three terms are rotations, so no asymmetric shift contribution.
159///
160/// ```text
161/// ROR64(x,28).lo = (rs1>>28) ^ (rs2<<4)
162/// ROR64(x,34).lo = ROR64(x,32+2).lo  = (rs2>>2)  ^ (rs1<<30)
163/// ROR64(x,39).lo = ROR64(x,32+7).lo  = (rs2>>7)  ^ (rs1<<25)
164/// ```
165#[inline(always)]
166#[doc(hidden)]
167#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
168pub fn sha512sum0r(rs1: u32, rs2: u32) -> u32 {
169    // TODO: Miri is excluded because corresponding intrinsic is not implemented there
170    cfg_select! {
171        all(not(miri), target_arch = "riscv32", target_feature = "zknh") => {
172            // SAFETY: Compile-time checked for supported feature
173            unsafe { core::arch::riscv32::sha512sum0r(rs1, rs2) }
174        }
175        _ => (rs1 >> 28) ^ (rs2 << 4) ^ (rs2 >> 2) ^ (rs1 << 30) ^ (rs2 >> 7) ^ (rs1 << 25),
176    }
177}
