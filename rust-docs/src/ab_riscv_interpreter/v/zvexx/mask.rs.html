55        match self {
56            // Mask-register logical instructions (§16.1).
57            // These compute the body elements [vstart, vl); prestart bits [0, vstart) are
58            // undisturbed and the tail (past vl) is tail-agnostic (realised here as
59            // undisturbed). They still require vtype to be valid (vill=0); any vector
60            // instruction must be rejected when vill is set, regardless of whether it uses
61            // SEW or vl.
62            Self::Vmandn { vd, vs2, vs1 } => {
63                if !ext_state.vector_instructions_allowed() {
64                    ::core::hint::cold_path();
65                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
66                        address: PackedAddress::new(
67                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
68                        ),
69                    });
70                }
71                if ext_state.vtype().is_none() {
72                    ::core::hint::cold_path();
73                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
74                        address: PackedAddress::new(
75                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
76                        ),
77                    });
78                }
79                // SAFETY: all VReg values are valid indices < 32; `vl <= VLEN` and
80                // `vstart <= vl` are architectural invariants; snapshot-before-write inside
81                // the helper means vd may overlap vs2 or vs1 safely.
82                unsafe {
83                    zvexx_mask_helpers::execute_mask_logical_op(ext_state, vd, vs2, vs1, |a, b| {
84                        a && !b
85                    });
86                }
87            }

256            // vcpop.m (§16.2): count set bits in vs2 over active elements, write to GPR rd.
257            Self::Vcpop { rd, vs2, vm } => {
258                if !ext_state.vector_instructions_allowed() {
259                    ::core::hint::cold_path();
260                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
261                        address: PackedAddress::new(
262                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
263                        ),
264                    });
265                }
266                // vcpop/vfirst require a valid vtype to know vl, but do not use SEW.
267                if ext_state.vtype().is_none() {
268                    ::core::hint::cold_path();
269                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
270                        address: PackedAddress::new(
271                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
272                        ),
273                    });
274                }
275                // SAFETY: `vl <= VLMAX <= VLEN`; `vstart <= vl` by spec invariant.
276                let rd_value = unsafe { zvexx_mask_helpers::execute_vcpop(ext_state, vs2, vm) };
277
278                return ExecutionResult::Continue {
279                    rd,
280                    value: rd_value,
281                };
282            }
283            // vfirst.m (§16.3): find lowest-numbered active set bit in vs2, write index to rd.
284            Self::Vfirst { rd, vs2, vm } => {
285                if !ext_state.vector_instructions_allowed() {
286                    ::core::hint::cold_path();
287                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
288                        address: PackedAddress::new(
289                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),

294                    ::core::hint::cold_path();
295                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
296                        address: PackedAddress::new(
297                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
298                        ),
299                    });
300                }
301                // SAFETY: same as `Vcpop`
302                let rd_value = unsafe { zvexx_mask_helpers::execute_vfirst(ext_state, vs2, vm) };
303
304                return ExecutionResult::Continue {
305                    rd,
306                    value: rd_value,
307                };
308            }
309            // vmsbf.m (§16.4): set-before-first mask bit.
310            // Constraints: vd != vs2 (overlap illegal), vm=false implies vd != v0.
311            Self::Vmsbf { vd, vs2, vm } => {
312                if !ext_state.vector_instructions_allowed() {
313                    ::core::hint::cold_path();
314                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
315                        address: PackedAddress::new(
316                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
317                        ),
318                    });
319                }
320                if ext_state.vtype().is_none() {
321                    ::core::hint::cold_path();
322                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
323                        address: PackedAddress::new(
324                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
325                        ),
326                    });
327                }
328                // Spec §16.4: vmsbf/vmsif/vmsof with vstart != 0 raise an illegal instruction
329                // exception.
330                if ext_state.vstart() != Vstart::ZERO {
331                    ::core::hint::cold_path();
332                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
333                        address: PackedAddress::new(
334                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
335                        ),
336                    });
337                }
338                // Per spec §16.4: vd must not overlap vs2
339                if vd == vs2 {
340                    ::core::hint::cold_path();
341                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
342                        address: PackedAddress::new(
343                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
344                        ),
345                    });
346                }
347                if !vm && vd == VReg::V0 {
348                    ::core::hint::cold_path();
349                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
350                        address: PackedAddress::new(
351                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
352                        ),
353                    });
354                }
355                let vl = ext_state.vl();
356                // SAFETY: `vd != vs2` checked above; `vd != v0` when masked checked above;
357                // `vstart == 0` checked above; `vl <= VLEN`.
358                unsafe {
359                    zvexx_mask_helpers::execute_vmsbf(ext_state, vd, vs2, vm, vl);
360                }
361            }
362            // vmsof.m (§16.5): set-only-first mask bit.
363            // Same overlap constraints as vmsbf.
364            Self::Vmsof { vd, vs2, vm } => {
365                if !ext_state.vector_instructions_allowed() {
366                    ::core::hint::cold_path();

464            // viota.m (§16.8): write prefix popcount of vs2 bits as SEW-wide elements into vd.
465            // Constraints: vd must not overlap vs2 or v0 (when masked); vd alignment per LMUL;
466            // vstart must be zero (mandatory trap per spec §16.8). There is no SEW-width
467            // constraint: if SEW is too narrow to hold the prefix count the result simply wraps
468            // (truncates to SEW), matching the spec's "integer operations wrap around on overflow"
469            // rule rather than raising an exception.
470            Self::Viota { vd, vs2, vm } => {
471                if !ext_state.vector_instructions_allowed() {
472                    ::core::hint::cold_path();
473                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
474                        address: PackedAddress::new(
475                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
476                        ),
477                    });
478                }
479                let Some(vtype) = ext_state.vtype() else {
480                    ::core::hint::cold_path();
481                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
482                        address: PackedAddress::new(
483                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
484                        ),
485                    });
486                };
487                // Spec §16.8: viota.m with vstart != 0 raises an illegal instruction exception.
488                if ext_state.vstart() != Vstart::ZERO {
489                    ::core::hint::cold_path();
490                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
491                        address: PackedAddress::new(
492                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
493                        ),
494                    });
495                }
496                let group_regs = vtype.vlmul().register_count().get();
497                let vd_idx = vd.to_bits();
498                if !vd_idx.is_multiple_of(group_regs) || vd_idx + group_regs > 32 {
499                    ::core::hint::cold_path();
500                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
501                        address: PackedAddress::new(
502                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
503                        ),
504                    });
505                }
506                // vd must not overlap vs2; vs2 is always a single mask register (group size 1).
507                let vd_start = u32::from(vd.to_bits());
508                let vs2_start = u32::from(vs2.to_bits());
509                if vd_start < vs2_start + 1 && vs2_start < vd_start + u32::from(group_regs) {
510                    ::core::hint::cold_path();
511                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
512                        address: PackedAddress::new(
513                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
514                        ),
515                    });
516                }
517                if !vm && vd == VReg::V0 {
518                    ::core::hint::cold_path();
519                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
520                        address: PackedAddress::new(
521                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
522                        ),
523                    });
524                }
525                let sew = vtype.vsew();
526                let vl = ext_state.vl();
527                // SAFETY: vd alignment checked above; vd group does not overlap vs2 checked above;
528                // `vm=false` implies `vd != v0` checked above; vstart == 0 checked above;
529                // `vl <= VLMAX = group_regs * VLEN.bytes() / sew_bytes`, all element indices valid.
530                unsafe {
531                    zvexx_mask_helpers::execute_viota(ext_state, vd, vs2, vm, vl, sew);
532                }
533            }
534            // vid.v (§16.9): write element index i as SEW-wide integer into vd[i].
535            // Constraints: vm=false implies vd != v0; vd alignment per LMUL.
536            Self::Vid { vd, vm } => {
537                if !ext_state.vector_instructions_allowed() {
538                    ::core::hint::cold_path();
539                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
540                        address: PackedAddress::new(
541                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
542                        ),
543                    });
544                }
545                let Some(vtype) = ext_state.vtype() else {
546                    ::core::hint::cold_path();
547                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
548                        address: PackedAddress::new(
549                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
550                        ),
551                    });
552                };
553                let group_regs = vtype.vlmul().register_count().get();
554                let vd_idx = vd.to_bits();
555                if !vd_idx.is_multiple_of(group_regs) || vd_idx + group_regs > 32 {
556                    ::core::hint::cold_path();
557                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
558                        address: PackedAddress::new(
559                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
560                        ),
561                    });
562                }
563                if !vm && vd == VReg::V0 {
564                    ::core::hint::cold_path();
565                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
566                        address: PackedAddress::new(
567                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
568                        ),
569                    });
570                }
571                let sew = vtype.vsew();
572                // SAFETY: vd alignment checked above; `vm=false` implies `vd != v0` checked above;
573                // `vl <= VLMAX = group_regs * VLEN.bytes() / sew_bytes`, all element indices valid.
574                unsafe {
575                    zvexx_mask_helpers::execute_vid(ext_state, vd, vm, sew);
576                }
577            }
578        }
579
580        ExecutionResult::ContinueNoWrite
581    }
