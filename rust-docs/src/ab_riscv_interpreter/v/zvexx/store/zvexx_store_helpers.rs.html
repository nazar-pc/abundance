11/// Write `eew`-sized data from `buf[..eew.bytes()]` to memory at `addr` (little-endian)
12#[inline(always)]
13#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
14fn write_mem_element<Reg>(
15    memory: &mut impl VirtualMemory,
16    addr: u64,
17    eew: Eew,
18    buf: [u8; const { usize::from(Eew::MAX_BYTES) }],
19) -> Result<(), VirtualMemoryError>
20where
21    Reg: Register,
22{
23    write_bytes::<Reg, _>(
24        memory,
25        addr,
26        buf.get(..usize::from(eew.bytes_width()))
27            .expect("Element width never exceeds `Eew::MAX_BYTES`; qed"),
28    )
29}

31/// Write `src` starting at effective address `address`, wrapping around at the end of the
32/// `XLEN`-bit address space
33#[inline(always)]
34#[doc(hidden)]
35#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
36pub fn write_bytes<Reg, Memory>(
37    memory: &mut Memory,
38    address: u64,
39    src: &[u8],
40) -> Result<(), VirtualMemoryError>
41where
42    Reg: Register,
43    Memory: VirtualMemory,
44{
45    let before_wrap = bytes_before_wrap::<Reg>(address, src.len());
46    let Some((head, tail)) = src.split_at_checked(before_wrap) else {
47        cold_path();
48        return Err(VirtualMemoryError::OutOfBoundsWrite { address });
49    };
50    memory.write_slice(address, head)?;
51    if !tail.is_empty() {
52        cold_path();
53        memory.write_slice(0, tail)?;
54    }
55    Ok(())
56}

58/// Execute a unit-stride or unit-stride segment store.
59///
60/// Segment stride between elements is `nf * eew.bytes()`. Field `f` for element `i` is at
61/// `base + i * nf * eew.bytes() + f * eew.bytes()`. When `nf == 1` this degenerates to a
62/// plain unit-stride store.
63///
64/// Elements `vstart..vl` of `vs3` are stored, where `vl` is the architectural one for regular
65/// stores.
66#[inline(always)]
67#[doc(hidden)]
68#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
69pub fn execute_unit_stride_store<Reg, Env, Memory>(
70    env: &mut Env,
71    memory: &mut Memory,
72    vs3: VRegSegmentGroup<{ Env::VLEN }>,
73    vm: bool,
74    base: u64,
75) -> Result<(), ExecutionError<Reg::Type>>
76where
77    Reg: Register,
78    Env: VectorRegistersExt<Reg>,
79    [(); SUPPORTED_ELEN_VLEN::<{ Env::ELEN }, { Env::VLEN }>]:,
80    Memory: VirtualMemory,
81{
82    let vl = vs3.first().vl();
83    let vstart = env.vstart();
84    let eew = vs3.first().eew();
85    let elem_bytes = eew.bytes_width();
86    let nf = vs3.nf();
87
88    // Unmasked non-segment store is a plain copy of the contiguous element range of the register
89    // group into a contiguous memory range, as long as the whole range is writable. If it is not,
90    // the element-wise path below is what determines the faulting element and writes everything
91    // before it, which is the same outcome a partially written copy would have.
92    if vm
93        && nf.fields_per_segment() == 1
94        && let Some(count) = u32::from(vl.get()).checked_sub(u32::from(u16::from(vstart)))
95        && count > 0
96    {
97        let first = u16::from(vstart);
98        let addr = effective_address::<Reg>(base, u64::from(first) * u64::from(elem_bytes));
99        // A range that wraps around the end of the address space is not contiguous in memory
100        if let Some(data) = env.read_vregs().elements_bytes(vs3.first(), first, count)
101            && !access_wraps::<Reg>(addr, data.len())
102            && memory.write_slice(addr, data).is_ok()
103        {
104            env.reset_vstart();
105            return Ok(());
106        }
107        cold_path();
108    }
109
110    let segment_stride = u64::from(nf.fields_per_segment() * elem_bytes);
111    let mask_buf = snapshot_mask(env.read_vregs(), vm);
112    for element in vs3.elements(u16::from(vstart)) {
113        let i = element.index();
114        if !vm && !mask_bit(&mask_buf, i) {
115            continue;
116        }
117        let elem_base = effective_address::<Reg>(base, u64::from(i) * segment_stride);
118        let fields = element.read_fields(env.read_vregs());
119        for (f, field) in (0..nf.fields_per_segment()).zip(fields) {
120            let addr = effective_address::<Reg>(elem_base, u64::from(f * elem_bytes));
121            let data = field.to_le_bytes();
122            // Record the current element index in `vstart` so that, on a memory fault, the failing
123            // element can be identified and the operation can be restarted
124            if let Err(error) = write_mem_element::<Reg>(memory, addr, eew, data) {
125                cold_path();
126                env.set_vstart(Vstart::from(i));
127                return Err(ExecutionError::from(error));
128            }
129        }
130    }
131    env.reset_vstart();
132    Ok(())
133}

135/// Execute a strided or strided-segment store.
136///
137///   `base + i * stride + f * eew.bytes()`, wrapping around modulo `2^XLEN`
138///   `effective_address::<Reg>(base, i.wrapping_mul(stride) as u64).wrapping_add(f * eew.bytes())`
139///
140/// `stride` is the raw XLEN register value reinterpreted as a signed integer, matching the RVV
141/// specification where the stride operand is a two's-complement signed offset.
142#[inline(always)]
143#[doc(hidden)]
144#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
145pub fn execute_strided_store<Reg, Env, Memory>(
146    env: &mut Env,
147    memory: &mut Memory,
148    vs3: VRegSegmentGroup<{ Env::VLEN }>,
149    vm: bool,
150    base: u64,
151    stride: i64,
152) -> Result<(), ExecutionError<Reg::Type>>
153where
154    Reg: Register,
155    Env: VectorRegistersExt<Reg>,
156    [(); SUPPORTED_ELEN_VLEN::<{ Env::ELEN }, { Env::VLEN }>]:,
157    Memory: VirtualMemory,
158{
159    let vstart = env.vstart();
160    let eew = vs3.first().eew();
161    let elem_bytes = eew.bytes_width();
162    let mask_buf = snapshot_mask(env.read_vregs(), vm);
163    for element in vs3.elements(u16::from(vstart)) {
164        let i = element.index();
165        if !vm && !mask_bit(&mask_buf, i) {
166            continue;
167        }
168        let elem_base =
169            effective_address::<Reg>(base, i64::from(i).wrapping_mul(stride).cast_unsigned());
170        let fields = element.read_fields(env.read_vregs());
171        for (f, field) in (0..vs3.nf().fields_per_segment()).zip(fields) {
172            let addr = effective_address::<Reg>(elem_base, u64::from(f * elem_bytes));
173            let data = field.to_le_bytes();
174            // Record the current element index in `vstart` so that, on a memory fault, the failing
175            // element can be identified and the operation can be restarted
176            if let Err(error) = write_mem_element::<Reg>(memory, addr, eew, data) {
177                cold_path();
178                env.set_vstart(Vstart::from(i));
179                return Err(ExecutionError::from(error));
180            }
181        }
182    }
183    env.reset_vstart();
184    Ok(())
185}

187/// Execute an indexed (unordered or ordered) store or indexed-segment store.
188///
189/// The effective address of element `i`, field `f` is:
190///   `base + index[i] + f * eew.bytes()`
191/// where `index[i]` is element `i` of the index register group `vs2`, interpreted as an
192/// unsigned integer of its element width.
193///
194/// `vs2` must have the same `vl` as `vs3`, which holds for groups created from the same
195/// configuration, nothing is stored otherwise.
196#[inline(always)]
197#[doc(hidden)]
198#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
199pub fn execute_indexed_store<Reg, Env, Memory>(
200    env: &mut Env,
201    memory: &mut Memory,
202    vs3: VRegSegmentGroup<{ Env::VLEN }>,
203    vs2: VRegGroup<{ Env::VLEN }>,
204    vm: bool,
205    base: u64,
206) -> Result<(), ExecutionError<Reg::Type>>
207where
208    Reg: Register,
209    Env: VectorRegistersExt<Reg>,
210    [(); SUPPORTED_ELEN_VLEN::<{ Env::ELEN }, { Env::VLEN }>]:,
211    Memory: VirtualMemory,
212{
213    let Some(indexed) = vs3.with_index(vs2) else {
214        cold_path();
215        return Ok(());
216    };
217    let vstart = env.vstart();
218    let data_eew = vs3.first().eew();
219    let data_elem_bytes = data_eew.bytes_width();
220    let mask_buf = snapshot_mask(env.read_vregs(), vm);
221    for indexed in indexed.elements(u16::from(vstart)) {
222        let element = indexed.element();
223        let i = element.index();
224        if !vm && !mask_bit(&mask_buf, i) {
225            continue;
226        }
227        let offset = indexed.read_index(env.read_vregs());
228        let elem_base = effective_address::<Reg>(base, offset);
229        let fields = element.read_fields(env.read_vregs());
230        for (f, field) in (0..vs3.nf().fields_per_segment()).zip(fields) {
231            let addr =
232                effective_address::<Reg>(elem_base, u64::from(f) * u64::from(data_elem_bytes));
233            let data = field.to_le_bytes();
234            // Record the current element index in `vstart` so that, on a memory fault, the failing
235            // element can be identified and the operation can be restarted
236            if let Err(error) = write_mem_element::<Reg>(memory, addr, data_eew, data) {
237                cold_path();
238                env.set_vstart(Vstart::from(i));
239                return Err(ExecutionError::from(error));
240            }
241        }
242    }
243    env.reset_vstart();
244    Ok(())
