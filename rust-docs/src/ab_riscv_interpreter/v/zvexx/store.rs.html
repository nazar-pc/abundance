30#[instruction_execution]
31impl<Reg, Regs, ExtState, Memory, PC, InstructionHandler, CustomError>
32    ExecutableInstruction<Regs, ExtState, Memory, PC, InstructionHandler, CustomError>
33    for ZveXxStoreInstruction<Reg>
34where
35    Reg: Register,
36    Regs: RegisterFile<Reg>,
37    ExtState: VectorRegistersExt<Reg, CustomError>,
38    [(); SUPPORTED_ELEN_VLEN::<{ ExtState::ELEN }, { ExtState::VLEN }>]:,
39    Memory: VirtualMemory,
40    PC: ProgramCounter<Reg::Type, Memory, CustomError>,
41{
42    #[inline(always)]
43    #[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
44    fn execute(
45        self,
46        Rs1Rs2OperandValues {
47            rs1_value,
48            rs2_value,
49        }: Rs1Rs2OperandValues<<Self::Reg as Register>::Type>,
50        _regs: &mut Regs,
51        ext_state: &mut ExtState,
52        memory: &mut Memory,
53        program_counter: &mut PC,
54        _system_instruction_handler: &mut InstructionHandler,
55    ) -> ExecutionResult<Self::Reg, CustomError> {
56        match self {
57            // Whole-register store: stores `nreg` consecutive registers starting at `vs3` directly
58            // to memory as a flat byte array of `EVL = nreg * VLEN.bytes()` bytes. `vs3` must be
59            // aligned to `nreg`. Ignores vtype, vl, masking. Honors `vstart` in byte
60            // units: the first `vstart` bytes are skipped. If `vstart >= EVL`, the
61            // instruction is a no-op.
62            Self::Vsr { vs3, rs1: _, nreg } => {
63                let nreg = nreg.num_registers();
64                if !ext_state.vector_instructions_allowed() {
65                    ::core::hint::cold_path();
66                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
67                        address: PackedAddress::new(
68                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
69                        ),
70                    });
71                }
72                if !vs3.to_bits().is_multiple_of(nreg) {
73                    ::core::hint::cold_path();
74                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
75                        address: PackedAddress::new(
76                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
77                        ),
78                    });
79                }
80                let vlenb = u64::from(ExtState::VLEN.bytes());
81                let evl = u64::from(nreg) * vlenb;
82                let vstart = ext_state.vstart();
83                if u64::from(u16::from(vstart)) < evl {
84                    let base = rs1_value.as_u64();
85                    let mut byte_off = u64::from(u16::from(vstart));
86                    while byte_off < evl {
87                        let reg_off = byte_off / vlenb;
88                        let in_reg = (byte_off % vlenb) as usize;
89                        // SAFETY: the decoder guarantees `nreg` in {1,2,4,8} and `vs3` is
90                        // `nreg`-aligned (checked above), so `vs3.to_bits() + nreg - 1 <= 31`
91                        let reg = unsafe {
92                            VReg::from_bits(vs3.to_bits() + reg_off as u8).unwrap_unchecked()
93                        };
94                        // SAFETY: `in_reg < VLEN.bytes()` by construction
95                        let src =
96                            unsafe { ext_state.read_vregs().get(reg).get_unchecked(in_reg..) };
97                        if let Err(error) = memory.write_slice(base + byte_off, src) {
98                            ext_state.set_vstart(Vstart::from(byte_off as u16));
99                            return ExecutionResult::Err(ExecutionError::from(error));
100                        }
101                        byte_off += src.len() as u64;
102                    }
103                }
104                ext_state.reset_vstart();
105            }
106            // Mask store: stores `ceil(vl / 8)` bytes from `vs3` to memory with no masking.
107            // Does not require a valid vtype: when vill is set vl is 0, so zero bytes are written.
108            // Honors `vstart` at byte granularity: the first `vstart / 8` bytes are skipped.
109            Self::Vsm { vs3, rs1: _ } => {
110                if !ext_state.vector_instructions_allowed() {
111                    ::core::hint::cold_path();
112                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
113                        address: PackedAddress::new(
114                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
115                        ),
116                    });
117                }
118                let vl = ext_state.vl();
119                let evl_bytes = vl.bytes();
120                let start_byte = ext_state.vstart();
121                if u16::from(start_byte) < evl_bytes {
122                    let base = rs1_value.as_u64();
123                    // SAFETY: `evl_bytes = vl.div_ceil(8) <= VLEN / 8 = VLEN.bytes()` because
124                    // `vl <= VLMAX <= VLEN`, so the slice `start_byte..evl_bytes` is in bounds of
125                    // the `VLEN.bytes()`-byte source register
126                    let src = unsafe {
127                        ext_state.read_vregs().get(vs3).get_unchecked(
128                            usize::from(u16::from(start_byte))..usize::from(evl_bytes),
129                        )
130                    };
131                    memory
132                        .write_slice(base + u64::from(u16::from(start_byte)), src)
133                        .map_err(ExecutionError::from)?;
134                }
135                ext_state.reset_vstart();
136            }
137            // Unit-stride store.
138            //
139            // Source EMUL = EEW/SEW * LMUL, computed via `data_register_count`. This gives
140            // `group_regs` such that `VLMAX = group_regs * VLEN.bytes() / eew.bytes()` matches the
141            // architectural `vl`.
142            Self::Vse {
143                vs3,
144                rs1: _,
145                vm,
146                eew,
147            } => {
148                if !ext_state.vector_instructions_allowed() {
149                    ::core::hint::cold_path();
150                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
151                        address: PackedAddress::new(
152                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
153                        ),
154                    });
155                }
156                let Some(vtype) = ext_state.vtype() else {
157                    ::core::hint::cold_path();
158                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
159                        address: PackedAddress::new(
160                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
161                        ),
162                    });
163                };
164                let group_regs = vtype.vlmul().data_register_count(eew, vtype.vsew()).ok_or(
165                    ExecutionError::IllegalInstruction {
166                        address: PackedAddress::new(
167                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
168                        ),
169                    },
170                )?;
171                zvexx_load_helpers::check_register_group_alignment::<Reg, _, _, _>(
172                    program_counter,
173                    vs3,
174                    group_regs,
175                )?;
176                // SAFETY:
177                // - alignment: `check_register_group_alignment` verified `vs3 % group_regs == 0`
178                //   and `vs3 + group_regs <= 32`
179                // - `vl <= group_regs * VLEN.bytes() / eew.bytes()`: `group_regs` is the EMUL
180                //   computed for this `eew` and `vtype`, so this VLMAX equals the architectural
181                //   VLMAX that bounds `vl`
182                // - vs3/v0 overlap: stores read vs3 as a source; the spec does not restrict
183                //   source/v0 overlap
184                unsafe {
185                    zvexx_store_helpers::execute_unit_stride_store(
186                        ext_state,
187                        memory,
188                        vs3,
189                        vm,
190                        rs1_value.as_u64(),
191                        eew,
192                        group_regs,
193                        Nf::N1,
194                    )?;
195                }
196            }
197            // Strided store
198            Self::Vsse {
199                vs3,
200                rs1: _,
201                rs2: _,
202                vm,
203                eew,
204            } => {
205                if !ext_state.vector_instructions_allowed() {
206                    ::core::hint::cold_path();
207                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
208                        address: PackedAddress::new(
209                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
210                        ),
211                    });
212                }
213                let Some(vtype) = ext_state.vtype() else {
214                    ::core::hint::cold_path();
215                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
216                        address: PackedAddress::new(
217                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
218                        ),
219                    });
220                };
221                let group_regs = vtype.vlmul().data_register_count(eew, vtype.vsew()).ok_or(
222                    ExecutionError::IllegalInstruction {
223                        address: PackedAddress::new(
224                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
225                        ),
226                    },
227                )?;
228                zvexx_load_helpers::check_register_group_alignment::<Reg, _, _, _>(
229                    program_counter,
230                    vs3,
231                    group_regs,
232                )?;
233                let stride = rs2_value.as_i64();
234                // SAFETY: same preconditions as `Vse`.
235                unsafe {
236                    zvexx_store_helpers::execute_strided_store(
237                        ext_state,
238                        memory,
239                        vs3,
240                        vm,
241                        rs1_value.as_u64(),
242                        stride,
243                        eew,
244                        group_regs,
245                        Nf::N1,
246                    )?;
247                }
248            }
249            // Indexed-unordered store. Ordering between elements is not guaranteed.
250            Self::Vsuxei {
251                vs3,
252                rs1: _,
253                vs2,
254                vm,
255                eew: index_eew,
256            } => {
257                if !ext_state.vector_instructions_allowed() {
258                    ::core::hint::cold_path();
259                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
260                        address: PackedAddress::new(
261                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
262                        ),
263                    });
264                }
265                let Some(vtype) = ext_state.vtype() else {
266                    ::core::hint::cold_path();
267                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
268                        address: PackedAddress::new(
269                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
270                        ),
271                    });
272                };
273                let data_eew = vtype.vsew().as_eew();
274                let data_group_regs = vtype.vlmul().register_count();
275                let index_group_regs = vtype
276                    .vlmul()
277                    .index_register_count(index_eew, vtype.vsew())
278                    .ok_or(ExecutionError::IllegalInstruction {
279                        address: PackedAddress::new(
280                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
281                        ),
282                    })?;
283                zvexx_load_helpers::check_register_group_alignment::<Reg, _, _, _>(
284                    program_counter,
285                    vs3,
286                    data_group_regs,
287                )?;
288                zvexx_load_helpers::check_register_group_alignment::<Reg, _, _, _>(
289                    program_counter,
290                    vs2,
291                    index_group_regs,
292                )?;
293                // SAFETY:
294                // - `vs3` alignment/bounds: `check_register_group_alignment` verified both
295                // - `vs2` alignment/bounds: `check_register_group_alignment` verified both
296                // - `vl <= data_group_regs * VLEN.bytes() / data_eew.bytes()`: `data_group_regs` is
297                //   the EMUL that bounds `vl`
298                // - `vl <= index_group_regs * VLEN.bytes() / index_eew.bytes()`:
299                //   `index_register_count` returns the EMUL for the index group, which by the same
300                //   argument bounds `vl`
301                // - vs3/v0 overlap: stores read vs3 as a source; no restriction
302                unsafe {
303                    zvexx_store_helpers::execute_indexed_store(
304                        ext_state,
305                        memory,
306                        vs3,
307                        vs2,
308                        vm,
309                        rs1_value.as_u64(),
310                        data_eew,
311                        index_eew,
312                        data_group_regs,
313                        Nf::N1,
314                    )?;
315                }
316            }
317            // Indexed-ordered store. Elements must be written in element order.
318            // The ordering constraint is visible only to other harts/devices; the implementation
319            // here is already sequential, so no additional logic is needed.
320            Self::Vsoxei {
321                vs3,
322                rs1: _,
323                vs2,
324                vm,

333                    });
334                }
335                let Some(vtype) = ext_state.vtype() else {
336                    ::core::hint::cold_path();

342                };

344                let data_group_regs = vtype.vlmul().register_count();
345                let index_group_regs = vtype
346                    .vlmul()
347                    .index_register_count(index_eew, vtype.vsew())
348                    .ok_or(ExecutionError::IllegalInstruction {
349                        address: PackedAddress::new(
350                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),

355                    vs3,
356                    data_group_regs,
357                )?;
358                zvexx_load_helpers::check_register_group_alignment::<Reg, _, _, _>(
359                    program_counter,
360                    vs2,
361                    index_group_regs,
362                )?;
363                // SAFETY: identical precondition argument to `Vsuxei`
364                unsafe {
365                    zvexx_store_helpers::execute_indexed_store(
366                        ext_state,
367                        memory,
368                        vs3,
369                        vs2,
370                        vm,
371                        rs1_value.as_u64(),
372                        data_eew,
373                        index_eew,
374                        data_group_regs,
375                        Nf::N1,
376                    )?;
377                }
378            }
379            // Unit-stride segment store: `nf` fields per element, stored contiguously
380            Self::Vsseg {
381                vs3,
382                rs1: _,
383                eew,
384                vm_nf,
385            } => {
386                let vm = vm_nf.vm();
387                let nf = vm_nf.nf();
388                if !ext_state.vector_instructions_allowed() {
389                    ::core::hint::cold_path();
390                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
391                        address: PackedAddress::new(
392                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
393                        ),
394                    });
395                }
396                let Some(vtype) = ext_state.vtype() else {
397                    ::core::hint::cold_path();
398                    return ExecutionResult::Err(ExecutionError::IllegalInstruction {
399                        address: PackedAddress::new(
400                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),
401                        ),
402                    });
403                };
404                let group_regs = vtype.vlmul().data_register_count(eew, vtype.vsew()).ok_or(
405                    ExecutionError::IllegalInstruction {
406                        address: PackedAddress::new(
407                            program_counter.old_pc(zvexx_helpers::INSTRUCTION_SIZE),

410                )?;
411                zvexx_store_helpers::validate_segment_store_registers::<Reg, _, _, _>(
412                    program_counter,
413                    vs3,
414                    group_regs,
415                    nf,
416                )?;
417                // SAFETY:
418                // - `validate_segment_store_registers` guarantees `vs3 % group_regs == 0` and `vs3
419                //   + nf * group_regs <= 32`
420                // - `vl <= group_regs * VLEN.bytes() / eew.bytes()`: same EMUL argument as `Vse`
421                // - vs3/v0 overlap: stores read vs3 as a source; no restriction
422                unsafe {
423                    zvexx_store_helpers::execute_unit_stride_store(
424                        ext_state,
425                        memory,
426                        vs3,
427                        vm,
428                        rs1_value.as_u64(),
429                        eew,
430                        group_regs,
431                        nf,
432                    )?;
433                }
434            }
435            // Strided segment store
436            Self::Vssseg {
437                vs3,
438                rs1: _,
439                rs2: _,
440                eew,
