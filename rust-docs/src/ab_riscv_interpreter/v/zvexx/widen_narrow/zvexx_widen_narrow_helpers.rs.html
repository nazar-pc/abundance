12/// Check that a widening destination `vd` is aligned to `wide_group_regs` and fits within
13/// `[0,32)`, without any source overlap check
14#[inline(always)]
15#[doc(hidden)]
16#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
17pub fn check_vd_widen_no_src_check<Reg, Memory, PC>(
18    program_counter: &PC,
19    vd: VReg,
20    wide_group_regs: NonZeroU8,
21) -> Result<(), ExecutionError<Reg::Type>>
22where
23    Reg: Register,
24    PC: ProgramCounter<Reg::Type, Memory>,
25{
26    let wide_group_regs = wide_group_regs.get();
27    let vd_idx = vd.to_bits();
28    if !vd_idx.is_multiple_of(wide_group_regs) || vd_idx + wide_group_regs > 32 {
29        cold_path();
30        return Err(ExecutionError::IllegalInstruction {
31            address: PackedAddress::new(program_counter.old_pc(INSTRUCTION_SIZE)),
32        });
33    }
34    Ok(())
35}

37/// Check that an extension source `vs2` is aligned to `src_group_regs`, fits in `[0,32)`, and only
38/// overlaps `vd` (which occupies `group_regs` registers) in a manner permitted by the spec.
39///
40/// Per the vector spec §5.2, the destination EEW (SEW) of an extension is greater than the source
41/// EEW (SEW/factor), so the destination may overlap the source only when the source EMUL is at
42/// least 1 and the overlap is in the highest-numbered part of the destination register group (e.g.
43/// `vzext.vf4 v0, v6` with LMUL=8, where the narrow source `{v6,v7}` aliases the high registers of
44/// the wide `{v0..v7}` destination). Any other overlap is illegal.
45#[inline(always)]
46#[doc(hidden)]
47#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
48pub fn check_vs_ext_alignment<Reg, Memory, PC>(
49    program_counter: &PC,
50    vs2: VReg,
51    src_group_regs: NonZeroU8,
52    vd: VReg,
53    group_regs: NonZeroU8,
54) -> Result<(), ExecutionError<Reg::Type>>
55where
56    Reg: Register,
57    PC: ProgramCounter<Reg::Type, Memory>,
58{
59    let src_group_regs = src_group_regs.get();
60    let group_regs = group_regs.get();
61    let vs2_idx = vs2.to_bits();
62    if !vs2_idx.is_multiple_of(src_group_regs) || vs2_idx + src_group_regs > 32 {
63        cold_path();
64        return Err(ExecutionError::IllegalInstruction {
65            address: PackedAddress::new(program_counter.old_pc(INSTRUCTION_SIZE)),
66        });
67    }
68    // The wide destination (group_regs) may overlap the narrow source (src_group_regs) only in the
69    // highest-numbered part of the destination group, and only when the source EMUL >= 1.
70    if widen_src_overlap_illegal(vd.to_bits(), group_regs, vs2_idx, src_group_regs) {
71        cold_path();
72        return Err(ExecutionError::IllegalInstruction {
73            address: PackedAddress::new(program_counter.old_pc(INSTRUCTION_SIZE)),
74        });
75    }
76    Ok(())
77}

79/// Check that a widening destination `vd` is aligned to `wide_group_regs`, fits within `[0, 32)`,
80/// and only overlaps the `group_regs`-register narrow source(s) starting at `vs_a`/`vs_b` in a
81/// manner permitted by the spec.
82///
83/// `wide_group_regs` is the pre-computed register count for the wide EMUL (2*LMUL), obtained via
84/// `Vlmul::index_register_count(wide_eew, sew)`. `group_regs` is the narrow LMUL register count.
85///
86/// Per the vector spec §5.2, a destination whose EEW (2*SEW) is greater than a source's EEW (SEW)
87/// may overlap that source only when the source EMUL is at least 1 and the overlap is in the
88/// highest-numbered part of the destination register group (e.g. `vwsubu.wv v2, v14, v3` with
89/// LMUL=1, where the narrow `v3` aliases the high register of the wide `{v2, v3}` destination).
90/// Any other overlap is illegal.
91#[inline(always)]
92#[doc(hidden)]
93#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
94pub fn check_vd_widen_alignment<Reg, Memory, PC>(
95    program_counter: &PC,
96    vd: VReg,
97    vs_a: VReg,
98    vs_b_opt: Option<VReg>,
99    group_regs: NonZeroU8,
100    wide_group_regs: NonZeroU8,
101) -> Result<(), ExecutionError<Reg::Type>>
102where
103    Reg: Register,
104    PC: ProgramCounter<Reg::Type, Memory>,
105{
106    let wide_group_regs = wide_group_regs.get();
107    let group_regs = group_regs.get();
108    let vd_idx = vd.to_bits();
109    if !vd_idx.is_multiple_of(wide_group_regs) || vd_idx + wide_group_regs > 32 {
110        cold_path();
111        return Err(ExecutionError::IllegalInstruction {
112            address: PackedAddress::new(program_counter.old_pc(INSTRUCTION_SIZE)),
113        });
114    }
115    if widen_src_overlap_illegal(vd_idx, wide_group_regs, vs_a.to_bits(), group_regs) {
116        cold_path();
117        return Err(ExecutionError::IllegalInstruction {
118            address: PackedAddress::new(program_counter.old_pc(INSTRUCTION_SIZE)),
119        });
120    }
121    if let Some(vs_b) = vs_b_opt
122        && widen_src_overlap_illegal(vd_idx, wide_group_regs, vs_b.to_bits(), group_regs)
123    {
124        cold_path();
125        return Err(ExecutionError::IllegalInstruction {
126            address: PackedAddress::new(program_counter.old_pc(INSTRUCTION_SIZE)),
127        });
128    }
129    Ok(())
130}

132/// Returns `true` when a narrow source group of `group_regs` registers starting at `vs_idx`
133/// overlaps the wide destination group (`wide_group_regs` registers starting at `vd_idx`) in a way
134/// that is *not* permitted by the spec.
135///
136/// Overlap is only legal when the source EMUL is at least 1 - which, on widening, is exactly when
137/// the destination register count strictly exceeds the narrow source count (for fractional LMUL
138/// both counts collapse to 1) - and the source occupies the highest-numbered registers of the
139/// destination group.
140#[inline(always)]
141#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
142fn widen_src_overlap_illegal(vd_idx: u8, wide_group_regs: u8, vs_idx: u8, group_regs: u8) -> bool {
143    if !ranges_overlap(vd_idx, wide_group_regs, vs_idx, group_regs) {
144        return false;
145    }
146    let high_part_overlap =
147        wide_group_regs > group_regs && vs_idx == vd_idx + wide_group_regs - group_regs;
148    !high_part_overlap
149}

176/// Check that a narrowing destination `vd` is aligned to `group_regs`, fits within `[0, 32)`, and
177/// only overlaps the wide source `vs2` (which occupies `wide_group_regs` registers starting at
178/// `vs2`) in a manner permitted by the spec.
179///
180/// Per spec §11.7, narrowing instructions permit `vd` to alias the *low* half of the wide `vs2`
181/// register group (i.e. `vd == vs2`) - any other overlap (e.g. `vd` aliasing only the high part of
182/// `vs2`'s group) is illegal.
183#[inline(always)]
184#[doc(hidden)]
185#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
186pub fn check_vd_narrow_alignment<Reg, Memory, PC>(
187    program_counter: &PC,
188    vd: VReg,
189    group_regs: NonZeroU8,
190    vs2: VReg,
191    wide_group_regs: NonZeroU8,
192) -> Result<(), ExecutionError<Reg::Type>>
193where
194    Reg: Register,
195    PC: ProgramCounter<Reg::Type, Memory>,
196{
197    let group_regs = group_regs.get();
198    let vd_idx = vd.to_bits();
199    let vs2_idx = vs2.to_bits();
200    let overlaps = ranges_overlap(vd_idx, group_regs, vs2_idx, wide_group_regs.get());
201    if !vd_idx.is_multiple_of(group_regs)
202        || vd_idx + group_regs > 32
203        || (overlaps && vd_idx != vs2_idx)
204    {
205        cold_path();
206        return Err(ExecutionError::IllegalInstruction {
207            address: PackedAddress::new(program_counter.old_pc(INSTRUCTION_SIZE)),
208        });
209    }
210    Ok(())
211}

213/// Returns `true` when `[a_start, a_start+a_len)` overlaps `[b_start, b_start+b_len)`.
214#[inline(always)]
215#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
216fn ranges_overlap(a_start: u8, a_len: u8, b_start: u8, b_len: u8) -> bool {
217    a_start < b_start + b_len && b_start < a_start + a_len
218}

220/// Return whether mask bit `i` is set in the mask byte slice (LSB-first within each byte).
221#[inline(always)]
222#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
223fn mask_bit(mask: &[u8], i: u16) -> bool {
224    mask.get(usize::from(i / u8::BITS as u16))
225        .is_some_and(|b| (b >> (i % u8::BITS as u16)) & 1 != 0)
226}

228/// Snapshot the mask register into a stack buffer.
229///
230/// When `vm=true` (unmasked), all bytes are `0xff`.
231///
232/// # Safety
233/// `vl <= VLEN` must hold
234#[inline(always)]
235#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
236unsafe fn snapshot_mask<const VLEN: Vlen>(
237    vregs: &VectorRegisterFile<VLEN>,
238    vm: bool,
239    vl: Vl,
240) -> [u8; VLENB_USIZE::<VLEN>] {
241    let mut buf = [0u8; _];
242    if vm {
243        buf = [0xffu8; _];
244    } else {
245        let mask_bytes = usize::from(vl.bytes());
246        // SAFETY: `mask_bytes <= VLEN.bytes()` by precondition
247        unsafe {
248            buf.get_unchecked_mut(..mask_bytes)
249                .copy_from_slice(vregs.get(VReg::V0).get_unchecked(..mask_bytes));
250        }
251    }
252    buf
253}

255/// Read the low `sew.bytes_width()` of the element `elem_i` from the register group `base_reg`,
256/// zero-extended to `u64`.
257///
258/// # Safety
259/// `base_reg + elem_i / (VLEN.bytes() / sew.bytes_width()) < 32`
260#[inline(always)]
261#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
262unsafe fn read_element_u64<const VLEN: Vlen>(
263    vregs: &VectorRegisterFile<VLEN>,
264    base_reg: VReg,
265    elem_i: u16,
266    sew: Vsew,
267) -> u64 {
268    let sew_bytes = u32::from(sew.bytes_width());
269    let elems_per_reg = VLEN.bytes() / sew_bytes;
270    let reg_off = u32::from(elem_i) / elems_per_reg;
271    let byte_off = (u32::from(elem_i) % elems_per_reg) * sew_bytes;
272    // SAFETY: `base_reg + reg_off < 32` by caller's precondition
273    let reg = unsafe {
274        vregs.get(VReg::from_bits(base_reg.to_bits() + reg_off as u8).unwrap_unchecked())
275    };
276    // SAFETY: `byte_off + sew_bytes <= VLEN.bytes()`
277    let src = unsafe { reg.get_unchecked(byte_off as usize..(byte_off + sew_bytes) as usize) };
278    let mut buf = [0u8; 8];
279    // SAFETY: `sew_bytes <= 8`
280    unsafe { buf.get_unchecked_mut(..sew_bytes as usize) }.copy_from_slice(src);
281    u64::from_le_bytes(buf)
282}

284/// Write the low `sew.bytes_width()` of `value` into element `elem_i` in register group `base_reg`.
285///
286/// # Safety
287/// `base_reg + elem_i / (VLEN.bytes() / sew.bytes_width()) < 32`
288#[inline(always)]
289#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
290unsafe fn write_element_u64<const VLEN: Vlen>(
291    vregs: &mut VectorRegisterFile<VLEN>,
292    base_reg: VReg,
293    elem_i: u16,
294    sew: Vsew,
295    value: u64,
296) {
297    let sew_bytes = u32::from(sew.bytes_width());
298    let elems_per_reg = VLEN.bytes() / sew_bytes;
299    let reg_off = u32::from(elem_i) / elems_per_reg;
300    let byte_off = (u32::from(elem_i) % elems_per_reg) * sew_bytes;
301    let buf = value.to_le_bytes();
302    // SAFETY: `base_reg + reg_off < 32` by caller's precondition
303    let reg = unsafe {
304        vregs.get_mut(VReg::from_bits(base_reg.to_bits() + reg_off as u8).unwrap_unchecked())
305    };
306    // SAFETY: `byte_off + sew_bytes <= VLEN.bytes()`
307    let dst = unsafe { reg.get_unchecked_mut(byte_off as usize..(byte_off + sew_bytes) as usize) };
308    // SAFETY: `sew_bytes <= 8`
309    dst.copy_from_slice(unsafe { buf.get_unchecked(..sew_bytes as usize) });
310}

312/// Sign-extend the low `sew.bits_width()` of `val` to `i64`.
313#[inline(always)]
314#[doc(hidden)]
315#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
316pub fn sign_extend_bits(val: u64, sew: Vsew) -> i64 {
317    let shift = u64::BITS - u32::from(sew.bits_width());
318    (val.cast_signed() << shift) >> shift
319}

321/// Interpret a scalar operand as an unsigned SEW-wide value.
322///
323/// RVV widening scalar instructions (.vx/.wx) conceptually use a scalar
324/// operand whose width matches the current SEW, not the full XLEN width.
325///
326/// For example on RV64:
327///
328/// SEW=8:
329///     val = 0x0000_0000_0000_01ff
330///     result = 0x0000_0000_0000_00ff
331///
332/// SEW=16:
333///     val = 0x0000_0000_0000_01ff
334///     result = 0x0000_0000_0000_01ff
335///
336/// SEW=32:
337///     val = 0xffff_ffff_1234_5678
338///     result = 0x0000_0000_1234_5678
339///
340/// This helper performs that SEW-width truncation without sign extension.
341#[inline(always)]
342#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
343fn scalar_unsigned_for_sew(val: u64, sew: Vsew) -> u64 {
344    val & (u64::MAX >> (u64::BITS - u32::from(sew.bits_width())))
345}

347/// Interpret a scalar operand as a signed SEW-wide value.
348///
349/// The scalar is first truncated to SEW bits, then sign-extended back to
350/// 64 bits.
351///
352/// For example on RV64:
353///
354/// SEW=8:
355///     val = 0x0000_0000_0000_00ff
356///     result = 0xffff_ffff_ffff_ffff (-1)
357///
358/// SEW=8:
359///     val = 0x0000_0000_0000_007f
360///     result = 0x0000_0000_0000_007f (+127)
361///
362/// SEW=16:
363///     val = 0x0000_0000_0000_ffff
364///     result = 0xffff_ffff_ffff_ffff (-1)
365///
366/// This matches the signed widening behavior required by instructions such
367/// as vwadd.vx and vwsub.vx.
368#[inline(always)]
369#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
370fn scalar_signed_for_sew(val: u64, sew: Vsew) -> u64 {
371    sign_extend_bits(val, sew).cast_unsigned()
372}

374/// Execute a widening integer add/subtract.
375///
376/// Each source element is SEW-wide; the destination element is 2×SEW-wide.
377/// `ZERO_EXTEND_AB` selects unsigned or signed widening for sources (unsigned = zero-extend,
378/// signed = sign-extend).
379///
380/// `op` receives `(wide_a: u64, wide_b: u64) -> u64`.
381///
382/// # Safety
383/// - `vd` aligned to `2*group_regs`, fits in `[0,32)`, does not overlap `vs2` or `src` (verified by
384///   caller)
385/// - `vs2` aligned to `group_regs`, fits in `[0,32)` (verified by caller)
386/// - `src` register (when `WidenSrc::Vreg`) aligned to `group_regs`, fits in `[0,32)` (verified by
387///   caller)
388/// - `vl <= group_regs * VLEN.bytes() / sew.bytes_width()` (all elements fit)
389/// - SEW < 64
390/// - When `vm=false`: `vd.to_bits() != 0`
391#[inline(always)]
392#[doc(hidden)]
393#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
394pub unsafe fn execute_widen_op<const ZERO_EXTEND_AB: bool, Reg, Env, F>(
395    env: &mut Env,
396    vd: VReg,
397    vs2: VReg,
398    src: OpSrc,
399    vm: bool,
400    sew: Vsew,
401    op: F,
402) where
403    Reg: Register,
404    Env: VectorRegistersExt<Reg>,
405    [(); SUPPORTED_ELEN_VLEN::<{ Env::ELEN }, { Env::VLEN }>]:,
406    F: Fn(u64, u64) -> u64,
407{
408    let vl = env.vl();
409    let vstart = env.vstart();
410    // SAFETY: Caller guarantees SEW < 64, hence this is always valid
411    let wide_sew = unsafe { sew.double_width().unwrap_unchecked() };
412
413    // SAFETY: `vl <= VLMAX <= VLEN`
414    let mask_buf = unsafe { snapshot_mask(env.read_vregs(), vm, vl) };
415
416    for i in vstart.range_to(vl) {
417        if !mask_bit(&mask_buf, i) {
418            continue;
419        }
420        // SAFETY: `vs2` aligned to `group_regs`;
421        // `i < vl <= group_regs * (VLEN.bytes() / sew.bytes_width())`
422        let raw_a = unsafe { read_element_u64(env.read_vregs(), vs2, i, sew) };
423        let wide_a = if ZERO_EXTEND_AB {
424            raw_a
425        } else {
426            sign_extend_bits(raw_a, sew).cast_unsigned()
427        };
428        let wide_b = match src {
429            OpSrc::Vreg(vs1_base) => {
430                // SAFETY: same argument as vs2
431                let raw_b = unsafe { read_element_u64(env.read_vregs(), vs1_base, i, sew) };
432                if ZERO_EXTEND_AB {
433                    raw_b
434                } else {
435                    sign_extend_bits(raw_b, sew).cast_unsigned()
436                }
437            }
438            OpSrc::Scalar(val) => {
439                if ZERO_EXTEND_AB {
440                    scalar_unsigned_for_sew(val, sew)
441                } else {
442                    scalar_signed_for_sew(val, sew)
443                }
444            }
445        };
446        let result = op(wide_a, wide_b);
447        // SAFETY: `vd` aligned to `2*group_regs`;
448        // `i < vl <= group_regs * (VLEN.bytes() / sew.bytes_width())` so
449        // `i < 2*group_regs * (VLEN.bytes() / wide_sew.bytes_width())` - element fits in the wide
450        // group
451        unsafe {
452            write_element_u64(env.write_vregs(), vd, i, wide_sew, result);
453        }
454    }
455    env.mark_vs_dirty();
456    env.reset_vstart();
457}

459/// Execute a widening add/subtract where `vs2` is already 2×SEW wide.
460///
461/// `vs2` is read at `wide_sew.bytes_width()`; `src` (narrow) is read at `sew.bytes_width()` and
462/// widened. `ZERO_EXTEND_B` selects unsigned vs signed widening for the narrow source operand.
463///
464/// # Safety
465/// - `vd` aligned to `2*group_regs`, fits in `[0,32)`, does not overlap `vs2` or `src`
466/// - `vs2` aligned to `2*group_regs`, fits in `[0,32)` (wide source)
467/// - `src` register (when `WidenSrc::Vreg`) aligned to `group_regs`, fits in `[0,32)`
468/// - `vl <= group_regs * VLEN.bytes() / sew.bytes_width()`
469/// - SEW < 64
470/// - When `vm=false`: `vd.to_bits() != 0`
471#[inline(always)]
472#[doc(hidden)]
473#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
474pub unsafe fn execute_widen_w_op<const ZERO_EXTEND_B: bool, Reg, Env, F>(
475    env: &mut Env,
476    vd: VReg,
477    vs2: VReg,
478    src: OpSrc,
479    vm: bool,
480    sew: Vsew,
481    op: F,
482) where
483    Reg: Register,
484    Env: VectorRegistersExt<Reg>,
485    [(); SUPPORTED_ELEN_VLEN::<{ Env::ELEN }, { Env::VLEN }>]:,
486    F: Fn(u64, u64) -> u64,
487{
488    let vl = env.vl();
489    let vstart = env.vstart();
490    // SAFETY: Caller guarantees SEW < 64, hence this is always valid
491    let wide_sew = unsafe { sew.double_width().unwrap_unchecked() };
492
493    // SAFETY: `vl <= VLEN`
494    let mask_buf = unsafe { snapshot_mask(env.read_vregs(), vm, vl) };
495
496    for i in vstart.range_to(vl) {
497        if !mask_bit(&mask_buf, i) {
498            continue;
499        }
500        // vs2 is already 2×SEW; read at wide width
501        // SAFETY: `vs2` aligned to `2*group_regs`; element `i` fits within it
502        let wide_a = unsafe { read_element_u64(env.read_vregs(), vs2, i, wide_sew) };
503        let wide_b = match src {
504            OpSrc::Vreg(vs1) => {
505                // SAFETY: `vs1` is aligned to `group_regs` and fits within `[0, 32)`,
506                // verified by caller; `i < vl <= group_regs * (VLEN.bytes() / sew.bytes_width())`,
507                // so `vs1_base + i / elems_per_reg < vs1_base + group_regs <= 32`
508                let raw_b = unsafe { read_element_u64(env.read_vregs(), vs1, i, sew) };
509                if ZERO_EXTEND_B {
510                    raw_b
511                } else {
512                    sign_extend_bits(raw_b, sew).cast_unsigned()
513                }
514            }
515            OpSrc::Scalar(val) => {
516                if ZERO_EXTEND_B {
517                    scalar_unsigned_for_sew(val, sew)
518                } else {
519                    scalar_signed_for_sew(val, sew)
520                }
521            }
522        };
523        let result = op(wide_a, wide_b);
524        // SAFETY: same as `execute_widen_op` for vd
525        unsafe {
526            write_element_u64(env.write_vregs(), vd, i, wide_sew, result);
527        }
528    }
529    env.mark_vs_dirty();
530    env.reset_vstart();
531}

533/// Execute a narrowing right-shift.
534///
535/// `vs2` is 2×SEW wide; the shift amount comes from `src` (SEW-wide or scalar).
536/// The shift amount is masked to `log2(2*SEW)` bits per spec §12.6.
537/// `ARITHMETIC` selects sign-extending (true) vs zero-extending (false) before shifting.
538///
539/// # Safety
540/// - `vd` aligned to `group_regs`, fits in `[0,32)`
541/// - `vs2` aligned to `wide_group_regs`, fits in `[0,32)`; aliasing with the low half of `vs2` is
542///   permitted per spec §11.7 - reads complete before writes to any overlapping element since the
543///   destination SEW is half the source SEW
544/// - `src` register (when `OpSrc::Vreg`) aligned to `group_regs`, fits in `[0,32)`
545/// - `vl <= group_regs * VLEN.bytes() / sew.bytes_width()`
546/// - SEW < 64
547/// - When `vm=false`: `vd.to_bits() != 0`
548#[inline(always)]
549#[doc(hidden)]
550#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
551pub unsafe fn execute_narrow_shift<const ARITHMETIC: bool, Reg, Env>(
552    env: &mut Env,
553    vd: VReg,
554    vs2: VReg,
555    src: OpSrc,
556    vm: bool,
557    sew: Vsew,
558) where
559    Reg: Register,
560    Env: VectorRegistersExt<Reg>,
561    [(); SUPPORTED_ELEN_VLEN::<{ Env::ELEN }, { Env::VLEN }>]:,
562{
563    let vl = env.vl();
564    let vstart = env.vstart();
565    // SAFETY: Caller guarantees SEW < 64, hence this is always valid
566    let wide_sew = unsafe { sew.double_width().unwrap_unchecked() };
567    // Shift amount mask: log2(2*SEW) bits = log2(SEW) + 1 bits
568    let shamt_mask = u64::from(wide_sew.bits_width() - 1);
569
570    // SAFETY: `vl <= VLEN`
571    let mask_buf = unsafe { snapshot_mask(env.read_vregs(), vm, vl) };
572
573    for i in vstart.range_to(vl) {
574        if !mask_bit(&mask_buf, i) {
575            continue;
576        }
577        // SAFETY: `vs2` is the wide source group
578        let wide_val = unsafe { read_element_u64(env.read_vregs(), vs2, i, wide_sew) };
579        let shamt = match src {
580            OpSrc::Vreg(vs1_base) => {
581                // SAFETY: `vs1` is aligned to `group_regs` and fits within `[0, 32)`,
582                // verified by caller; `i < vl <= group_regs * (VLEN.bytes() / sew.bytes_width())`,
583                // so `vs1_base + i / elems_per_reg < vs1_base + group_regs <= 32`
584                let raw = unsafe { read_element_u64(env.read_vregs(), vs1_base, i, sew) };
585                raw & shamt_mask
586            }
587            // Scalar shift amount: only the low log2(2*SEW) bits are used per spec
588            OpSrc::Scalar(val) => val & shamt_mask,
589        };
590        let result_wide = if ARITHMETIC {
591            // Sign-extend to i64 first, then shift arithmetically as i64 to
592            // preserve sign bits, then cast back. Shifting u64 after cast_unsigned()
593            // would be a logical shift and lose sign bits.
594            (sign_extend_bits(wide_val, wide_sew) >> shamt).cast_unsigned()
595        } else {
596            wide_val >> shamt
597        };
598        // Truncate to SEW bits
599        let result = result_wide & ((1u64 << sew.bits_width()) - 1);
600        // SAFETY: `vd` is the narrow destination group
601        unsafe {
602            write_element_u64(env.write_vregs(), vd, i, sew, result);
603        }
604    }
605    env.mark_vs_dirty();
606    env.reset_vstart();
607}

609/// Execute an integer extension (vzext/vsext).
610///
611/// Source element width is `sew.divide_by_factor(factor).bytes_width()`; destination is
612/// `sew.bytes_width()`. `SIGN` selects sign- or zero-extension.
613///
614/// The source EMUL = LMUL / factor; the source register group is `max(1, group_regs / factor)`
615/// registers.
616///
617/// # Safety
618/// - `vd` aligned to `group_regs`, fits in `[0,32)`
619/// - `vs2` aligned to `src_group_regs`, fits in `[0,32)`, does not overlap `vd`
620/// - `vl <= group_regs * VLEN.bytes() / sew.bytes_width()`
621/// - `sew.divide_by_factor(factor).is_some()`
622/// - When `vm=false`: `vd.to_bits() != 0`
623#[inline(always)]
624#[doc(hidden)]
625#[cfg_attr(feature = "no-panic", no_panic_const::no_panic)]
626pub unsafe fn execute_extension<const SIGN: bool, Reg, Env>(
627    env: &mut Env,
628    vd: VReg,
629    vs2: VReg,
630    vm: bool,
631    sew: Vsew,
632    factor: VsewFactor,
633) where
634    Reg: Register,
635    Env: VectorRegistersExt<Reg>,
636    [(); SUPPORTED_ELEN_VLEN::<{ Env::ELEN }, { Env::VLEN }>]:,
637{
638    let vl = env.vl();
639    let vstart = env.vstart();
640    // SAFETY: Caller guarantees SEW >= factor*8 and valid according to function contract
641    let src_sew = unsafe { sew.divide_by_factor(factor).unwrap_unchecked() };
642
643    // SAFETY: `vl <= VLEN`
644    let mask_buf = unsafe { snapshot_mask(env.read_vregs(), vm, vl) };
645
646    for i in vstart.range_to(vl) {
647        if !mask_bit(&mask_buf, i) {
648            continue;
649        }
650        // SAFETY: vs2 group covers `vl` narrow elements
651        let raw = unsafe { read_element_u64(env.read_vregs(), vs2, i, src_sew) };
652        let result = if SIGN {
653            sign_extend_bits(raw, src_sew).cast_unsigned()
654        } else {
655            raw
656        };
657        // SAFETY: vd group covers `vl` wide elements
658        unsafe {
659            write_element_u64(env.write_vregs(), vd, i, sew, result);
660        }
661    }
662    env.mark_vs_dirty();
663    env.reset_vstart();
664}
