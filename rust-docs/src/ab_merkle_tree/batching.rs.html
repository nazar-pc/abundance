3use crate::MerkleHasher;
4use core::mem::MaybeUninit;
5use core::{hint, mem};

7/// [`MerkleHasher::PARALLEL_PAIRS`] that is checked to be supported
8pub(crate) const PARALLEL_PAIRS<H: MerkleHasher>: usize = {
9    assert!(
10        H::PARALLEL_PAIRS.is_power_of_two() && H::PARALLEL_PAIRS <= 64,
11        "`MerkleHasher::PARALLEL_PAIRS` must be a power of two that is not larger than 64"
12    );
13
14    H::PARALLEL_PAIRS
15};
16/// Number of leaves of [`MerkleHasher::PARALLEL_PAIRS`] pairs, which are hashed as a batch
17pub(crate) const BATCH_NUM_LEAVES<H: MerkleHasher>: usize = PARALLEL_PAIRS::<H> * 2;
18/// Number of consecutive leaves that trees buffer to hash them together as perfect subtrees.
19///
20/// 8 times [`MerkleHasher::PARALLEL_PAIRS`] keeps the extra stack small (4 KiB for BLAKE3 with
21/// AVX-512), while a larger buffer barely helps. Hashers without parallelism don't benefit from
22/// it, so the buffer is a single leaf for them, which trees don't use.
23pub(crate) const BUFFER_NUM_LEAVES<H: MerkleHasher>: usize = if PARALLEL_PAIRS::<H> == 1 {
24    1
25} else {
26    PARALLEL_PAIRS::<H> * 8
27};

29// TODO: Combine `NUM_PAIRS` and `NUM_LEAVES` into a single generic parameter once compiler is
30//  smart enough to deal with it
31/// Similar to [`MerkleHasher::hash_pairs()`], but takes pairs as a flat list of nodes.
32///
33/// # Panics
34/// Panics when `NUM_LEAVES != NUM_PAIRS * 2`.
35#[inline(always)]
36#[cfg_attr(feature = "no-panic", no_panic::no_panic)]
37pub(crate) fn hash_pairs<const NUM_PAIRS: usize, const NUM_LEAVES: usize, H>(
38    hasher: &H,
39    leaves: &[[u8; H::OUT_LEN]; NUM_LEAVES],
40) -> [[u8; H::OUT_LEN]; NUM_PAIRS]
41where
42    H: MerkleHasher,
43{
44    // This protects the invariant of the function
45    assert_eq!(NUM_LEAVES, NUM_PAIRS * 2);
46
47    // SAFETY: Same size (checked above) and alignment
48    let pairs = unsafe {
49        mem::transmute::<&[[u8; H::OUT_LEN]; NUM_LEAVES], &[[[u8; H::OUT_LEN]; 2]; NUM_PAIRS]>(
50            leaves,
51        )
52    };
53    hasher.hash_pairs(pairs)
54}

56/// Root of a perfect subtree with `nodes` at its lowest level.
57///
58/// Levels are hashed in place: parents are written to the front of `nodes`, such that no memory
59/// is needed beyond the output of a single [`MerkleHasher::hash_pairs()`] call. Contents of `nodes`
60/// are meaningless afterward.
61///
62/// # Safety
63/// `nodes.len()` must be a power of two.
64#[inline(always)]
65#[cfg_attr(feature = "no-panic", no_panic::no_panic)]
66pub(crate) unsafe fn perfect_subtree_root<H>(
67    hasher: &H,
68    mut nodes: &mut [[u8; H::OUT_LEN]],
69) -> [u8; H::OUT_LEN]
70where
71    H: MerkleHasher,
72{
73    // Levels with at least a batch of nodes are hashed one batch at a time, parents of each batch
74    // take the place of nodes that were hashed already
75    while nodes.len() >= BATCH_NUM_LEAVES::<H> {
76        let batches = nodes.as_chunks_mut::<{ BATCH_NUM_LEAVES::<H> }>().0;
77        for index in 0..batches.len() {
78            let parents = hash_pairs::<{ PARALLEL_PAIRS::<H> }, _, _>(hasher, &batches[index]);
79            // Parents of two consecutive batches fill the first of them, which was hashed already
80            batches[index / 2]
81                .as_chunks_mut::<{ PARALLEL_PAIRS::<H> }>()
82                .0[index % 2] = parents;
83        }
84        nodes = nodes.split_at_mut(nodes.len() / 2).0;
85    }
86
87    // Levels with fewer nodes than a batch are hashed at once, which is at most
88    // `MerkleHasher::PARALLEL_PAIRS` nodes
89    while nodes.len() > 1 {
90        // Guards always hold for levels that get here, they only keep arms with more nodes than
91        // `MerkleHasher::PARALLEL_PAIRS` from being compiled
92        match nodes.len() {
93            2 => hash_level::<1, 2, _>(hasher, nodes),
94            4 if PARALLEL_PAIRS::<H> >= 4 => hash_level::<2, 4, _>(hasher, nodes),
95            8 if PARALLEL_PAIRS::<H> >= 8 => hash_level::<4, 8, _>(hasher, nodes),
96            16 if PARALLEL_PAIRS::<H> >= 16 => hash_level::<8, 16, _>(hasher, nodes),
97            32 if PARALLEL_PAIRS::<H> >= 32 => hash_level::<16, 32, _>(hasher, nodes),
98            64 if PARALLEL_PAIRS::<H> >= 64 => hash_level::<32, 64, _>(hasher, nodes),
99            // SAFETY: Levels here have a power of two nodes, which is at least 2 and at most
100            // `MerkleHasher::PARALLEL_PAIRS`, which in turn is at most 64
101            _ => unsafe { hint::unreachable_unchecked() },
102        }
103        nodes = nodes.split_at_mut(nodes.len() / 2).0;
104    }
105
106    // SAFETY: A power of two is not zero, so a single node remains
107    *unsafe { nodes.get_unchecked(0) }
108}

110/// Hash `NUM_NODES` nodes as `NUM_PAIRS` pairs, parents are written to the front of `nodes`
111#[inline(always)]
112#[cfg_attr(feature = "no-panic", no_panic::no_panic)]
113fn hash_level<const NUM_PAIRS: usize, const NUM_NODES: usize, H>(
114    hasher: &H,
115    nodes: &mut [[u8; H::OUT_LEN]],
116) where
117    H: MerkleHasher,
118{
119    if let Some(level) = nodes.as_mut_array::<NUM_NODES>() {
120        let parents = hash_pairs::<NUM_PAIRS, NUM_NODES, _>(hasher, level);
121        *level
122            .first_chunk_mut()
123            .expect("There are fewer parents than nodes; qed") = parents;
124    }
125}

127/// Hash leaves `buffer[start..end]` as the fewest perfect subtrees that start at multiples of
128/// their sizes, calling `push(offset, height, root)` for each of them from left to right, where
129/// `offset` is the index of the first leaf of the subtree in the buffer.
130///
131/// Contents of `buffer[start..end]` are meaningless afterward, other leaves are not changed.
132///
133/// # Safety
134/// `end` must not be larger than `BUFFER_LEN` and leaves `buffer[start..end]` must be initialized.
135#[inline(always)]
136#[cfg_attr(feature = "no-panic", no_panic::no_panic)]
137pub(crate) unsafe fn hash_buffered_leaves<const BUFFER_LEN: usize, H, Push>(
138    hasher: &H,
139    buffer: &mut [MaybeUninit<[u8; H::OUT_LEN]>; BUFFER_LEN],
140    mut start: usize,
141    end: usize,
142    mut push: Push,
143) where
144    H: MerkleHasher,
145    Push: FnMut(usize, u32, [u8; H::OUT_LEN]),
146{
147    while start < end {
148        // The largest subtree that starts at a multiple of its size and ends within the range
149        let height = (end - start).ilog2().min(start.trailing_zeros());
150        let subtree_end = start + 2usize.pow(height);
151        // SAFETY: `start < subtree_end <= end <= BUFFER_LEN` and leaves within the range are
152        // initialized
153        let leaves = unsafe {
154            buffer
155                .get_unchecked_mut(start..subtree_end)
156                .assume_init_mut()
157        };
158        // SAFETY: Number of leaves is a power of two
159        let root = unsafe { perfect_subtree_root(hasher, leaves) };
160        push(start, height, root);
161        start = subtree_end;
162    }
163}
