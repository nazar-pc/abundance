18// The worst case is to have a slot with two pointers and two 32-bit size fields:
19// address + data + size + capacity
20const INTERNAL_ARGS_SIZE: usize =
21    usize::from(MAX_TOTAL_METHOD_ARGS) * (size_of::<*mut c_void>() * 2 + size_of::<u32>() * 2);
22
23// Only pointers and the structs below are written into `InternalArgs`. A single argument takes at
24// most `INTERNAL_ARGS_SIZE / MAX_TOTAL_METHOD_ARGS` bytes, and since sizes of written values are
25// multiples of the pointer alignment, which their alignment doesn't exceed, a cursor that starts at
26// the beginning of a buffer of pointers stays correctly aligned after every write.
27const _: () = {
28    assert!(
29        size_of::<*const Address>() + size_of::<FfiDataSizeCapacityRo>()
30            <= INTERNAL_ARGS_SIZE / usize::from(MAX_TOTAL_METHOD_ARGS)
31    );
32    assert!(
33        size_of::<*const Address>() + size_of::<FfiDataSizeCapacityRw>()
34            <= INTERNAL_ARGS_SIZE / usize::from(MAX_TOTAL_METHOD_ARGS)
35    );
36    assert!(size_of::<FfiDataSizeCapacityRo>().is_multiple_of(align_of::<*mut c_void>()));
37    assert!(size_of::<FfiDataSizeCapacityRw>().is_multiple_of(align_of::<*mut c_void>()));
38    assert!(align_of::<FfiDataSizeCapacityRo>() <= align_of::<*mut c_void>());
39    assert!(align_of::<FfiDataSizeCapacityRw>() <= align_of::<*mut c_void>());
40};

45    data_ptr: NonNull<u8>,
46    size: u32,
47    capacity: u32,
48}

58/// Read from an external arguments pointer and move it forward.
59///
60/// # Safety
61/// `external_args` must have enough capacity for the read value, and the current offset must
62/// have the correct alignment for the type being read.
63#[inline(always)]
64unsafe fn read_external_args<T>(external_args: &mut NonNull<c_void>) -> T {
65    // SAFETY: Guaranteed by the function contract
66    unsafe {
67        let value = external_args.cast::<T>().read();
68        *external_args = external_args.byte_add(size_of::<T>());
69        value
70    }
71}

73/// Write to an internal arguments pointer and move it forward.
74///
75/// # Safety
76/// `internal_args` must have enough capacity for the written value, and the current offset must
77/// have the correct alignment for the type being written.
78#[inline(always)]
79unsafe fn write_internal_args<T>(internal_args: &mut NonNull<c_void>, value: T) {
80    // SAFETY: Guaranteed by the function contract
81    unsafe {
82        internal_args.cast::<T>().write(value);
83        *internal_args = internal_args.byte_add(size_of::<T>());
84    }
85}

87/// Stores details about arguments that need to be processed after FFI call
88#[derive(Copy, Clone)]
89enum PostProcessing {
90    Slot {
91        /// Offset into `internal_args` where the corresponding slot entry is located.
92        ///
93        /// NOTE: An assertion above ensures `u8` is large enough to store all possible offsets.
94        internal_args_ptr: NonNull<c_void>,
95        slot_index: SlotIndex,
96        /// Whether a slot written must be non-empty.
97        ///
98        /// This is the case for state in `#[init]` methods.
99        must_be_not_empty: bool,
100    },
101    Output {
102        /// Offset into `InternalArgs` where the guest will store a pointer to potentially updated
103        /// output contents.
104        ///
105        /// NOTE: An assertion above ensures `u8` is large enough to store all possible offsets.
106        internal_args_ptr: NonNull<c_void>,
107        /// Offset into `ExternalArgs` where the host will potentially update output contents.
108        ///
109        /// This is strictly smaller than `internal_args_ptr`, hence the same type.
110        external_args_ptr: NonNull<c_void>,
111    },
112}

114/// Special container that allows aliasing of `Env` stored inside it and tracks the nested context
115/// holding slots.
116///
117/// The nested context is tracked here even when it is given to `Env`, so that the host never reads
118/// anything back from `Env` that the guest had exclusive access to (and could have replaced).
119///
120/// Variants with pointers are only created by `Self::insert_ro()`, `Self::insert_rw()` and
121/// `Self::initialize()`, which `Drop` and `Self::into_slots()` rely on.
122enum MaybeEnv<Env, Context> {
123    None(Context),
124    ReadOnly(*mut UnsafeCell<Env>, Context),
125    ReadWrite(*mut UnsafeCell<Env>, Context),
126}
127
128impl<Env, Context> Drop for MaybeEnv<Env, Context> {
129    #[inline(always)]
130    fn drop(&mut self) {
131        match self {
132            MaybeEnv::None(_) => {}
133            &mut (MaybeEnv::ReadOnly(env, _) | MaybeEnv::ReadWrite(env, _)) => {
134                // SAFETY: Created with `Box::into_raw()` in `insert_ro()`/`insert_rw()` and only
135                // freed here. The guest gets a pointer to it only for the duration of the FFI call,
136                // during which `self` (owned by `make_ffi_call()`) can't be dropped.
137                let _: Box<_> = unsafe { Box::from_raw(env) };
138            }
139        }
140    }
141}

143impl<'env> MaybeEnv<MaybeUninit<Env<'env>>, ()> {
144    /// Insert a new value and get a pointer to it, value must be initialized later with
145    /// [`Self::initialize()`]
146    #[inline(always)]
147    fn insert_ro(&mut self) -> *const MaybeUninit<Env<'env>> {
148        let env = Box::into_raw(Box::new(UnsafeCell::new(MaybeUninit::uninit())));
149        let env_ptr = {
150            // SAFETY: Valid pointer from `Box::into_raw()` right above, nothing else references
151            // the value yet
152            let env_ref = unsafe { env.as_ref_unchecked() };
153            env_ref.get().cast_const()
154        };
155        *self = Self::ReadOnly(env, ());
156        env_ptr
157    }
158
159    /// Insert a new value and get a pointer to it, value must be initialized later with
160    /// [`Self::initialize()`]
161    #[inline(always)]
162    fn insert_rw(&mut self) -> *mut MaybeUninit<Env<'env>> {
163        let env = Box::into_raw(Box::new(UnsafeCell::new(MaybeUninit::uninit())));
164        let env_ptr = {
165            // SAFETY: Valid pointer from `Box::into_raw()` right above, nothing else references
166            // the value yet
167            let env_ref = unsafe { env.as_ref_unchecked() };
168            env_ref.get()
169        };
170        *self = Self::ReadWrite(env, ());
171        env_ptr
172    }
173
174    /// # Safety
175    /// Nothing must have a live reference to `self` or its internals
176    #[inline(always)]
177    unsafe fn initialize<'slots, CreateNestedContext>(
178        self,
179        slots: NestedSlots<'slots>,
180        env_state: EnvState,
181        create_nested_context: CreateNestedContext,
182    ) -> MaybeEnv<Env<'env>, NonNull<NativeExecutorContext<'slots>>>
183    where
184        CreateNestedContext:
185            FnOnce(NestedSlots<'slots>, bool) -> &'env mut NativeExecutorContext<'slots>,
186        'slots: 'env,
187    {
188        // Only `#[view]` methods can be called through `&Env`, and without `Env` the nested context
189        // isn't used for calls at all and only holds onto slots
190        let allow_env_mutation = matches!(self, Self::ReadWrite(..));
191        let mut context = NonNull::from_mut(create_nested_context(slots, allow_env_mutation));
192
193        match self {
194            Self::None(()) => MaybeEnv::None(context),
195            Self::ReadOnly(env_ro, ()) => {
196                // SAFETY: Created from an exclusive reference above. The reference given to
197                // `Env` is derived from this pointer, the pointer itself is only used again in
198                // `Self::into_slots()`, after the guest returned.
199                let env = Env::with_executor_context(env_state, unsafe { context.as_mut() });
200                {
201                    // SAFETY: Valid pointer from `Box::into_raw()` in `Self::insert_ro()`, which is
202                    // only freed when `self` is dropped. Only the pointer to it was given out so
203                    // far, and nothing has a live reference to it as per function contract.
204                    let env_ro = unsafe { env_ro.as_mut_unchecked() };
205                    env_ro.get_mut().write(env);
206                }
207                // Very explicit cast to the initialized value since it was just written to
208                let env_ro = env_ro.cast::<UnsafeCell<Env<'env>>>();
209
210                // Prevent destructor from running and de-allocating `Env`
211                mem::forget(self);
212
213                MaybeEnv::ReadOnly(env_ro, context)
214            }
215            Self::ReadWrite(env_rw, ()) => {
216                // SAFETY: Created from an exclusive reference above. The reference given to
217                // `Env` is derived from this pointer, the pointer itself is only used again in
218                // `Self::into_slots()`, after the guest returned.
219                let env = Env::with_executor_context(env_state, unsafe { context.as_mut() });
220                {
221                    // SAFETY: Valid pointer from `Box::into_raw()` in `Self::insert_rw()`, which is
222                    // only freed when `self` is dropped. Only the pointer to it was given out so
223                    // far, and nothing has a live reference to it as per function contract.
224                    let env_rw = unsafe { env_rw.as_mut_unchecked() };
225                    env_rw.get_mut().write(env);
226                }
227                // Very explicit cast to the initialized value since it was just written to
228                let env_rw = env_rw.cast::<UnsafeCell<Env<'env>>>();
229
230                // Prevent destructor from running and de-allocating `Env`
231                mem::forget(self);
232
233                MaybeEnv::ReadWrite(env_rw, context)
234            }
235        }
236    }
237}

239impl<'env, 'slots> MaybeEnv<Env<'env>, NonNull<NativeExecutorContext<'slots>>> {
240    /// Free `Env` (if present) and get slots of the nested context
241    #[inline(always)]
242    fn into_slots(self) -> &'env mut NestedSlots<'slots> {
243        let mut context = match self {
244            MaybeEnv::None(context)
245            | MaybeEnv::ReadOnly(_, context)
246            | MaybeEnv::ReadWrite(_, context) => context,
247        };
248        // The host doesn't use `Env` anymore
249        drop(self);
250        // SAFETY: Created in `Self::initialize()` from an exclusive reference to the nested context
251        // that is borrowed for `'env`. The only reference derived from it was given to the guest
252        // through `Env`, and the guest can only use it during the FFI call, while `self` (owned by
253        // `make_ffi_call()`) can't be consumed. Consuming `self` also ensures the host doesn't
254        // access `Env` or create another reference to the nested context afterward.
255        unsafe { context.as_mut() }.slots.get_mut()
256    }
257}

259/// Call method of a contract through its FFI function.
260///
261/// # Safety
262/// `external_args` must point to arguments laid out as
263/// [`ExternalArgs`](ab_contracts_common::method::ExternalArgs) of the method described by
264/// `method_details`, with all pointers in them valid for the whole call according to the method
265/// signature. The native executor trusts contracts and their callers to uphold this, see
266/// [`NativeExecutor`](crate::NativeExecutor).
267#[inline(always)]
268#[expect(clippy::too_many_arguments, reason = "Internal API")]
269pub(super) unsafe fn make_ffi_call<'slots, CreateNestedContext>(
270    allow_env_mutation: bool,
271    is_allocate_new_address_method: bool,
272    parent_slots: &'slots mut NestedSlots<'slots>,
273    contract: Address,
274    method_details: MethodDetails,
275    external_args: NonNull<c_void>,
276    env_state: EnvState,
277    create_nested_context: CreateNestedContext,
278) -> Result<(), ContractError>
279where
280    CreateNestedContext: FnOnce(NestedSlots<'slots>, bool) -> NativeExecutorContext<'slots>,
281{
282    let MethodDetails {
283        recommended_state_capacity,
284        recommended_slot_capacity,
285        recommended_tmp_capacity,
286        mut method_metadata,
287        ffi_fn,
288    } = method_details;
289
290    // Allocate a buffer that will contain incrementally built `InternalArgs` that the method
291    // expects, according to its metadata.
292    //
293    // Writes into it are in bounds and aligned: there are at most `MAX_TOTAL_METHOD_ARGS`
294    // arguments including `self` (checked below), each written in one go, which is within bounds
295    // and keeps the cursor aligned as asserted next to `INTERNAL_ARGS_SIZE`.
296    let mut internal_args =
297        MaybeUninit::<[*mut c_void; INTERNAL_ARGS_SIZE / size_of::<*const c_void>()]>::uninit();
298    let mut post_processing = ArrayVec::<_, { usize::from(MAX_TOTAL_METHOD_ARGS) }>::new_const();
299
300    let method_metadata_decoder =
301        MethodMetadataDecoder::new(&mut method_metadata, MethodsContainerKind::Unknown);
302    let (mut arguments_metadata_decoder, method_metadata_item) =
303        match method_metadata_decoder.decode_next() {
304            Ok(result) => result,
305            Err(error) => {
306                error!(%error, "Method metadata decoding error");
307                return Err(ContractError::InternalError);
308            }
309        };
310    #[expect(
311        clippy::rest_pattern_accessible_field,
312        reason = "Do not need other fields"
313    )]
314    let MethodMetadataItem {
315        method_kind,
316        num_arguments,
317        ..
318    } = method_metadata_item;
319
320    let number_of_arguments = usize::from(num_arguments) + usize::from(method_kind.has_self());
321
322    if number_of_arguments > usize::from(MAX_TOTAL_METHOD_ARGS) {
323        debug!(%number_of_arguments, "Too many arguments");
324        return Err(ContractError::BadInput);
325    }
326
327    let internal_args = NonNull::new(internal_args.as_mut_ptr().cast::<c_void>())
328        .expect("Taken from non-null instance; qed");
329    // This pointer will be moving as the data structure is being constructed, while `internal_args`
330    // will keep pointing to the beginning
331    let internal_args_cursor = &mut internal_args.clone();
332    // This pointer will be moving as the data structure is being read, while `external_args` will
333    // keep pointing to the beginning
334    let external_args_cursor = &mut external_args.clone();
335
336    // `view_only == true` when only `#[view]` method is allowed
337    let (view_only, mut slots) = match method_kind {
338        MethodKind::Init
339        | MethodKind::UpdateStateless
340        | MethodKind::UpdateStatefulRo
341        | MethodKind::UpdateStatefulRw => {
342            if !allow_env_mutation {
343                warn!(allow_env_mutation, "Only `#[view]` methods are allowed");
344                return Err(ContractError::Forbidden);
345            }
346
347            let Some(slots) = parent_slots.new_nested_rw() else {
348                error!("Unexpected creation of non-read-only slots from read-only slots");
349                return Err(ContractError::InternalError);
350            };
351
352            (false, slots)
353        }
354        MethodKind::ViewStateless | MethodKind::ViewStateful => {
355            let slots = parent_slots.new_nested_ro();
356            (true, slots)
357        }
358    };
359
360    let mut maybe_env = MaybeEnv::None(());
361
362    // Handle `&self` and `&mut self`
363    match method_kind {
364        MethodKind::Init | MethodKind::UpdateStateless | MethodKind::ViewStateless => {
365            // No state handling is needed
366        }
367        MethodKind::UpdateStatefulRo | MethodKind::ViewStateful => {
368            let state_bytes = slots
369                .use_ro(SlotKey {
370                    owner: contract,
371                    contract: Address::SYSTEM_STATE,
372                })
373                .ok_or(ContractError::Forbidden)?;
374
375            if state_bytes.is_empty() {
376                warn!("Contract does not have state yet, can't call stateful method before init");
377                return Err(ContractError::Forbidden);
378            }
379
380            // SAFETY: In bounds of `internal_args` and aligned, see its allocation
381            unsafe {
382                write_internal_args(
383                    internal_args_cursor,
384                    FfiDataSizeCapacityRo {
385                        data_ptr: NonNull::from_ref(state_bytes.as_slice()).as_non_null_ptr(),
386                        size: state_bytes.len(),
387                        capacity: state_bytes.len(),
388                    },
389                );
390            }
391        }
392        MethodKind::UpdateStatefulRw => {
393            if view_only {
394                warn!("Only `#[view]` methods are allowed");
395                return Err(ContractError::Forbidden);
396            }
397
398            let slot_key = SlotKey {
399                owner: contract,
400                contract: Address::SYSTEM_STATE,
401            };
402            let (slot_index, state_bytes) = slots
403                .use_rw(slot_key, recommended_state_capacity)
404                .ok_or(ContractError::Forbidden)?;
405
406            if state_bytes.is_empty() {
407                warn!("Contract does not have state yet, can't call stateful method before init");
408                return Err(ContractError::Forbidden);
409            }
410
411            post_processing.push(PostProcessing::Slot {
412                internal_args_ptr: *internal_args_cursor,
413                slot_index,
414                must_be_not_empty: false,
415            });
416
417            // SAFETY: In bounds of `internal_args` and aligned, see its allocation
418            unsafe {
419                write_internal_args(
420                    internal_args_cursor,
421                    FfiDataSizeCapacityRw {
422                        data_ptr: state_bytes.as_mut_ptr(),
423                        size: state_bytes.len(),
424                        capacity: state_bytes.capacity(),
425                    },
426                );
427            }
428        }
429    }
430
431    let mut new_address_ptr = None;
432
433    // Handle all other arguments one by one
434    for argument_index in 0..num_arguments {
435        let argument_kind = match arguments_metadata_decoder.decode_next() {
436            Some(Ok(item)) => item.argument_kind,
437            Some(Err(error)) => {
438                error!(%error, "Argument metadata decoding error");
439                return Err(ContractError::InternalError);
440            }
441            None => {
442                error!("Argument not found, invalid metadata");
443                return Err(ContractError::InternalError);
444            }
445        };
446
447        match argument_kind {
448            ArgumentKind::EnvRo => {
449                // Allocate and create a pointer now, the actual value will be inserted towards the
450                // end of the function
451                let env_ro = maybe_env.insert_ro().cast::<Env<'_>>();
452                // SAFETY: In bounds of `internal_args` and aligned, see its allocation
453                unsafe {
454                    write_internal_args(internal_args_cursor, env_ro);
455                }
456
457                // Size for `#[env]` is implicit and doesn't need to be added to `InternalArgs`
458            }
459            ArgumentKind::EnvRw => {
460                if view_only {
461                    return Err(ContractError::Forbidden);
462                }
463
464                // Allocate and create a pointer now, the actual value will be inserted towards the
465                // end of the function
466                let env_rw = maybe_env.insert_rw().cast::<Env<'_>>();
467
468                // SAFETY: In bounds of `internal_args` and aligned, see its allocation
469                unsafe {
470                    write_internal_args(internal_args_cursor, env_rw);
471                }
472
473                // Size for `#[env]` is implicit and doesn't need to be added to `InternalArgs`
474            }
475            ArgumentKind::TmpRo | ArgumentKind::SlotRo => {
476                let tmp = matches!(argument_kind, ArgumentKind::TmpRo);
477
478                let (owner, contract) = if tmp {
479                    if view_only {
480                        return Err(ContractError::Forbidden);
481                    }
482
483                    // Null contact is used implicitly for `#[tmp]` since it is not possible for
484                    // this contract to write something there directly
485                    (&contract, Address::NULL)
486                } else {
487                    // SAFETY: Arguments before this one were read in metadata order, so the
488                    // cursor points to this slot's field in `ExternalArgs`, which is a pointer to
489                    // the address, valid for reads during the call as per function contract
490                    (
491                        unsafe { &*read_external_args::<*const Address>(external_args_cursor) },
492                        contract,
493                    )
494                };
495
496                let slot_key = SlotKey {
497                    owner: *owner,
498                    contract,
499                };
500                let slot_bytes = slots.use_ro(slot_key).ok_or(ContractError::Forbidden)?;
501
502                // SAFETY: In bounds of `internal_args` and aligned, see its allocation
503                unsafe {
504                    if !tmp {
505                        write_internal_args(internal_args_cursor, owner);
506                    }
507                    write_internal_args(
508                        internal_args_cursor,
509                        FfiDataSizeCapacityRo {
510                            data_ptr: NonNull::from_ref(slot_bytes.as_slice()).as_non_null_ptr(),
511                            size: slot_bytes.len(),
512                            capacity: slot_bytes.len(),
513                        },
514                    );
515                }
516            }
517            ArgumentKind::TmpRw | ArgumentKind::SlotRw => {
518                if view_only {
519                    return Err(ContractError::Forbidden);
520                }
521
522                let tmp = matches!(argument_kind, ArgumentKind::TmpRw);
523
524                let (owner, contract, capacity) = if tmp {
525                    // Null contact is used implicitly for `#[tmp]` since it is not possible for
526                    // this contract to write something there directly
527                    (&contract, Address::NULL, recommended_tmp_capacity)
528                } else {
529                    // SAFETY: Arguments before this one were read in metadata order, so the
530                    // cursor points to this slot's field in `ExternalArgs`, which is a pointer to
531                    // the address, valid for reads during the call as per function contract
532                    let address =
533                        unsafe { &*read_external_args::<*const Address>(external_args_cursor) };
534
535                    (address, contract, recommended_slot_capacity)
536                };
537
538                let slot_key = SlotKey {
539                    owner: *owner,
540                    contract,
541                };
542                let (slot_index, slot_bytes) = slots
543                    .use_rw(slot_key, capacity)
544                    .ok_or(ContractError::Forbidden)?;
545
546                if !tmp {
547                    // SAFETY: In bounds of `internal_args` and aligned, see its allocation
548                    unsafe {
549                        write_internal_args(internal_args_cursor, owner);
550                    }
551                }
552
553                post_processing.push(PostProcessing::Slot {
554                    internal_args_ptr: *internal_args_cursor,
555                    slot_index,
556                    must_be_not_empty: false,
557                });
558
559                // SAFETY: In bounds of `internal_args` and aligned, see its allocation
560                unsafe {
561                    write_internal_args(
562                        internal_args_cursor,
563                        FfiDataSizeCapacityRw {
564                            data_ptr: slot_bytes.as_mut_ptr(),
565                            size: slot_bytes.len(),
566                            capacity: slot_bytes.capacity(),
567                        },
568                    );
569                }
570            }
571            ArgumentKind::Input => {
572                // SAFETY: Arguments before this one were read in metadata order, so the cursor
573                // points to this input's pointer, size and capacity in `ExternalArgs` as per
574                // function contract. Writing is in bounds of `internal_args` and aligned, see its
575                // allocation.
576                unsafe {
577                    let data_size_capacity =
578                        read_external_args::<FfiDataSizeCapacityRw>(external_args_cursor);
579                    write_internal_args(internal_args_cursor, data_size_capacity);
580                }
581            }
582            ArgumentKind::Output | ArgumentKind::Return => {
583                let last_argument = argument_index == num_arguments - 1;
584                // `#[init]` method returns the state of the contract and needs to be stored
585                // accordingly
586                if matches!((method_kind, last_argument), (MethodKind::Init, true)) {
587                    if view_only {
588                        return Err(ContractError::Forbidden);
589                    }
590
591                    let slot_key = SlotKey {
592                        owner: contract,
593                        contract: Address::SYSTEM_STATE,
594                    };
595                    let (slot_index, state_bytes) = slots
596                        .use_rw(slot_key, recommended_state_capacity)
597                        .ok_or(ContractError::Forbidden)?;
598
599                    if !state_bytes.is_empty() {
600                        debug!("Can't initialize already initialized contract");
601                        return Err(ContractError::Forbidden);
602                    }
603
604                    if matches!(argument_kind, ArgumentKind::Return) {
605                        // SAFETY: In bounds of `internal_args` and aligned, see its allocation
606                        unsafe {
607                            // The return type is `TrivialType` and doesn't have size/capacity
608                            write_internal_args(internal_args_cursor, state_bytes.as_mut_ptr());
609                        }
610                        // SAFETY: The return type is the state type (enforced by `#[contract]` for
611                        // `#[init]`), which is `TrivialType`, so its recommended capacity in
612                        // metadata is its size, and `use_rw()` above ensured at least that much
613                        // capacity. The bytes are not read until the method returns, and on
614                        // success it has written the whole state through the pointer above. On
615                        // error slots are reset and the bytes are discarded. It is more efficient
616                        // to just set the length here right away than do explicit post-processing
617                        // below.
618                        unsafe {
619                            state_bytes.set_len(recommended_state_capacity);
620                        }
621                    } else {
622                        post_processing.push(PostProcessing::Slot {
623                            internal_args_ptr: *internal_args_cursor,
624                            slot_index,
625                            must_be_not_empty: true,
626                        });
627
628                        // SAFETY: In bounds of `internal_args` and aligned, see its allocation
629                        unsafe {
630                            write_internal_args(
631                                internal_args_cursor,
632                                FfiDataSizeCapacityRw {
633                                    data_ptr: state_bytes.as_mut_ptr(),
634                                    size: 0,
635                                    capacity: state_bytes.capacity(),
636                                },
637                            );
638                        }
639                    }
640                } else {
641                    if last_argument && is_allocate_new_address_method {
642                        // SAFETY: Arguments before this one were read in metadata order, so the
643                        // cursor points to the return value pointer in `ExternalArgs` of
644                        // `AddressAllocator::allocate_address()` as per function contract. Writing
645                        // is in bounds of `internal_args` and aligned, see its allocation.
646                        unsafe {
647                            let address = read_external_args::<*mut Address>(external_args_cursor);
648                            write_internal_args(internal_args_cursor, address);
649                            new_address_ptr.replace(address);
650                        }
651                    } else if matches!(argument_kind, ArgumentKind::Return) {
652                        // SAFETY: Arguments before this one were read in metadata order, so the
653                        // cursor points to the return value pointer in `ExternalArgs` as per
654                        // function contract. Writing is in bounds of `internal_args` and aligned,
655                        // see its allocation.
656                        unsafe {
657                            // The return type is `TrivialType` and doesn't have size/capacity
658                            let data = read_external_args::<*mut u8>(external_args_cursor);
659                            write_internal_args(internal_args_cursor, data);
660                        }
661                    } else {
662                        post_processing.push(PostProcessing::Output {
663                            internal_args_ptr: *internal_args_cursor,
664                            external_args_ptr: *external_args_cursor,
665                        });
666
667                        // SAFETY: Arguments before this one were read in metadata order, so the
668                        // cursor points to this output's pointer, size and capacity in
669                        // `ExternalArgs` as per function contract. Writing is in bounds of
670                        // `internal_args` and aligned, see its allocation.
671                        unsafe {
672                            let data_size_capacity =
673                                read_external_args::<FfiDataSizeCapacityRw>(external_args_cursor);
674                            write_internal_args(internal_args_cursor, data_size_capacity);
675                        }
676                    }
677                }
678            }
679        }
680    }
681
682    let mut nested_context = None;
683    // SAFETY: `internal_args` only holds a raw pointer to `Env` and no references exist
684    let maybe_env = unsafe {
685        maybe_env.initialize(slots, env_state, |slots, allow_env_mutation| {
686            nested_context.insert(create_nested_context(slots, allow_env_mutation))
687        })
688    };
689
690    let internal_args = internal_args.cast::<c_void>();
691    // SAFETY: `internal_args` was built according to `method_metadata`, which `#[contract]`
692    // generated together with `ffi_fn` for the same method (the native executor trusts contracts,
693    // including manual `Contract` implementations, to pair them correctly, see `NativeExecutor`).
694    // Pointers in it stay valid for the whole call:
695    // * state and slot buffers belong to `slots` and are registered as accessed there, so nested
696    //   calls can't modify or reallocate them
697    // * `Env` is allocated in `maybe_env` and only freed when it is consumed after the call, the
698    //   nested context lives in `nested_context`, which outlives the call
699    // * the rest comes from `external_args`, valid during the call as per function contract
700    let result = Result::<(), ContractError>::from(unsafe { ffi_fn(internal_args) });
701
702    let slots = maybe_env.into_slots();
703
704    if let Err(error) = result {
705        slots.reset();
706
707        return Err(error);
708    }
709
710    // Catch new address allocation and add it to new contracts in slots for code and other things
711    // to become usable for it
712    if let Some(new_address_ptr) = new_address_ptr {
713        // Assert that the API has the expected shape
714        let _: fn(&mut AddressAllocator, &mut Env<'_>) -> Result<Address, ContractError> =
715            AddressAllocator::allocate_address;
716        // SAFETY: `new_address_ptr` is the return value pointer of
717        // `AddressAllocator::allocate_address()` (checked above), which the guest initialized since
718        // the call succeeded
719        let new_address = unsafe { new_address_ptr.read() };
720        if !slots.add_new_contract(new_address) {
721            warn!("Failed to add new contract returned by address allocator");
722            return Err(ContractError::InternalError);
723        }
724    }
725
726    for &entry in &post_processing {
727        match entry {
728            PostProcessing::Slot {
729                internal_args_ptr,
730                slot_index,
731                must_be_not_empty,
732            } => {
733                // SAFETY: Points to the `FfiDataSizeCapacityRw` written into `internal_args` for
734                // this slot above, which is still alive, and the guest no longer accesses it
735                let FfiDataSizeCapacityRw {
736                    data_ptr,
737                    size,
738                    capacity: _,
739                } = unsafe { internal_args_ptr.cast().read() };
740
741                if must_be_not_empty && size == 0 {
742                    error!(
743                        %size,
744                        "Contract returned empty size where it is not allowed, likely state of \
745                        `#[init]` method"
746                    );
747                    return Err(ContractError::BadOutput);
748                }
749
750                let slot_bytes = slots
751                    .access_used_rw(slot_index)
752                    .expect("Was used above and must exist since `Slots` was not dropped yet; qed");
753
754                // Guest created a different allocation for slot, copy bytes
755                if !ptr::eq(data_ptr, slot_bytes.as_ptr()) {
756                    if data_ptr.is_null() {
757                        error!("Contract returned `null` pointer for slot data");
758                        return Err(ContractError::BadOutput);
759                    }
760                    // SAFETY: The guest replaced the slot contents with a different allocation,
761                    // whose first `size` bytes are initialized as required by `IoType` (the native
762                    // executor trusts contracts to uphold it, see `NativeExecutor`)
763                    let data =
764                        unsafe { slice::from_raw_parts(data_ptr.cast::<u8>(), size as usize) };
765                    slot_bytes.copy_from_slice(data);
766                    continue;
767                }
768
769                if size > slot_bytes.capacity() {
770                    error!(
771                        %size,
772                        capacity = %slot_bytes.capacity(),
773                        "Contract returned invalid size for slot data in source allocation"
774                    );
775                    return Err(ContractError::BadOutput);
776                }
777                // Otherwise, set the size to what guest claims
778                //
779                // SAFETY: Checked to be within capacity above, and the guest initialized the first
780                // `size` bytes as required by `IoType` (the native executor trusts contracts to
781                // uphold it, see `NativeExecutor`)
782                unsafe {
783                    slot_bytes.set_len(size);
784                }
785            }
786            PostProcessing::Output {
787                internal_args_ptr,
788                external_args_ptr,
789            } => {
790                // SAFETY: Points to the `FfiDataSizeCapacityRw` written into `internal_args` for
791                // this output above, which is still alive, and the guest no longer accesses it
792                let source_size =
793                    unsafe { internal_args_ptr.cast::<FfiDataSizeCapacityRw>().read() }.size;
794                // SAFETY: Points to this output's pointer, size and capacity in `ExternalArgs`,
795                // which are valid for writes during the call as per function contract and not
796                // accessed by anything else, since the guest returned
797                let FfiDataSizeCapacityRw {
798                    data_ptr: _,
799                    size,
800                    capacity,
801                } = unsafe { external_args_ptr.cast::<FfiDataSizeCapacityRw>().as_mut() };
802
803                if source_size > *capacity {
804                    error!(
805                        size = %source_size,
806                        %capacity,
807                        "Contract returned invalid size for output in source allocation"
808                    );
809                    return Err(ContractError::BadOutput);
810                }
811
812                *size = source_size;
813            }
814        }
815    }
816
817    Ok(())
818}
