28/// Temporary storage items that will be pruned from the database eventually
29#[derive(Debug)]
30pub(crate) enum StorageItemTemporary {
31    Block(StorageItemTemporaryBlock),
32    SegmentHeaders(StorageItemTemporarySegmentHeaders),
33    SuperSegmentHeaders(StorageItemTemporarySuperSegmentHeaders),
34    ShardMaxLocalSegmentIndices(StorageItemTemporaryShardMaxLocalSegmentIndices),
35}
36
37impl StorageItem for StorageItemTemporary {
38    #[inline(always)]
39    fn total_bytes(&self) -> Result<u32, StorageItemError> {
40        match self {
41            Self::Block(block) => block.total_bytes(),
42            Self::SegmentHeaders(segment_headers) => segment_headers.total_bytes(),
43            Self::SuperSegmentHeaders(super_segment_headers) => super_segment_headers.total_bytes(),
44            Self::ShardMaxLocalSegmentIndices(shard_max_local_segment_indices) => {
45                shard_max_local_segment_indices.total_bytes()
46            }
47        }
48    }
49
50    #[inline(always)]
51    fn write<'a>(
52        &self,
53        buffer: &'a mut [MaybeUninit<u8>],
54    ) -> Result<StorageItemWriteResult<'a>, StorageItemError> {
55        let (variant, storage_item_size) = match self {
56            Self::Block(block) => (StorageItemBlockVariant::Block, block.write(buffer)?),
57            Self::SegmentHeaders(segment_headers) => (
58                StorageItemBlockVariant::SegmentHeaders,
59                segment_headers.write(buffer)?,
60            ),
61            Self::SuperSegmentHeaders(super_segment_headers) => (
62                StorageItemBlockVariant::SuperSegmentHeaders,
63                super_segment_headers.write(buffer)?,
64            ),
65            Self::ShardMaxLocalSegmentIndices(shard_max_local_segment_indices) => (
66                StorageItemBlockVariant::ShardMaxLocalSegmentIndices,
67                shard_max_local_segment_indices.write(buffer)?,
68            ),
69        };
70
71        let (storage_item_bytes, buffer) = buffer.split_at_mut(storage_item_size);
72        // SAFETY: Storage item bytes were just written to
73        let storage_item_bytes = unsafe { storage_item_bytes.assume_init_mut() };
74
75        Ok(StorageItemWriteResult {
76            storage_item_variant: variant as u8,
77            storage_item_bytes,
78            buffer,
79        })
80    }
81
82    #[inline(always)]
83    fn read(variant: u8, buffer: &SharedAlignedBuffer<Align16>) -> Result<Self, StorageItemError> {
84        let variant = StorageItemBlockVariant::try_from_bytes_exact(&[variant])
85            .copied()
86            .ok_or(StorageItemError::UnknownStorageItemVariant(variant))?;
87
88        Ok(match variant {
89            StorageItemBlockVariant::Block => Self::Block(StorageItemTemporaryBlock::read(buffer)?),
90            StorageItemBlockVariant::SegmentHeaders => {
91                Self::SegmentHeaders(StorageItemTemporarySegmentHeaders::read(buffer.as_slice())?)
92            }
93            StorageItemBlockVariant::SuperSegmentHeaders => Self::SuperSegmentHeaders(
94                StorageItemTemporarySuperSegmentHeaders::read(buffer.as_slice())?,
95            ),
96            StorageItemBlockVariant::ShardMaxLocalSegmentIndices => {
97                Self::ShardMaxLocalSegmentIndices(
98                    StorageItemTemporaryShardMaxLocalSegmentIndices::read(buffer.as_slice())?,
99                )
100            }
101        })
102    }
103}

108        PageGroupKind::Temporary
