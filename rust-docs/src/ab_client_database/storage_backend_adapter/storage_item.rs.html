15/// The minimum overhead that the storage item will have due to the way it is stored on disk
16pub(crate) const fn min_segment_item_overhead() -> usize {
17    // Align buffer used by storage item to 16 bytes
18    let prefix_size = StorageItemContainer::<()>::prefix_size().next_multiple_of(size_of::<u128>());
19
20    prefix_size + StorageItemContainer::<()>::suffix_size()
21}

23#[derive(Debug, thiserror::Error)]
24pub(crate) enum StorageItemError {
25    /// Buffer too small
26    #[error("Buffer too small: expected {expected}, actual {actual}")]
27    BufferTooSmall { expected: usize, actual: usize },
28    /// Buffer too large
29    #[error("Buffer too large, {extra_bytes} extra bytes were provided")]
30    BufferTooLarge { extra_bytes: usize },
31    /// Need more bytes
32    #[error("Need {0} more bytes")]
33    NeedMoreBytes(usize),
34    /// Unknown storage item variant
35    #[error("Unknown storage item variant {0}")]
36    UnknownStorageItemVariant(u8),
37    /// Invalid data length
38    #[error("Invalid data length {data_type}: expected {expected}, actual {actual}")]
39    InvalidDataLength {
40        data_type: &'static str,
41        expected: usize,
42        actual: usize,
43    },
44    /// Invalid data alignment
45    #[error("Invalid data alignment {data_type}")]
46    InvalidDataAlignment { data_type: &'static str },
47    /// Checksum mismatch
48    #[error("Checksum mismatch: expected {expected}, actual {actual}")]
49    ChecksumMismatch {
50        expected: Blake3Hash,
51        actual: Blake3Hash,
52    },
53    /// Repeat checksum mismatch
54    #[error("Repeat checksum mismatch: expected {expected}, actual {actual}")]
55    RepeatChecksumMismatch {
56        expected: Blake3Hash,
57        actual: Blake3Hash,
58    },
59    /// Storage item checksum mismatch
60    #[error("Storage item checksum mismatch: expected {expected}, actual {actual}")]
61    StorageItemChecksumMismatch {
62        expected: Blake3Hash,
63        actual: Blake3Hash,
64    },
65    /// Invalid buffer contents
66    #[error("Invalid buffer contents")]
67    InvalidBufferContents,
68    /// Storage item is too large
69    #[error("Storage item is too large")]
70    StorageItemTooLarge,
71}

73impl From<ReadError> for StorageItemError {
74    #[inline(always)]
75    fn from(error: ReadError) -> Self {
76        match error {
77            ReadError::NotEnoughBytes { missing } => {
78                Self::NeedMoreBytes(usize::try_from(missing.get()).unwrap_or(usize::MAX))
79            }
80            ReadError::Misaligned { alignment: _ } => Self::InvalidDataAlignment {
81                data_type: "storage item data",
82            },
83        }
84    }
85}

87/// The result of [`StorageItem::write()`] call
88pub(crate) struct StorageItemWriteResult<'a> {
89    /// Storage item variant
90    pub(crate) storage_item_variant: u8,
91    /// Bytes where storage item was written
92    pub(crate) storage_item_bytes: &'a [u8],
93    /// Remaining bytes of the input buffer
94    pub(crate) buffer: &'a mut [MaybeUninit<u8>],
95}

97pub(crate) trait StorageItem: fmt::Debug + Send + Sync + Sized + 'static {
98    /// Total number of bytes
99    fn total_bytes(&self) -> Result<u32, StorageItemError>;
100
101    /// Write a storage item to the provided buffer.
102    ///
103    /// The buffer is required to be aligned to 128-bit.
104    ///
105    /// Returns the storage item variant and the number of bytes written.
106    fn write<'a>(
107        &self,
108        buffer: &'a mut [MaybeUninit<u8>],
109    ) -> Result<StorageItemWriteResult<'a>, StorageItemError>;
110
111    /// The inverse of [`Self::write()`].
112    ///
113    /// The buffer references the memory of the storage item, which may be large. Parts of it that
114    /// are kept in memory for a long time should be copied.
115    fn read(variant: u8, buffer: &SharedAlignedBuffer<Align16>) -> Result<Self, StorageItemError>;
116}

124#[derive(Debug)]
125pub(super) struct StorageItemContainer<SI> {
126    pub(super) sequence_number: u64,
127    pub(super) storage_item: SI,
128}

130impl<SI> StorageItemContainer<SI> {
131    const fn prefix_size() -> usize {
132        // Sequence number + enum variant + storage item size + checksum
133        size_of::<u64>() + size_of::<u8>() + size_of::<u32>() + size_of::<Blake3Hash>()
134    }
135
136    const fn suffix_size() -> usize {
137        // Storage item checksum + repeat of prefix checksum
138        size_of::<Blake3Hash>() * 2
139    }
140}

142impl<SI> StorageItemContainer<SI>
143where
144    SI: StorageItem,
145{
146    /// Returns the number of pages necessary to write this storage item
147    pub(super) fn num_pages(&self) -> Result<u32, StorageItemError> {
148        let storage_item_size = self.storage_item.total_bytes()?;
149
150        let num_bytes = u32::try_from(min_segment_item_overhead())
151            .ok()
152            .and_then(|overhead| storage_item_size.checked_add(overhead))
153            .ok_or(StorageItemError::StorageItemTooLarge)?;
154
155        Ok(num_bytes.div_ceil(OwnedPages::ALIGN))
156    }
157
158    /// Write a storage item to the provided buffer of whole pages, which must be aligned to 16
159    /// bytes.
160    ///
161    /// Successful write means the whole buffer is fully initialized.
162    pub(super) fn write_to_pages(
163        &self,
164        mut buffer: &mut [MaybeUninit<u8>],
165    ) -> Result<(), StorageItemError> {
166        debug_assert!(
167            buffer.len().is_multiple_of(OwnedPages::ALIGN as usize),
168            "Buffer must contain whole pages"
169        );
170
171        // Align buffer used by storage item to 128 bytes
172        let prefix_bytes = buffer
173            .split_off_mut(..Self::prefix_size().next_multiple_of(size_of::<u128>()))
174            .expect("Always fits one page; qed");
175        let (prefix_bytes, prefix_padding_bytes) = prefix_bytes.split_at_mut(Self::prefix_size());
176        prefix_padding_bytes.write_filled(0);
177
178        let StorageItemWriteResult {
179            storage_item_variant,
180            storage_item_bytes,
181            mut buffer,
182        } = self.storage_item.write(buffer)?;
183
184        let buffer_len = buffer.len();
185        let suffix_bytes = buffer
186            .split_off_mut(..Self::suffix_size())
187            .ok_or_else(|| StorageItemError::NeedMoreBytes(Self::suffix_size() - buffer_len))?;
188
189        let (before_checksum, checksum_bytes) =
190            prefix_bytes.split_at_mut(size_of::<u64>() + size_of::<u8>() + size_of::<u32>());
191        let (sequence_number_bytes, remainder) = before_checksum.split_at_mut(size_of::<u64>());
192        let (storage_item_variant_bytes, storage_item_size_bytes) =
193            remainder.split_at_mut(size_of::<u8>());
194
195        // Write prefix
196        sequence_number_bytes.write_copy_of_slice(&self.sequence_number.to_le_bytes());
197        storage_item_variant_bytes[0].write(storage_item_variant);
198        let storage_item_size = u32::try_from(storage_item_bytes.len())
199            .map_err(|_error| StorageItemError::StorageItemTooLarge)?;
200        storage_item_size_bytes.write_copy_of_slice(&storage_item_size.to_le_bytes());
201        // SAFETY: Wrote to all components of `before_checksum` above
202        let checksum = single_block_hash(unsafe { before_checksum.assume_init_ref() })
203            .expect("Less than one block worth of data; qed");
204        checksum_bytes.write_copy_of_slice(&checksum);
205
206        let (storage_item_checksum_bytes, prefix_checksum_repeat_bytes) =
207            suffix_bytes.split_at_mut(size_of::<Blake3Hash>());
208
209        // Write suffix
210        let storage_item_checksum = *hash(storage_item_bytes).as_bytes();
211        storage_item_checksum_bytes.write_copy_of_slice(&storage_item_checksum);
212        prefix_checksum_repeat_bytes.write_copy_of_slice(&checksum);
213
214        if buffer.len() < OwnedPages::ALIGN as usize {
215            buffer.write_filled(0);
216        } else {
217            return Err(StorageItemError::BufferTooLarge {
218                extra_bytes: buffer.len(),
219            });
220        }
221
222        Ok(())
223    }
224
225    /// The inverse of [`Self::write_to_pages()`].
226    ///
227    /// Pages start with the storage item, which references them without copying.
228    ///
229    /// Returns the storage item container and the number of pages it occupies.
230    pub(super) fn read_from_pages(pages: &SharedPages) -> Result<(Self, u32), StorageItemError> {
231        let mut reader = pages.reader();
232
233        let prefix_bytes =
234            reader.read_bytes_array::<{ StorageItemContainer::<()>::prefix_size() }>()?;
235        let (sequence_number_bytes, remainder) = prefix_bytes.split_at(size_of::<u64>());
236        let (storage_item_variant_bytes, remainder) = remainder.split_at(size_of::<u8>());
237        let (storage_item_size_bytes, checksum_bytes) = remainder.split_at(size_of::<u32>());
238        let checksum = Blake3Hash::new(
239            single_block_hash(&prefix_bytes[..prefix_bytes.len() - checksum_bytes.len()])
240                .expect("Less than one block worth of data; qed"),
241        );
242
243        if checksum.as_slice() != checksum_bytes {
244            return Err(StorageItemError::ChecksumMismatch {
245                expected: checksum,
246                actual: Blake3Hash::new(checksum_bytes.try_into().expect("Correct length; qed")),
247            });
248        }
249
250        let sequence_number = u64::from_le_bytes(
251            sequence_number_bytes
252                .try_into()
253                .expect("Correct length; qed"),
254        );
255        let storage_item_variant = storage_item_variant_bytes[0];
256        let storage_item_size = u32::from_le_bytes(
257            storage_item_size_bytes
258                .try_into()
259                .expect("Correct length; qed"),
260        );
261
262        // Storage item starts at a multiple of 16 bytes after the prefix
263        reader.read_to_alignment::<Align16>()?;
264        let storage_item = reader.read_buffer::<Align16>(storage_item_size)?;
265
266        let suffix_bytes =
267            reader.read_bytes_array::<{ StorageItemContainer::<()>::suffix_size() }>()?;
268        let (storage_item_checksum_bytes, prefix_checksum_repeat_bytes) =
269            suffix_bytes.split_at(size_of::<Blake3Hash>());
270
271        if checksum.as_slice() != prefix_checksum_repeat_bytes {
272            return Err(StorageItemError::RepeatChecksumMismatch {
273                expected: checksum,
274                actual: Blake3Hash::new(
275                    prefix_checksum_repeat_bytes
276                        .try_into()
277                        .expect("Correct length; qed"),
278                ),
279            });
280        }
281
282        let storage_item_checksum = Blake3Hash::from(hash(&storage_item));
283        if storage_item_checksum.as_slice() != storage_item_checksum_bytes {
284            return Err(StorageItemError::StorageItemChecksumMismatch {
285                expected: storage_item_checksum,
286                actual: Blake3Hash::new(
287                    storage_item_checksum_bytes
288                        .try_into()
289                        .expect("Correct length; qed"),
290                ),
291            });
292        }
293
294        let storage_item = SI::read(storage_item_variant, &storage_item)?;
295        let num_pages = reader.position().div_ceil(SharedPages::ALIGN);
296
297        Ok((
298            Self {
299                sequence_number,
300                storage_item,
301            },
302            num_pages,
303        ))
304    }
