16/// Re-create the genesis segment on demand.
17///
18/// This is a bit of a hack and is useful for deriving of the genesis beacon chain segment that is a
19/// special case since we don't have enough data in the blockchain history itself during genesis to
20/// do the archiving.
21pub fn recreate_genesis_segment(
22    owned_genesis_block: &OwnedBeaconChainBlock,
23    erasure_coding: ErasureCoding,
24) -> NewArchivedSegment {
25    let encoded_block = encode_block(owned_genesis_block);
26
27    let block_outcome = Archiver::new(ShardIndex::BEACON_CHAIN, erasure_coding)
28        .add_block(encoded_block, Vec::new())
29        .expect("Genesis block is much smaller than `u32::MAX` bytes; qed");
30    // Since there is a single segment in the genesis super segment, the segment proof is empty and
31    // pieces are complete as they are
32    block_outcome
33        .archived_segments
34        .into_iter()
35        .next()
36        .expect("Genesis block fills at least one segment; qed")
37}

39/// Error for [`recreate_segment()`]
40#[derive(Debug, thiserror::Error)]
41pub enum RecreateSegmentError {
42    /// Read block error
43    #[error("Read block error: {0}")]
44    ReadBlockError(#[from] ReadBlockError),
45    /// Archiver instantiation error
46    #[error("Archiver instantiation error: {0}")]
47    ArchiverInstantiationError(#[from] ArchiverInstantiationError),
48    /// Encoded block is too large to archive
49    #[error("Encoded block {block_number} is too large to archive: {size} bytes")]
50    BlockTooLarge {
51        /// Block number
52        block_number: BlockNumber,
53        /// Encoded block size in bytes
54        size: usize,
55    },
56    /// Blocking task join error
57    #[error("Blocking task join error: {0}")]
58    BlockingTaskJoinError(#[from] JoinError),
59}

61/// Re-create a segment on demand.
62///
63/// `last_segment_header` corresponds to the last segment before the segment being re-created and
64/// indicates where segment archiving should start. `segment_proof` is the proof of the segment in
65/// its super segment, which is written into all pieces.
66///
67/// `extract_block_objects` allows extracting objects stored in blocks to translate them into global
68/// objects.
69///
70/// Returns `Ok(None)` if one of the segment blocks is already pruned.
71pub async fn recreate_segment<Block, CI, EBO>(
72    last_segment_header: Option<SegmentHeader>,
73    chain_info: &CI,
74    erasure_coding: ErasureCoding,
75    segment_proof: &SegmentProof,
76    mut extract_block_objects: EBO,
77) -> Result<Option<NewArchivedSegment>, RecreateSegmentError>
78where
79    Block: GenericOwnedBlock,
80    CI: ChainInfo<Block>,
81    EBO: FnMut(&Block) -> Vec<BlockObject>,
82{
83    let best_block_root = chain_info.best_root();
84
85    let (start_block_number, mut archiver) = if let Some(last_segment_header) = last_segment_header
86    {
87        let first_block_number = last_segment_header.last_archived_block.number.as_inner();
88
89        let archiver = {
90            let Some(first_block_root) = chain_info
91                .ancestor_header(first_block_number, &best_block_root)
92                .map(|header| *header.root())
93            else {
94                return Ok(None);
95            };
96            let block = chain_info.block(&first_block_root).await?;
97            let shard_index = block.header().header().prefix.shard_index;
98            let encoded_block = encode_block(&block);
99
100            Archiver::with_initial_state(
101                shard_index,
102                erasure_coding,
103                last_segment_header,
104                &encoded_block,
105                extract_block_objects(&block),
106            )?
107        };
108
109        (
110            first_block_number.saturating_add(BlockNumber::ONE),
111            archiver,
112        )
113    } else {
114        let best_header = chain_info.best_header();
115        let archiver = Archiver::new(best_header.header().prefix.shard_index, erasure_coding);
116
117        (BlockNumber::ZERO, archiver)
118    };
119
120    for block_number in start_block_number.. {
121        let (encoded_block, block_objects) = {
122            let Some(block_root) = chain_info
123                .ancestor_header(block_number, &best_block_root)
124                .map(|header| *header.root())
125            else {
126                return Ok(None);
127            };
128            let block = chain_info.block(&block_root).await?;
129
130            (encode_block(&block), extract_block_objects(&block))
131        };
132
133        let task_fut = spawn_blocking(move || {
134            let encoded_block_size = encoded_block.len();
135            let maybe_outcome = archiver.add_block(encoded_block, block_objects).ok_or(
136                RecreateSegmentError::BlockTooLarge {
137                    block_number,
138                    size: encoded_block_size,
139                },
140            );
141
142            (archiver, maybe_outcome)
143        });
144        let outcome;
145        (archiver, outcome) = task_fut.await?;
146        let outcome = outcome?;
147
148        // TODO: Return global objects once archiver API improves
149        if let Some(mut archived_segment) = outcome.archived_segments.into_iter().next() {
150            for piece in archived_segment.pieces.iter_mut() {
151                piece
152                    .header
153                    .segment_proof
154                    .copy_from_slice(segment_proof.as_slice());
155            }
156
157            archived_segment.pieces = archived_segment.pieces.to_shared();
158
159            return Ok(Some(archived_segment));
160        }
161    }
162
163    Ok(None)
164}
