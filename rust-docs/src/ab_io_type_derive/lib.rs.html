7/// Derive macro for `TrivialType` trait of `ab-io-type` crate.
8///
9/// See the documentation of the trait for supported types and requirements, which are checked at
10/// compile time.
11#[proc_macro_derive(TrivialType, attributes(trivial_type))]
12pub fn trivial_type_derive(input: proc_macro::TokenStream) -> proc_macro::TokenStream {
13    let input = parse_macro_input!(input as DeriveInput);
14
15    if !input.generics.params.is_empty() {
16        return Error::new(
17            input.ident.span(),
18            "`TrivialType` can't be derived on generic types",
19        )
20        .into_compile_error()
21        .into();
22    }
23
24    let validate = match parse_validate(&input) {
25        Ok(validate) => validate,
26        Err(error) => {
27            return error.to_compile_error().into();
28        }
29    };
30
31    let type_name = &input.ident;
32
33    let output = match &input.data {
34        Data::Struct(data_struct) => {
35            if let Err(error) = check_repr(
36                &input,
37                &["C", "transparent"],
38                "`TrivialType` on structs requires `#[repr(C)]` or `#[repr(transparent)]` without \
39                other options",
40            ) {
41                return error.to_compile_error().into();
42            }
43            let field_types = data_struct
44                .fields
45                .iter()
46                .map(|field| &field.ty)
47                .collect::<Vec<_>>();
48
49            let struct_metadata = match generate_struct_metadata(type_name, data_struct) {
50                Ok(struct_metadata) => struct_metadata,
51                Err(error) => {
52                    return error.to_compile_error().into();
53                }
54            };
55
56            // Fields are parsed in order: they follow each other without padding (asserted below),
57            // so each field's step consumes exactly the bytes of that field, and the bytes are a
58            // valid value of the struct when every field is a valid value
59            let checks = quote! {
60                let fields = bytes;
61                #(
62                    let (_value, fields) =
63                        <#field_types as ::ab_io_type::trivial_type::TrivialType>
64                            ::try_from_bytes_unaligned(fields)?;
65                )*
66                // Empty, there is no padding after the last field
67                let _: &[::core::primitive::u8] = fields;
68            };
69            let parsing_step = generate_parsing_step(checks, validate.as_ref());
70
71            // A `validate` function rejects some values whose fields are all valid, so such a type
72            // doesn't accept every bit pattern even if its fields do. Bounds are higher-ranked, so
73            // that a field type without the marker makes the bound false instead of being a
74            // "trivial bound" error. They also contain no type parameter, which keeps a future
75            // `try_as_dyn()` probe working.
76            let marker_impl = if validate.is_some() {
77                TokenStream::new()
78            } else {
79                quote! {
80                    // SAFETY: Padding is rejected by the assertions generated with the
81                    // `TrivialType` impl, so the bytes of a value are exactly the bytes of its
82                    // fields, and the impl only applies when every field accepts any bit pattern.
83                    // The parsing step only calls parsing steps of fields after checking the
84                    // length, which accept every pattern of the field's bytes, and there is no
85                    // `validate` function.
86                    #[automatically_derived]
87                    unsafe impl ::ab_io_type::trivial_type::AnyBitPatternMarker for #type_name
88                    where
89                        #(
90                            for<'__ab_io_type> #field_types:
91                                ::ab_io_type::trivial_type::AnyBitPatternMarker,
92                        )*
93                    {
94                    }
95                }
96            };
97
98            quote! {
99                const _: () = {
100                    // Assert statically that there is no unexpected padding that would be left
101                    // uninitialized and unsound to access
102                    ::core::assert!(
103                        0 == (
104                            ::core::mem::size_of::<#type_name>()
105                            #(- ::core::mem::size_of::<#field_types>() )*
106                        ),
107                        "Struct must not have implicit padding. Consider reordering fields, adding \
108                        `padding: [u8; N]` field where necessary or use `Unaligned<T>` wrapper for \
109                        types with larger alignment to reduce it to one byte."
110                    );
111
112                    // Assert that type doesn't exceed 32-bit size limit
113                    ::core::assert!(
114                        ::core::primitive::u32::MAX as ::core::primitive::usize
115                            >= ::core::mem::size_of::<#type_name>(),
116                        "Type size must be smaller than 2^32"
117                    );
118
119                    // Ensure metadata decodes completely, and capacity and alignment are correctly
120                    // decoded from it. Every level of nesting takes at least one byte of metadata,
121                    // so its length is enough as the max nesting depth.
122                    let (type_details, remaining_metadata) =
123                        ::ab_io_type::metadata::IoTypeMetadataKind::type_details::<{
124                            <#type_name as ::ab_io_type::trivial_type::TrivialType>::METADATA.len()
125                        }>(
126                            <#type_name as ::ab_io_type::trivial_type::TrivialType>::METADATA,
127                        )
128                            .expect(
129                                "Metadata must decode, which is not the case for layouts with \
130                                padding or misaligned fields"
131                            );
132                    ::core::assert!(remaining_metadata.is_empty());
133                    ::core::assert!(
134                        ::core::mem::size_of::<#type_name>()
135                            == type_details.recommended_capacity as ::core::primitive::usize
136                    );
137                    ::core::assert!(
138                        ::core::mem::align_of::<#type_name>()
139                            == type_details.alignment.get() as ::core::primitive::usize
140                    );
141                };
142
143                #[automatically_derived]
144                unsafe impl ::ab_io_type::trivial_type::TrivialType for #type_name
145                where
146                    #( #field_types: ::ab_io_type::trivial_type::TrivialType, )*
147                {
148                    const METADATA: &[::core::primitive::u8] = #struct_metadata;
149
150                    #parsing_step
151                }
152
153                #marker_impl
154            }
155        }
156        Data::Enum(data_enum) => {
157            // Require defined size of the discriminant instead of allowing compiler to guess. With
158            // `#[repr(u8)]` each variant is laid out like a `#[repr(C)]` struct with the
159            // discriminant as the first field, which padding assertions rely on, `#[repr(C, u8)]`
160            // has a different layout.
161            if let Err(error) = check_repr(
162                &input,
163                &["u8"],
164                "`TrivialType` on enums requires `#[repr(u8)]` without other options",
165            ) {
166                return error.to_compile_error().into();
167            }
168
169            // Metadata identifies variants by their index, so explicit discriminants must be equal
170            // to it, which makes implicit ones equal to it too. Assertions are associated constants
171            // of the enum, so `Self` in a discriminant means the same as in the enum definition.
172            // They are only evaluated when used, so each is used by its own `const _`, which
173            // reports all wrong discriminants.
174            let discriminant_assertions = data_enum
175                .variants
176                .iter()
177                // `#[repr(u8)]` allows at most 256 variants
178                .zip(0..=u8::MAX)
179                .filter_map(|(variant, index)| {
180                    let (_eq_token, discriminant) = variant.discriminant.as_ref()?;
181                    // Errors point at the discriminant
182                    let span = Span::call_site().located_at(discriminant.span());
183                    let name = format_ident!("__TRIVIAL_TYPE_DISCRIMINANT_{index}");
184                    let message = format!(
185                        "Discriminant of `{}` must be equal to its index {index}, variants are \
186                        identified by their index in `TrivialType` metadata",
187                        variant.ident
188                    );
189
190                    Some(quote_spanned! {span=>
191                        impl #type_name {
192                            // The index goes first, so the discriminant is inferred as `u8` like
193                            // in the enum definition
194                            #[doc(hidden)]
195                            const #name: () = ::core::assert!(#index == (#discriminant), #message);
196                        }
197
198                        const _: () = #type_name::#name;
199                    })
200                })
201                .collect::<Vec<_>>();
202
203            let field_types = data_enum
204                .variants
205                .iter()
206                .flat_map(|variant| &variant.fields)
207                .map(|field| &field.ty)
208                .collect::<Vec<_>>();
209
210            let padding_assertions = data_enum.variants.iter().map(|variant| {
211                let field_types = variant.fields.iter().map(|field| &field.ty);
212
213                quote! {
214                    // Assert statically that there is no unexpected padding that would be left
215                    // uninitialized and unsound to access
216                    ::core::assert!(
217                        0 == (
218                            ::core::mem::size_of::<#type_name>()
219                            - ::core::mem::size_of::<::core::primitive::u8>()
220                            #(- ::core::mem::size_of::<#field_types>() )*
221                        ),
222                        "Enum must not have implicit padding. Consider reordering fields, adding \
223                        `padding: [u8; N]` field where necessary or use `Unaligned<T>` wrapper for \
224                        types with larger alignment to reduce it to one byte."
225                    );
226                }
227            });
228
229            // The discriminant must be the index of a variant, which is its discriminant (asserted
230            // above), then fields of that variant are parsed in order. Each variant is laid out
231            // like a `#[repr(C)]` struct with the discriminant as the first field, without padding
232            // (asserted below), so each field's step consumes exactly the bytes of that field, and
233            // the bytes are a valid value of the enum when every field of the variant is a valid
234            // value.
235            let variant_arms = data_enum
236                .variants
237                .iter()
238                // `#[repr(u8)]` allows at most 256 variants
239                .zip(0..=u8::MAX)
240                .map(|(variant, index)| {
241                    let field_types = variant.fields.iter().map(|field| &field.ty);
242
243                    quote! {
244                        #index => {
245                            #(
246                                let (_value, fields) =
247                                    <#field_types as ::ab_io_type::trivial_type::TrivialType>
248                                        ::try_from_bytes_unaligned(fields)?;
249                            )*
250                            // Empty, there is no padding after the last field
251                            let _: &[::core::primitive::u8] = fields;
252                        }
253                    }
254                });
255            let checks = quote! {
256                let (&discriminant, fields) = <[::core::primitive::u8]>::split_first(bytes)?;
257                match discriminant {
258                    #( #variant_arms )*
259                    _ => {
260                        return ::core::option::Option::None;
261                    }
262                }
263            };
264            let parsing_step = generate_parsing_step(checks, validate.as_ref());
265
266            let enum_metadata = match generate_enum_metadata(type_name, data_enum) {
267                Ok(struct_metadata) => struct_metadata,
268                Err(error) => {
269                    return error.to_compile_error().into();
270                }
271            };
272
273            quote! {
274                #( #discriminant_assertions )*
275
276                const _: () = {
277                    #( #padding_assertions )*
278
279                    // Assert that type doesn't exceed 32-bit size limit
280                    ::core::assert!(
281                        ::core::primitive::u32::MAX as ::core::primitive::usize
282                            >= ::core::mem::size_of::<#type_name>(),
283                        "Type size must be smaller than 2^32"
284                    );
285
286                    // Ensure metadata decodes completely, and capacity and alignment are correctly
287                    // decoded from it. Every level of nesting takes at least one byte of metadata,
288                    // so its length is enough as the max nesting depth.
289                    let (type_details, remaining_metadata) =
290                        ::ab_io_type::metadata::IoTypeMetadataKind::type_details::<{
291                            <#type_name as ::ab_io_type::trivial_type::TrivialType>::METADATA.len()
292                        }>(
293                            <#type_name as ::ab_io_type::trivial_type::TrivialType>::METADATA,
294                        )
295                            .expect(
296                                "Metadata must decode, which is not the case for layouts with \
297                                padding or misaligned fields"
298                            );
299                    ::core::assert!(remaining_metadata.is_empty());
300                    ::core::assert!(
301                        ::core::mem::size_of::<#type_name>()
302                            == type_details.recommended_capacity as ::core::primitive::usize
303                    );
304                    ::core::assert!(
305                        ::core::mem::align_of::<#type_name>()
306                            == type_details.alignment.get() as ::core::primitive::usize
307                    );
308                };
309
310                #[automatically_derived]
311                unsafe impl ::ab_io_type::trivial_type::TrivialType for #type_name
312                where
313                    #( #field_types: ::ab_io_type::trivial_type::TrivialType, )*
314                {
315                    const METADATA: &[::core::primitive::u8] = #enum_metadata;
316
317                    #parsing_step
318                }
319            }
320        }
321        Data::Union(data_union) => {
322            return Error::new(
323                data_union.union_token.span(),
324                "`TrivialType` can be derived for structs and enums, but not unions",
325            )
326            .to_compile_error()
327            .into();
328        }
329    };
330
331    output.into()
332}

334/// Parse `#[trivial_type(..)]` attributes, returns the path of the `validate` function if there is
335/// one
336fn parse_validate(input: &DeriveInput) -> Result<Option<ExprPath>, Error> {
337    // Registering the helper attribute makes it legal on fields and variants too, where the derive
338    // would ignore it, and an ignored `validate` lets parsing create values that `unsafe` code
339    // relies on never existing
340    let nested_attrs = match &input.data {
341        Data::Struct(data_struct) => data_struct
342            .fields
343            .iter()
344            .flat_map(|field| &field.attrs)
345            .collect::<Vec<_>>(),
346        Data::Enum(data_enum) => data_enum
347            .variants
348            .iter()
349            .flat_map(|variant| {
350                variant
351                    .attrs
352                    .iter()
353                    .chain(variant.fields.iter().flat_map(|field| &field.attrs))
354            })
355            .collect::<Vec<_>>(),
356        Data::Union(data_union) => data_union
357            .fields
358            .named
359            .iter()
360            .flat_map(|field| &field.attrs)
361            .collect::<Vec<_>>(),
362    };
363    if let Some(attr) = nested_attrs
364        .into_iter()
365        .find(|attr| attr.path().is_ident("trivial_type"))
366    {
367        return Err(Error::new_spanned(
368            attr,
369            "`#[trivial_type(..)]` is only supported on the type",
370        ));
371    }
372
373    let mut validate = None;
374    let attrs = input
375        .attrs
376        .iter()
377        .filter(|attr| attr.path().is_ident("trivial_type"));
378    for attr in attrs {
379        let mut has_options = false;
380        attr.parse_nested_meta(|meta| {
381            has_options = true;
382            if !meta.path.is_ident("validate") {
383                return Err(meta.error("Unsupported `#[trivial_type(..)]` option"));
384            }
385            if validate.is_some() {
386                return Err(meta.error("`validate` is specified more than once"));
387            }
388            // Parsed like a path in an expression, so generic arguments are written as `::<..>`
389            // and a qualified self type is supported, like in the generated call
390            let path = meta.value()?.parse::<ExprPath>()?;
391            if let Some(attr) = path.attrs.first() {
392                return Err(Error::new_spanned(
393                    attr,
394                    "Attributes are not supported in `validate = path`",
395                ));
396            }
397            validate = Some(path);
398            Ok(())
399        })?;
400
401        if !has_options {
402            return Err(Error::new_spanned(
403                attr,
404                "`#[trivial_type(..)]` requires an option, like `validate = path`",
405            ));
406        }
407    }
408
409    Ok(validate)
410}

412/// Generate the parsing step of `TrivialType`, which checks the size, runs `checks` with `bytes` of
413/// the size of the type in scope, which must return `None` for invalid bytes, casts `bytes` to the
414/// type and calls the `validate` function with the value if there is one
415fn generate_parsing_step(checks: TokenStream, validate: Option<&ExprPath>) -> TokenStream {
416    let validation = if let Some(path) = validate {
417        // Errors, like a wrong signature or an `unsafe` function, point at the path in the
418        // attribute
419        let span = Span::call_site().located_at(path.span());
420        quote_spanned! {span=>
421            if !#path(::ab_io_type::unaligned::Unaligned::<Self>::as_inner(__ab_io_type_value)) {
422                return ::core::option::Option::None;
423            }
424        }
425    } else {
426        TokenStream::new()
427    };
428
429    // The value is the only local of the step with a prefixed name: a constant with the same name
430    // in the module of the type would turn the binding into a constant pattern, which compiles for
431    // types with a single value, and the step would return the constant instead of a reference
432    // into `bytes`. Constants named like the other locals only cause compile errors.
433    quote! {
434        #[inline(always)]
435        fn try_from_bytes_unaligned(
436            bytes: &[::core::primitive::u8],
437        ) -> ::core::option::Option<(
438            &::ab_io_type::unaligned::Unaligned<Self>,
439            &[::core::primitive::u8],
440        )> {
441            let (bytes, remainder) = <[::core::primitive::u8]>::split_at_checked(
442                bytes,
443                ::core::mem::size_of::<Self>(),
444            )?;
445            #checks
446            // SAFETY: `bytes` have the size of the type and `checks` return `None` unless every
447            // field, of the variant whose index is the discriminant for an enum, is a valid value.
448            // Variants have discriminants equal to their indices and fields follow each other
449            // without padding (both asserted next to the type), and `Unaligned<Self>` has
450            // alignment 1.
451            let __ab_io_type_value = unsafe {
452                &*<[::core::primitive::u8]>::as_ptr(bytes)
453                    .cast::<::ab_io_type::unaligned::Unaligned<Self>>()
454            };
455            // The `validate` function, if any, rejects values whose invariants don't hold. It is
456            // called outside of the `unsafe` block above, so it can't be an `unsafe` function.
457            #validation
458            ::core::option::Option::Some((__ab_io_type_value, remainder))
459        }
460    }
461}

463/// Check that `#[repr(..)]` attributes of the type contain at least one of `allowed` options and no
464/// other options, returns an error with `message` otherwise
465fn check_repr(input: &DeriveInput, allowed: &[&str], message: &str) -> Result<(), Error> {
466    let mut found_allowed = false;
467
468    // All `#[repr(..)]` attributes are combined by the compiler
469    let repr_attrs = input
470        .attrs
471        .iter()
472        .filter(|attr| attr.path().is_ident("repr"));
473    for repr_attr in repr_attrs {
474        repr_attr.parse_nested_meta(|meta| {
475            if allowed.iter().any(|option| meta.path.is_ident(option)) {
476                found_allowed = true;
477                Ok(())
478            } else {
479                Err(meta.error(message))
480            }
481        })?;
482    }
483
484    if !found_allowed {
485        return Err(Error::new(input.ident.span(), message));
486    }
487
488    Ok(())
489}

491fn generate_struct_metadata(ident: &Ident, data_struct: &DataStruct) -> Result<TokenStream, Error> {
492    let inner_struct_metadata = generate_inner_struct_metadata(ident, &data_struct.fields)
493        .collect::<Result<Vec<_>, _>>()?;
494
495    // Encodes the following:
496    // * Type: struct
497    // * The rest as inner struct metadata
498    Ok(quote! {{
499        #[inline(always)]
500        const fn metadata() -> (
501            [::core::primitive::u8; ::ab_io_type::metadata::MAX_METADATA_CAPACITY],
502            ::core::primitive::usize,
503        )
504        {
505            ::ab_io_type::metadata::concat_metadata_sources(&[
506                &[::ab_io_type::metadata::IoTypeMetadataKind::Struct as ::core::primitive::u8],
507                #( #inner_struct_metadata )*
508            ])
509        }
510
511        // The reference extends the lifetime of metadata scratch to that of the constant
512        let (metadata_scratch, metadata_size) = &metadata();
513        // `split_at()` is called by path, with method call syntax a method of a trait in scope that
514        // is implemented for arrays would be called instead
515        <[_]>::split_at(metadata_scratch, *metadata_size).0
516    }})
517}

519fn generate_enum_metadata(ident: &Ident, data_enum: &DataEnum) -> Result<TokenStream, Error> {
520    let type_name_string = ident.to_string();
521    let type_name_bytes = type_name_string.as_bytes();
522
523    let type_name_bytes_len = u8::try_from(type_name_bytes.len()).map_err(|_error| {
524        Error::new(
525            ident.span(),
526            format!(
527                "Name of the enum must not be more than {} bytes in length",
528                u8::MAX
529            ),
530        )
531    })?;
532    let num_variants = u8::try_from(data_enum.variants.len()).map_err(|_error| {
533        Error::new(
534            ident.span(),
535            format!("Enum must not have more than {} variants", u8::MAX),
536        )
537    })?;
538    // Encodes the following:
539    // * Type: enum
540    // * Length of enum name in bytes (u8)
541    // * Enum name as UTF-8 bytes
542    // * Number of variants (u8)
543    let enum_metadata_header = {
544        let enum_metadata_header = [Literal::u8_unsuffixed(type_name_bytes_len)]
545            .into_iter()
546            .chain(
547                type_name_bytes
548                    .iter()
549                    .map(|&char| Literal::byte_character(char)),
550            )
551            .chain([Literal::u8_unsuffixed(num_variants)]);
552
553        quote! {
554            &[
555                ::ab_io_type::metadata::IoTypeMetadataKind::Enum as ::core::primitive::u8,
556                #( #enum_metadata_header, )*
557            ]
558        }
559    };
560
561    // Encodes each variant as inner struct, fields of tuple variants have zero-length names like
562    // fields of tuple structs
563    let inner = data_enum
564        .variants
565        .iter()
566        .flat_map(|variant| generate_inner_struct_metadata(&variant.ident, &variant.fields))
567        .collect::<Result<Vec<TokenStream>, Error>>()?;
568
569    Ok(quote! {{
570        #[inline(always)]
571        const fn metadata() -> (
572            [::core::primitive::u8; ::ab_io_type::metadata::MAX_METADATA_CAPACITY],
573            ::core::primitive::usize,
574        )
575        {
576            ::ab_io_type::metadata::concat_metadata_sources(&[
577                #enum_metadata_header,
578                #( #inner )*
579            ])
580        }
581
582        // The reference extends the lifetime of metadata scratch to that of the constant
583        let (metadata_scratch, metadata_size) = &metadata();
584        // `split_at()` is called by path, with method call syntax a method of a trait in scope that
585        // is implemented for arrays would be called instead
586        <[_]>::split_at(metadata_scratch, *metadata_size).0
587    }})
588}

590fn generate_inner_struct_metadata<'a>(
591    ident: &'a Ident,
592    fields: &'a Fields,
593) -> impl Iterator<Item = Result<TokenStream, Error>> + 'a {
594    iter::once_with(move || generate_inner_struct_metadata_header(ident, fields))
595        .chain(generate_fields_metadata(fields))
596}
597
598fn generate_inner_struct_metadata_header(
599    ident: &Ident,
600    fields: &Fields,
601) -> Result<TokenStream, Error> {
602    let ident_string = ident.to_string();
603    let ident_bytes = ident_string.as_bytes();
604
605    let ident_bytes_len = u8::try_from(ident_bytes.len()).map_err(|_error| {
606        Error::new(
607            ident.span(),
608            format!(
609                "Identifier must not be more than {} bytes in length",
610                u8::MAX
611            ),
612        )
613    })?;
614    let num_fields = u8::try_from(fields.len()).map_err(|_error| {
615        Error::new(
616            fields.span(),
617            format!("Must not have more than {} fields", u8::MAX),
618        )
619    })?;
620
621    // Encodes the following:
622    // * Length of identifier in bytes (u8)
623    // * Identifier as UTF-8 bytes
624    // * Number of fields (u8)
625    Ok({
626        let struct_metadata_header = [Literal::u8_unsuffixed(ident_bytes_len)]
627            .into_iter()
628            .chain(
629                ident_bytes
630                    .iter()
631                    .map(|&char| Literal::byte_character(char)),
632            )
633            .chain([Literal::u8_unsuffixed(num_fields)]);
634
635        quote! {
636            &[#( #struct_metadata_header, )*],
637        }
638    })
639}
640
641fn generate_fields_metadata(
642    fields: &Fields,
643) -> impl Iterator<Item = Result<TokenStream, Error>> + '_ {
644    // Encodes the following for each field:
645    // * Length of the field name in bytes (u8, zero for tuple fields)
646    // * Field name as UTF-8 bytes
647    // * Recursive metadata of the field's type
648    fields.iter().map(move |field| {
649        let field_metadata = if let Some(field_name) = &field.ident {
650            let field_name_string = field_name.to_string();
651            let field_name_bytes = field_name_string.as_bytes();
652            let field_name_len = u8::try_from(field_name_bytes.len()).map_err(|_error| {
653                Error::new(
654                    field.span(),
655                    format!(
656                        "Name of the field must not be more than {} bytes in length",
657                        u8::MAX
658                    ),
659                )
660            })?;
661
662            let field_metadata = [Literal::u8_unsuffixed(field_name_len)].into_iter().chain(
663                field_name_bytes
664                    .iter()
665                    .map(|&char| Literal::byte_character(char)),
666            );
667
668            quote! { #( #field_metadata, )* }
669        } else {
670            quote! { 0, }
671        };
672        let field_type = &field.ty;
673
674        Ok(quote! {
675            &[ #field_metadata ],
676            <#field_type as ::ab_io_type::trivial_type::TrivialType>::METADATA,
677        })
678    })
679}
