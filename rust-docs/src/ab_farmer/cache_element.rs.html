1//! Elements of piece caches stored on disk.
2//!
3//! An element contains piece index, piece and checksum of both, followed by zero padding to whole
4//! pages, such that elements can be written with direct I/O without read-modify-write. An element
5//! that only contains zeroes is empty.
6
7use ab_aligned_buffer::alignment::Align4096;
8use ab_core_primitives::hashes::Blake3Hash;
9use ab_core_primitives::pieces::{Piece, PieceIndex};
10use ab_direct_io_file::OwnedPages;
11use std::ops::Range;
12use tracing::debug;
13
14/// Range of the piece within an element in bytes
15const PIECE_RANGE: Range<u32> = {
16    assert!(PieceIndex::SIZE + Piece::SIZE <= u32::MAX as usize);
17    PieceIndex::SIZE as u32..(PieceIndex::SIZE + Piece::SIZE) as u32
18};
19/// Size of piece index, piece and checksum in bytes
20const CONTENTS_SIZE: u32 = {
21    let size = PieceIndex::SIZE + Piece::SIZE + Blake3Hash::SIZE;
22    assert!(size <= u32::MAX as usize);
23    size as u32
24};
25/// Size of a cache element in pages
26pub(crate) const ELEMENT_PAGES: u32 = CONTENTS_SIZE.div_ceil(OwnedPages::ALIGN);
27/// Size of a cache element in bytes, a whole number of pages
28pub(crate) const ELEMENT_SIZE: u32 = ELEMENT_PAGES * OwnedPages::ALIGN;
29
30/// Checksum of a cache element doesn't match its contents
31#[derive(Debug, Copy, Clone, thiserror::Error)]
32#[error("Checksum mismatch")]
33pub(crate) struct ChecksumMismatch;
34
35/// Encode an element with `piece` stored under `piece_index`
36pub(crate) fn encode(piece_index: PieceIndex, piece: &Piece) -> OwnedPages {
37    let mut element = OwnedPages::with_capacity(ELEMENT_SIZE);
38    element
39        .append(&piece_index.to_bytes())
40        .expect("Capacity for the whole element is reserved above; qed");
41    element
42        .append(piece.as_ref())
43        .expect("Capacity for the whole element is reserved above; qed");
44    let checksum = blake3::hash(element.as_slice());
45    element
46        .append(checksum.as_bytes())
47        .expect("Capacity for the whole element is reserved above; qed");
48    element
49        .pad_to_alignment::<Align4096>()
50        .expect("Capacity for the whole element is reserved above; qed");
51    element
52}

54/// Verify the checksum of an element and return the piece index stored in it, `None` means the
55/// element is empty
56pub(crate) fn verify_checksum(
57    element: &[u8; ELEMENT_SIZE as usize],
58) -> Result<Option<PieceIndex>, ChecksumMismatch> {
59    let (piece_index_bytes, remaining_bytes) = element.split_at(PieceIndex::SIZE);
60    let (piece_bytes, remaining_bytes) = remaining_bytes.split_at(Piece::SIZE);
61    // The rest of the element after the checksum is zero padding
62    let (expected_checksum, _padding) = remaining_bytes.split_at(Blake3Hash::SIZE);
63
64    // Verify checksum
65    let actual_checksum = {
66        let mut hasher = blake3::Hasher::new();
67        hasher.update(piece_index_bytes);
68        hasher.update(piece_bytes);
69        *hasher.finalize().as_bytes()
70    };
71    if actual_checksum != expected_checksum {
72        if element.iter().all(|&byte| byte == 0) {
73            return Ok(None);
74        }
75
76        debug!(
77            actual_checksum = %hex::encode(actual_checksum),
78            expected_checksum = %hex::encode(expected_checksum),
79            "Hash doesn't match, corrupted piece in cache"
80        );
81
82        return Err(ChecksumMismatch);
83    }
84
85    let piece_index = PieceIndex::from_bytes(
86        piece_index_bytes
87            .try_into()
88            .expect("Statically known to have correct size; qed"),
89    );
90    Ok(Some(piece_index))
91}

93/// Convert an element into the piece stored in it without copying, see [`verify_checksum()`] for
94/// checking whether it contains one.
95///
96/// Returns `None` if the element is shorter than [`ELEMENT_SIZE`].
97pub(crate) fn into_piece(element: OwnedPages) -> Option<Piece> {
98    let piece = element.into_shared().get_slice(PIECE_RANGE)?;
99    Piece::try_from(piece).ok()
100}
