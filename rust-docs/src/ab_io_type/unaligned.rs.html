10/// Wrapper type for `Data` that is the same size, but doesn't need to be aligned/has alignment of
11/// one byte.
12///
13/// This is similar to `#[repr(packed)]`, but makes sure to only expose safe API when dealing with
14/// the contents. For example, if `Data` is unaligned and has fields, it is not sounds having
15/// references to its fields. This data structure prevents such invalid invariants.
16#[derive(Debug, Default, Copy, Clone, Ord, PartialOrd, Eq, PartialEq, Hash)]
17#[cfg_attr(feature = "scale-codec", derive(Decode, MaxEncodedLen))]
18#[cfg_attr(feature = "serde", derive(Deserialize))]
19#[cfg_attr(feature = "serde", serde(transparent))]
20#[repr(C, packed)]
21pub struct Unaligned<Data>(Data)
22where
23    Data: TrivialType;
24

46// SAFETY: Same bytes and validity as `Data`, the parsing step parses `Data`
47unsafe impl<Data> TrivialType for Unaligned<Data>
48where
49    Data: TrivialType,
50{
51    const ALIGNMENT: Alignment = u8::ALIGNMENT;
52    const METADATA: &[u8] = {
53        const fn metadata(inner_metadata: &[u8]) -> ([u8; MAX_METADATA_CAPACITY], usize) {
54            concat_metadata_sources(&[&[IoTypeMetadataKind::Unaligned as u8], inner_metadata])
55        }
56
57        // The reference extends the lifetime of metadata scratch to that of the constant
58        let (metadata_scratch, metadata_size) = &metadata(Data::METADATA);
59        metadata_scratch.split_at(*metadata_size).0
60    };
61
62    #[inline(always)]
63    unsafe fn parse_step<const ALIGNED: bool>(bytes: &[u8]) -> Option<(&Unaligned<Self>, &[u8])> {
64        // SAFETY: The unaligned flavour has no requirements. It is used even if `ALIGNED`, since
65        // `Self` has alignment 1 and `bytes` aren't necessarily aligned for `Data` then.
66        let (_value, remainder) = unsafe { Data::parse_step::<false>(bytes) }?;
67        // SAFETY: `bytes` start with a valid value of `Data` (parsing step contract), which has the
68        // same bytes and validity as `Self`, `Unaligned<Self>` has alignment 1 and the size of
69        // `Self`
70        let value = unsafe { bytes.as_ptr().cast::<Unaligned<Self>>().as_ref_unchecked() };
71        Some((value, remainder))
72    }
73}
