29// TODO: Make this a `const` trait, with `const` impls generated by the derive, once const traits

32/// Simple wrapper data type that is designed in such a way that its serialization/deserialization
33/// is the same as the type itself.
34///
35/// Serialization takes the bytes of a value. Deserialization parses bytes: the parsing step checks
36/// that they are a valid value of the type, which is a length check for types that accept every
37/// bit pattern (see [`AnyBitPattern`]). Bytes that are not aligned for the type are parsed with the
38/// methods of [`Unaligned`], like `Unaligned::<T>::try_from_bytes()`.
39///
40/// A valid value is one whose every field is a valid value of its type, like `0` or `1` for a
41/// `bool` and the index of a variant for the discriminant of an enum, and that satisfies the
42/// invariants `unsafe` code relies on. Parsing may reject more, like a length of
43/// `FixedCapacityElements` above its capacity, but doesn't check encoding rules, like the bytes of
44/// its length prefix after the length.
45///
46/// # Safety
47/// * There is no padding, every byte is initialized and there is no interior mutability
48/// * The alignment is at most [`Alignment::MAX`]
49/// * [`Self::SIZE`] is not overridden
50/// * [`Self::ALIGNMENT`] is the alignment that [`Self::METADATA`] decodes to, and it is at least
51///   `align_of::<Self>()`
52/// * The parsing step `parse_step::<ALIGNED>()` returns `Some((value, remainder))` only if `bytes`
53///   start with `size_of::<Self>()` bytes that are a valid value of `Self`, `value` points to the
54///   beginning of `bytes` and `remainder` is the rest of `bytes` after those bytes. This holds for
55///   both flavours: the aligned one (`ALIGNED` is `true`) gets `bytes` aligned for `Self`, the
56///   unaligned one gets `bytes` of any alignment
57///
58/// Implementations are expected to be composites of existing `TrivialType`s without custom
59/// alignment, whose alignment is then at most [`Alignment::MAX`] as well. Prefer
60/// `#[derive(TrivialType)]`, which upholds this contract.
61///
62/// # Deriving
63/// `#[derive(TrivialType)]` supports structs and enums and checks the following requirements at
64/// compile time:
65/// * no generic parameters, and types of all fields implement `TrivialType`
66/// * structs are `#[repr(C)]` or `#[repr(transparent)]` without other options
67/// * enums are `#[repr(u8)]` without other options
68/// * no `#[repr(align(N))]` or `#[repr(packed)]` in any `#[repr(..)]` attribute, [`Unaligned`] can
69///   be used to reduce the alignment of a field to one byte instead
70/// * no implicit padding, explicit `padding: [u8; N]` fields can be added where necessary. Padding
71///   is determined by [`TrivialType::ALIGNMENT`] of the field types. Each variant of an enum is
72///   laid out like a `#[repr(C)]` struct with the discriminant as the first field, so fields of all
73///   variants must have the same total size
74/// * explicit discriminants of enum variants are equal to the index of the variant, metadata
75///   identifies variants by their index and doesn't record discriminants
76/// * at most 255 fields in a struct or enum variant and at most 255 variants in an enum, names of
77///   the type, its fields and variants are at most 255 bytes long
78/// * size is smaller than 2^32 bytes
79/// * metadata, including metadata of field types, is at most [`MAX_METADATA_CAPACITY`] bytes,
80///   otherwise const evaluation panics with an opaque error
81/// * structs parse their fields in order, enums check the discriminant and parse the fields of that
82///   variant
83/// * deriving means that any combination of valid field values is a valid value of the type,
84///   private fields and `#[non_exhaustive]` included, because parsing creates values without
85///   calling constructors, unless `#[trivial_type(validate = path)]` rejects it
86/// * structs implement [`AnyBitPatternMarker`] when all their fields do, unless they use
87///   `#[trivial_type(validate = path)]`
88///
89/// Invariants that `unsafe` code relies on are checked with `#[trivial_type(validate = path)]`.
90/// `path` is called as `path(value: &Self) -> bool` after all fields were parsed and must return
91/// `false` for every value that `unsafe` code relying on the invariants can't handle. It must not
92/// panic, since parsing is used in code that is verified not to panic, and a `const fn` is
93/// recommended, the derive may become `const` later. Such types don't implement
94/// [`AnyBitPatternMarker`].
95///
96/// It is called by every parsing method, including those of arrays of the type, of [`Unaligned`]
97/// and of containers through [`IoType::try_from_ptr()`] and [`IoType::try_from_mut_ptr()`]. The
98/// unsafe methods that skip parsing, [`IoType::from_ptr()`], [`IoType::from_mut_ptr()`] and derives
99/// of other crates, like SCALE's `Decode` and serde's `Deserialize`, don't call it.
100pub unsafe trait TrivialType
101where
102    Self: Copy + 'static,
103{
104    // TODO: Make it `final` once supported:
105    //  https://internals.rust-lang.org/t/final-associated-constants/24655
106    const SIZE: u32 = {
107        assert!(
108            u32::MAX as usize >= size_of::<Self>(),
109            "Type size must be smaller than 2^32"
110        );
111        size_of::<Self>() as u32
112    };
113    /// Alignment of the type in metadata.
114    ///
115    /// Integers and `bool` are aligned to their size, `()` and [`Unaligned`] to 1 byte, other types
116    /// to the largest alignment of their contents. It is never smaller than `align_of::<Self>()`,
117    /// so memory aligned to it is aligned for `Self`.
118    const ALIGNMENT: Alignment;
119    /// Data structure metadata in binary form, describing shape and types of the contents, see
120    /// [`IoTypeMetadataKind`] for encoding details.
121    const METADATA: &[u8];
122
123    /// Parse the beginning of `bytes`, which are aligned for `Self` if `ALIGNED`.
124    ///
125    /// Returns the value and the remaining bytes, or `None` if `bytes` are shorter than the type or
126    /// don't start with a valid value. `#[derive(TrivialType)]` implements it, the other parsing
127    /// methods are built on it. Values in memory that is not aligned for `Self` are read with the
128    /// methods of [`Unaligned<Self>`], like `Unaligned::<Self>::try_from_bytes()`.
129    ///
130    /// Both flavours must accept the same values. The aligned one can use aligned loads, which
131    /// matters on targets without fast unaligned access, where unaligned loads wider than a byte
132    /// compile to several byte loads.
133    ///
134    /// # Safety
135    /// If `ALIGNED`, `bytes` are aligned for `Self`.
136    #[doc(hidden)]
137    unsafe fn parse_step<const ALIGNED: bool>(bytes: &[u8]) -> Option<(&Unaligned<Self>, &[u8])>;
138
139    /// Parse the beginning of aligned `bytes`, returns the value and the remaining bytes.
140    ///
141    /// Returns `None` if `bytes` are not aligned for `Self`, are too short or don't start with a
142    /// valid value.
143    #[inline(always)]
144    final fn try_from_bytes(bytes: &[u8]) -> Option<(&Self, &[u8])> {
145        if !bytes.as_ptr().cast::<Self>().is_aligned() {
146            return None;
147        }
148        // SAFETY: Aligned as checked above
149        let (_value, remainder) = unsafe { Self::parse_step::<true>(bytes) }?;
150        // SAFETY: `bytes` start with a valid value of `Self` (parsing step contract), aligned as
151        // checked above
152        let value = unsafe { bytes.as_ptr().cast::<Self>().as_ref_unchecked() };
153        Some((value, remainder))
154    }
155
156    /// Like [`Self::try_from_bytes()`], but returns `None` if there are more bytes than the value
157    #[inline(always)]
158    final fn try_from_bytes_exact(bytes: &[u8]) -> Option<&Self> {
159        let (value, remainder) = Self::try_from_bytes(bytes)?;
160        remainder.is_empty().then_some(value)
161    }
162
163    /// Like [`Self::try_from_bytes()`], but returns an exclusive reference.
164    ///
165    /// Only valid values of `Self` can be written through it, so the bytes stay a valid value.
166    #[inline(always)]
167    final fn try_from_bytes_mut(bytes: &mut [u8]) -> Option<(&mut Self, &mut [u8])> {
168        if !bytes.as_ptr().cast::<Self>().is_aligned() {
169            return None;
170        }
171        // SAFETY: Aligned as checked above
172        unsafe { Self::parse_step::<true>(bytes) }?;
173        let (value, remainder) = bytes.split_at_mut_checked(size_of::<Self>())?;
174        // SAFETY: `bytes` start with a valid value of `Self` (parsing step contract), aligned as
175        // checked above, only valid values of `Self` can be written through the reference, and
176        // they have no padding, so the bytes stay initialized. It is derived from `bytes`, not
177        // from the shared reference the parsing step returned.
178        let value = unsafe { value.as_mut_ptr().cast::<Self>().as_mut_unchecked() };
179        Some((value, remainder))
180    }
181
182    /// Like [`Self::try_from_bytes_mut()`], but returns `None` if there are more bytes than the
183    /// value
184    #[inline(always)]
185    final fn try_from_bytes_mut_exact(bytes: &mut [u8]) -> Option<&mut Self> {
186        let (value, remainder) = Self::try_from_bytes_mut(bytes)?;
187        remainder.is_empty().then_some(value)
188    }
189
190    /// Create a reference to a value at the beginning of aligned `bytes`, returns the value and
191    /// the remaining bytes.
192    ///
193    /// Returns `None` if `bytes` are not aligned for `Self` or are too short. Unlike
194    /// [`Self::try_from_bytes()`] it doesn't parse the bytes, since any bytes are a valid value.
195    #[inline(always)]
196    final fn from_bytes(bytes: &[u8]) -> Option<(&Self, &[u8])>
197    where
198        Self: AnyBitPattern,
199    {
200        if !bytes.as_ptr().cast::<Self>().is_aligned() {
201            return None;
202        }
203        let (value, remainder) = bytes.split_at_checked(size_of::<Self>())?;
204        // SAFETY: `value` is `size_of::<Self>()` initialized bytes, aligned as checked above, and
205        // any of them are a valid value of `Self` (`AnyBitPatternMarker`)
206        let value = unsafe { value.as_ptr().cast::<Self>().as_ref_unchecked() };
207        Some((value, remainder))
208    }
209
210    /// Like [`Self::from_bytes()`], but returns `None` if there are more bytes than the value
211    #[inline(always)]
212    final fn from_bytes_exact(bytes: &[u8]) -> Option<&Self>
213    where
214        Self: AnyBitPattern,
215    {
216        let (value, remainder) = Self::from_bytes(bytes)?;
217        remainder.is_empty().then_some(value)
218    }
219
220    /// Similar to [`Self::from_bytes()`], but for any type and without any checks
221    ///
222    /// # Safety
223    /// `bytes` are aligned for `Self` and start with a valid value of `Self`.
224    #[inline(always)]
225    final unsafe fn from_bytes_unchecked(bytes: &[u8]) -> &Self {
226        // SAFETY: Guaranteed by function contract
227        unsafe { bytes.as_ptr().cast::<Self>().as_ref_unchecked() }
228    }
229
230    /// Like [`Self::from_bytes()`], but returns an exclusive reference
231    #[inline(always)]
232    final fn from_bytes_mut(bytes: &mut [u8]) -> Option<(&mut Self, &mut [u8])>
233    where
234        Self: AnyBitPattern,
235    {
236        if !bytes.as_ptr().cast::<Self>().is_aligned() {
237            return None;
238        }
239        let (value, remainder) = bytes.split_at_mut_checked(size_of::<Self>())?;
240        // SAFETY: `value` is `size_of::<Self>()` initialized bytes, aligned as checked above, and
241        // any of them are a valid value of `Self` (`AnyBitPatternMarker`), values written through
242        // the reference have no padding, so the bytes stay initialized
243        let value = unsafe { value.as_mut_ptr().cast::<Self>().as_mut_unchecked() };
244        Some((value, remainder))
245    }
246
247    /// Like [`Self::from_bytes_mut()`], but returns `None` if there are more bytes than the value
248    #[inline(always)]
249    final fn from_bytes_mut_exact(bytes: &mut [u8]) -> Option<&mut Self>
250    where
251        Self: AnyBitPattern,
252    {
253        let (value, remainder) = Self::from_bytes_mut(bytes)?;
254        remainder.is_empty().then_some(value)
255    }
256
257    /// Similar to [`Self::from_bytes_mut()`], but for any type and without any checks
258    ///
259    /// # Safety
260    /// `bytes` are aligned for `Self` and start with a valid value of `Self`.
261    #[inline(always)]
262    final unsafe fn from_bytes_mut_unchecked(bytes: &mut [u8]) -> &mut Self {
263        // SAFETY: Guaranteed by function contract
264        unsafe { bytes.as_mut_ptr().cast::<Self>().as_mut_unchecked() }
265    }
266
267    /// Access the underlying byte representation of a data structure
268    #[inline(always)]
269    final fn as_bytes(&self) -> &[u8; SIZE_OF::<Self>] {
270        // SAFETY: All bits are valid for reading as bytes, see `TrivialType` description
271        unsafe { ptr::from_ref(self).cast::<[u8; _]>().as_ref_unchecked() }
272    }
273
274    /// Access the underlying mutable byte representation of a data structure
275    #[inline(always)]
276    final fn as_bytes_mut(&mut self) -> &mut [u8; SIZE_OF::<Self>]
277    where
278        Self: AnyBitPattern,
279    {
280        // SAFETY: Any bytes written are a valid value of `Self` (`AnyBitPatternMarker`)
281        unsafe { self.as_bytes_mut_unchecked() }
282    }
283
284    /// Similar to [`Self::as_bytes_mut()`], but for any type.
285    ///
286    /// # Safety
287    /// Only bytes of a valid value of `Self` may be written to the returned array.
288    #[inline(always)]
289    final unsafe fn as_bytes_mut_unchecked(&mut self) -> &mut [u8; SIZE_OF::<Self>] {
290        // SAFETY: All bits are valid for reading as bytes, see `TrivialType` description, only
291        // bytes of a valid value are written (function contract)
292        unsafe { ptr::from_mut(self).cast::<[u8; _]>().as_mut_unchecked() }
293    }
294}

297//  skips impls whose bounds mention a type parameter inside a constant, like the array size limit

300/// Every bit pattern of the type is a valid value.
301///
302/// Together with [`TrivialType`] it provides [`AnyBitPattern`], which is the bound to use.
303/// `#[derive(TrivialType)]` implements it for structs whose fields all implement it, unless they
304/// use `#[trivial_type(validate = path)]`.
305///
306/// # Safety
307/// Every pattern of `size_of::<Self>()` initialized bytes is a valid value of `Self` that no
308/// invariant of `Self` excludes, not even one that only safe code relies on (like a length that
309/// must not exceed a capacity, or one that `#[trivial_type(validate = path)]` checks), so bytes of
310/// `Self` never need to be checked. If `Self` implements [`TrivialType`], both flavours of its
311/// parsing step `parse_step::<ALIGNED>()` accept every such pattern.
312#[diagnostic::on_unimplemented(message = "`{Self}` doesn't accept every bit pattern")]
313pub unsafe trait AnyBitPatternMarker {}

315/// [`TrivialType`] whose every bit pattern is a valid value.
316///
317/// Implemented for every type that implements [`TrivialType`] and [`AnyBitPatternMarker`].
318#[diagnostic::on_unimplemented(message = "`{Self}` doesn't accept every bit pattern")]
319pub trait AnyBitPattern
320where
321    Self: TrivialType + AnyBitPatternMarker,

330/// Parsing step of a type whose every bit pattern is a valid value
331#[inline(always)]
332fn try_from_bytes_any_bit_pattern<T>(bytes: &[u8]) -> Option<(&Unaligned<T>, &[u8])>
333where
334    T: TrivialType + AnyBitPatternMarker,
335{
336    let (bytes, remainder) = bytes.split_at_checked(size_of::<T>())?;
337    // SAFETY: `Unaligned<T>` has alignment 1 and the size of `T`, `bytes` are that many
338    // initialized bytes, and any of them are a valid value of `T` (`AnyBitPatternMarker`)
339    let value = unsafe { bytes.as_ptr().cast::<Unaligned<T>>().as_ref_unchecked() };
340    Some((value, remainder))
341}

343/// Returns the bytes after `count` valid values of `T` at the beginning of `bytes`.
344///
345/// # Safety
346/// If `ALIGNED`, `bytes` are aligned for `T`.
347#[inline(always)]
348pub(crate) unsafe fn try_skip_valid<T, const ALIGNED: bool>(
349    bytes: &[u8],
350    count: usize,
351) -> Option<&[u8]>
352where
353    T: TrivialType,
354{
355    // Zero-sized values have no bytes, so only a check like a `validate` hook can reject one, and
356    // it rejects either all of them or none, so one check replaces `count` steps that don't consume
357    // bytes
358    if size_of::<T>() == 0 {
359        if count > 0 {
360            // SAFETY: If `ALIGNED`, `bytes` are aligned for `T` (function contract)
361            unsafe { T::parse_step::<ALIGNED>(bytes) }?;
362        }
363        return Some(bytes);
364    }
365    // With all elements split off first, every step gets exactly the bytes of one element, so for
366    // types whose step only checks the length the loop is optimized away
367    let (elements, remainder) = bytes.split_at_checked(count.checked_mul(size_of::<T>())?)?;
368    for element in elements.as_chunks::<{ SIZE_OF::<T> }>().0 {
369        // SAFETY: If `ALIGNED`, `bytes` are aligned for `T` (function contract), and each element
370        // is at an offset into them that is a multiple of the size of `T`, and so of its alignment
371        unsafe { T::parse_step::<ALIGNED>(element) }?;
372    }
373    Some(remainder)
374}

402        Some((value, remainder))
403    }

538const fn array_metadata(size: u32, inner_metadata: &[u8]) -> ([u8; MAX_METADATA_CAPACITY], usize) {

546// TODO: Change `usize` to `u32` once it is possible:
547//  https://github.com/rust-lang/rust/issues/156745
548// SAFETY: Elements follow each other without padding and are `TrivialType`, the parsing step checks
549// every element
550unsafe impl<const SIZE: usize, T> TrivialType for [T; SIZE]
551where
552    T: TrivialType,
553    [(); SUPPORTED_ARRAY::<T, SIZE>]:,
554{
555    const ALIGNMENT: Alignment = T::ALIGNMENT;
556    // Casting `SIZE` to `u32` is lossless, which is checked by the `SUPPORTED_ARRAY` bound
557    const METADATA: &[u8] = {
558        // The reference extends the lifetime of metadata scratch to that of the constant
559        let (metadata_scratch, metadata_size) = &array_metadata(SIZE as u32, T::METADATA);
560        metadata_scratch.split_at(*metadata_size).0
561    };
562
563    #[inline(always)]
564    unsafe fn parse_step<const ALIGNED: bool>(bytes: &[u8]) -> Option<(&Unaligned<Self>, &[u8])> {
565        let (bytes, remainder) = bytes.split_at_checked(size_of::<Self>())?;
566        // SAFETY: If `ALIGNED`, `bytes` are aligned for `Self` (function contract), which has the
567        // alignment of `T`
568        unsafe { try_skip_valid::<T, ALIGNED>(bytes, SIZE) }?;
569        // SAFETY: Elements follow each other without padding and each of them is a valid value as
570        // checked above, `Unaligned<Self>` has alignment 1 and the size of `Self`
571        let value = unsafe { bytes.as_ptr().cast::<Unaligned<Self>>().as_ref_unchecked() };
572        Some((value, remainder))
573    }
574}
575
576// SAFETY: Elements follow each other without padding, any bit pattern of each is a valid value,
577// which the parsing step of elements accepts. There is no array size limit here, `AnyBitPattern`
578// requires `[T; N]: TrivialType`, which has it.
579unsafe impl<const SIZE: usize, T> AnyBitPatternMarker for [T; SIZE] where T: AnyBitPatternMarker {}

581// SAFETY: All `TrivialType` instances are valid for `IoType`
582unsafe impl<T> IoType for T
583where
584    T: TrivialType,
585{
586    const METADATA: &[u8] = T::METADATA;
587
588    type PointerType = T;
589
590    type WithLifetime<'b> = T;
591
592    type Reborrowed<'b> = &'b mut T;
593
594    #[inline(always)]
595    fn size(&self) -> u32 {
596        T::SIZE
597    }
598
599    #[inline(always)]
600    fn capacity(&self) -> u32 {
601        self.size()
602    }
603
604    #[inline(always)]
605    #[track_caller]
606    unsafe fn set_size(&mut self, size: u32) {
607        debug_assert_eq!(size, T::SIZE, "`set_size` called with invalid input");
608    }
609
610    #[inline(always)]
611    #[track_caller]
612    unsafe fn from_ptr<'a>(
613        ptr: &'a NonNull<Self::PointerType>,
614        size: &'a u32,
615        capacity: u32,
616    ) -> impl Deref<Target = Self::WithLifetime<'a>> + 'a {
617        debug_assert!(ptr.is_aligned(), "Misaligned pointer");
618        debug_assert_eq!(*size, T::SIZE, "Invalid size");
619        debug_assert!(
620            *size <= capacity,
621            "Size {size} must not exceed capacity {capacity}"
622        );
623
624        // SAFETY: guaranteed by this function signature
625        unsafe { ptr.as_ref() }
626    }
627
628    #[inline(always)]
629    #[track_caller]
630    unsafe fn from_mut_ptr<'a>(
631        ptr: &'a mut NonNull<Self::PointerType>,
632        _size: &'a mut u32,
633        capacity: u32,
634    ) -> impl DerefMut<Target = Self::WithLifetime<'a>> + 'a {
635        debug_assert!(ptr.is_aligned(), "Misaligned pointer");
636        debug_assert!(
637            Self::SIZE <= capacity,
638            "Size {} must not exceed capacity {capacity}",
639            Self::SIZE
640        );
641
642        // SAFETY: guaranteed by this function signature
643        unsafe { ptr.as_mut() }
644    }
645
646    #[inline(always)]
647    #[track_caller]
648    unsafe fn try_from_ptr<'a>(
649        ptr: &'a NonNull<Self::PointerType>,
650        size: &'a u32,
651        capacity: u32,
652    ) -> Option<impl Deref<Target = Self::WithLifetime<'a>> + 'a> {
653        debug_assert!(ptr.is_aligned(), "Misaligned pointer");
654        debug_assert_eq!(*size, T::SIZE, "Invalid size");
655        debug_assert!(
656            *size <= capacity,
657            "Size {size} must not exceed capacity {capacity}"
658        );
659        // SAFETY: `ptr` is valid for reads of `size` bytes, which is the size of `T`, initialized
660        // and not written to during the call (function contract)
661        let bytes = unsafe { slice::from_raw_parts(ptr.as_ptr().cast::<u8>(), size_of::<T>()) };
662        // SAFETY: `ptr` is aligned for `T` (function contract)
663        unsafe { T::parse_step::<true>(bytes) }?;
664
665        // SAFETY: `size` is the size of `T` and doesn't exceed `capacity`, `ptr` is aligned and the
666        // memory is not written to while the returned value exists (function contract), and it
667        // holds a valid value of `T` as checked above
668        Some(unsafe { Self::from_ptr(ptr, size, capacity) })
669    }
670
671    #[inline(always)]
672    #[track_caller]
673    unsafe fn try_from_mut_ptr<'a>(
674        ptr: &'a mut NonNull<Self::PointerType>,
675        size: &'a mut u32,
676        capacity: u32,
677    ) -> Option<impl DerefMut<Target = Self::WithLifetime<'a>> + 'a> {
678        debug_assert!(ptr.is_aligned(), "Misaligned pointer");
679        debug_assert_eq!(*size, T::SIZE, "Invalid size");
680        debug_assert!(
681            *size <= capacity,
682            "Size {size} must not exceed capacity {capacity}"
683        );
684        // SAFETY: `ptr` is valid for reads of `size` bytes, which is the size of `T`, initialized
685        // and not accessed otherwise during the call (function contract), the slice is not used
686        // after parsing
687        let bytes = unsafe { slice::from_raw_parts(ptr.as_ptr().cast::<u8>(), size_of::<T>()) };
688        // SAFETY: `ptr` is aligned for `T` (function contract)
689        unsafe { T::parse_step::<true>(bytes) }?;
690
691        // SAFETY: `size` is the size of `T` and doesn't exceed `capacity`, `ptr` is aligned and the
692        // memory is not accessed otherwise while the returned value exists (function contract),
693        // and it holds a valid value of `T` as checked above
694        Some(unsafe { Self::from_mut_ptr(ptr, size, capacity) })
695    }
696
697    #[inline(always)]
698    fn as_ptr(&self) -> NonNull<Self::PointerType> {
699        NonNull::from_ref(self)
700    }
701
702    #[inline(always)]
703    fn as_mut_ptr(&mut self) -> NonNull<Self::PointerType> {
704        NonNull::from_mut(self)
705    }
706
707    #[inline(always)]
708    fn reborrow_mut(&mut self) -> Self::Reborrowed<'_> {
709        self
710    }
