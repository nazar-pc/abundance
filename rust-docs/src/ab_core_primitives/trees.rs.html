1//! Merkle Trees used by the protocol.
2//!
3//! Every tree hashes its inner nodes with its own key, so a node of one tree can't be interpreted
4//! as a node of another tree. Trees with a number of leaves that isn't known at compile time also
5//! commit to it.

59const {
60    let keys = [
61        Blake3MerkleHasher::KEY,
62        SegmentRootsMerkleHasher::BASE_KEY,
63        OwnSegmentsMerkleHasher::KEY,
64        ShardOwnSegmentsMerkleHasher::KEY,
65        LeafShardBodyMerkleHasher::KEY,
66        TransactionsMerkleHasher::BASE_KEY,
67        IntermediateShardBodyMerkleHasher::KEY,
68        LeafShardsSegmentsMerkleHasher::BASE_KEY,
69        ChildLeafShardBlocksMerkleHasher::BASE_KEY,
70        ChildIntermediateShardBlocksMerkleHasher::BASE_KEY,
71        BeaconChainBodyMerkleHasher::KEY,
72        IntermediateShardsSegmentsMerkleHasher::BASE_KEY,
73        BeaconChainHeaderMerkleHasher::KEY,
74        IntermediateShardHeaderMerkleHasher::KEY,
75        LeafShardHeaderMerkleHasher::KEY,
76        SuperSegmentMerkleHasher::BASE_KEY,
77        StateOwnersSparseMerkleHasher::KEY,
78        StateOwnerContractsSparseMerkleHasher::KEY,
79        BlockMerkleMountainRangeHasher::KEY,
80    ];
81
82    let mut i = 0;
83    while i < keys.len() {
84        let mut j = i + 1;
85        while j < keys.len() {
86            assert!(keys[i] != keys[j], "Keys of Merkle Trees must be unique");
87            j += 1;
88        }
89        i += 1;
90    }
91}

93/// Number of bits in [`Address`], which is the number of levels of state sparse Merkle Trees
94const ADDRESS_BITS: u8 = Address::SIZE as u8 * u8::BITS as u8;
