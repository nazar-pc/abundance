1//! Merkle Trees used by the protocol.
2//!
3//! Every tree hashes its inner nodes with its own key, so a node of one tree can't be interpreted
4//! as a node of another tree. Trees with a number of leaves that isn't known at compile time also
5//! commit to it.

63const {
64    let keys = [
65        Blake3MerkleHasher::KEY,
66        SegmentRootsMerkleHasher::BASE_KEY,
67        OwnSegmentsMerkleHasher::KEY,
68        ShardOwnSegmentsMerkleHasher::KEY,
69        LeafShardBodyMerkleHasher::KEY,
70        TransactionsMerkleHasher::BASE_KEY,
71        IntermediateShardBodyMerkleHasher::KEY,
72        LeafShardsSegmentsMerkleHasher::BASE_KEY,
73        ChildLeafShardBlocksMerkleHasher::BASE_KEY,
74        ChildIntermediateShardBlocksMerkleHasher::BASE_KEY,
75        BeaconChainBodyMerkleHasher::KEY,
76        IntermediateShardsSegmentsMerkleHasher::BASE_KEY,
77        BeaconChainHeaderMerkleHasher::KEY,
78        IntermediateShardHeaderMerkleHasher::KEY,
79        LeafShardHeaderMerkleHasher::KEY,
80        SuperSegmentMerkleHasher::BASE_KEY,
81        StateOwnersSparseMerkleHasher::KEY,
82        StateOwnerContractsSparseMerkleHasher::KEY,
83        BlockMerkleMountainRangeHasher::KEY,
84    ];
85
86    let mut i = 0;
87    while i < keys.len() {
88        let mut j = i + 1;
89        while j < keys.len() {
90            assert!(keys[i] != keys[j], "Keys of Merkle Trees must be unique");
91            j += 1;
92        }
93        i += 1;
94    }
95}

151// `MerkleHasher` is implemented for each hasher separately, a generic implementation would need
152// `<Self as MerkleHasher>::OUT_LEN` with generic `Self` in method signatures, which requires
153// `macroless_generic_const_args` that this crate doesn't use
154// TODO: Replace the macro with generic implementations once `macroless_generic_const_args` is
155//  enabled in this crate:
156// // `<Self as MerkleHasher>::OUT_LEN` doesn't unify with `OUT_LEN` in generic code, hence
157// // reshaping
158// impl<const KEY: [u8; KEY_LEN]> MerkleHasher for Blake3StaticMerkleHasher<KEY> {
159//     const OUT_LEN: usize = OUT_LEN;
160//
161//     #[inline(always)]
162//     fn hash_pair(
163//         &self,
164//         pair: &[[u8; <Self as MerkleHasher>::OUT_LEN]; 2],
165//     ) -> [u8; <Self as MerkleHasher>::OUT_LEN] {
166//         let hash =
167//             ab_blake3::single_block_keyed_hash(self.node_key(), pair.flatten_ref::<BLOCK_LEN>());
168//         // Reshaping inside `Option` results in fewer copies of the hash than after `expect()`
169//         hash.map(|hash| [hash].flatten())
170//             .expect("Exactly one block worth of data; qed")
171//     }
172//
173//     #[inline(always)]
174//     fn hash_pairs<const NUM_PAIRS: usize>(
175//         &self,
176//         pairs: &[[[u8; <Self as MerkleHasher>::OUT_LEN]; 2]; NUM_PAIRS],
177//     ) -> [[u8; <Self as MerkleHasher>::OUT_LEN]; NUM_PAIRS] {
178//         let pairs = pairs
179//             .as_slice()
180//             .flatten_each_ref::<BLOCK_LEN>()
181//             .as_array::<NUM_PAIRS>()
182//             .expect("Same number of pairs as in the input; qed");
183//         let mut hashes = [[0; <Self as MerkleHasher>::OUT_LEN]; NUM_PAIRS];
184//         ab_blake3::single_block_keyed_hash_many_exact(
185//             self.node_key(),
186//             pairs,
187//             hashes.rechunk_mut(),
188//         );
189//         hashes
190//     }
191//
192//     #[inline(always)]
193//     fn hash_single_leaf(
194//         &self,
195//         leaf: &[u8; <Self as MerkleHasher>::OUT_LEN],
196//     ) -> [u8; <Self as MerkleHasher>::OUT_LEN] {
197//         [self.single_leaf_root(array::from_ref(leaf).flatten_ref::<OUT_LEN>())].flatten()
198//     }
199// }
200//
201// // Same body as above
202// impl<const BASE_KEY: [u8; KEY_LEN]> MerkleHasher for Blake3SizedMerkleHasher<BASE_KEY> { ... }
203macro_rules! impl_merkle_hasher {
204    ($($hasher:ty),+ $(,)?) => {
205        $(
206            impl MerkleHasher for $hasher {
207                const OUT_LEN: usize = OUT_LEN;
208
209                #[inline(always)]
210                fn hash_pair(&self, pair: &[[u8; OUT_LEN]; 2]) -> [u8; OUT_LEN] {
211                    ab_blake3::single_block_keyed_hash(self.node_key(), pair.as_flattened())
212                        .expect("Exactly one block worth of data; qed")
213                }
214
215                #[inline(always)]
216                fn hash_pairs<const NUM_PAIRS: usize>(
217                    &self,
218                    pairs: &[[[u8; OUT_LEN]; 2]; NUM_PAIRS],
219                ) -> [[u8; OUT_LEN]; NUM_PAIRS] {
220                    let pairs = pairs
221                        .as_slice()
222                        .flatten_each_ref::<BLOCK_LEN>()
223                        .as_array::<NUM_PAIRS>()
224                        .expect("Same number of pairs as in the input; qed");
225                    let mut hashes = [[0; OUT_LEN]; NUM_PAIRS];
226                    ab_blake3::single_block_keyed_hash_many_exact(
227                        self.node_key(),
228                        pairs,
229                        &mut hashes,
230                    );
231                    hashes
232                }
233
234                #[inline(always)]
235                fn hash_single_leaf(&self, leaf: &[u8; OUT_LEN]) -> [u8; OUT_LEN] {
236                    self.single_leaf_root(leaf)
237                }
238            }
239        )+
240    };
241}
