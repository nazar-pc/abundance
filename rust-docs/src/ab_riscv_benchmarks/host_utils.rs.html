114/// Simple test memory implementation
115#[derive(Debug, Copy, Clone)]
116#[repr(align(16))]
117pub struct TestMemory<const BASE_ADDR: u64, const SIZE: usize> {
118    data: [u8; SIZE],
119}
120
121impl<const BASE_ADDR: u64, const SIZE: usize> VirtualMemory for TestMemory<BASE_ADDR, SIZE> {
122    #[inline(always)]
123    fn read<T>(&self, address: u64) -> Result<T, VirtualMemoryError>
124    where
125        T: BasicInt,
126    {
127        let offset = address.wrapping_sub(BASE_ADDR);
128
129        if offset.saturating_add(size_of::<T>() as u64) > self.data.len() as u64 {
130            cold_path();
131            return Err(VirtualMemoryError::OutOfBoundsRead { address });
132        }
133
134        // SAFETY: Only reading basic integers from initialized memory
135        unsafe {
136            Ok(self
137                .data
138                .as_ptr()
139                .cast::<T>()
140                .byte_add(offset as usize)
141                .read_unaligned())
142        }
143    }
144
145    #[inline(always)]
146    unsafe fn read_unchecked<T>(&self, address: u64) -> T
147    where
148        T: BasicInt,
149    {
150        // SAFETY: Guaranteed by function contract
151        unsafe {
152            let offset = address.unchecked_sub(BASE_ADDR) as usize;
153            self.data
154                .as_ptr()
155                .cast::<T>()
156                .byte_add(offset)
157                .read_unaligned()
158        }
159    }
160
161    fn read_slice(&self, address: u64, len: u32) -> Result<&[u8], VirtualMemoryError> {
162        let offset = address.wrapping_sub(BASE_ADDR);
163
164        if offset > self.data.len() as u64 {
165            cold_path();
166            return Err(VirtualMemoryError::OutOfBoundsRead { address });
167        }
168
169        self.data
170            .get(offset as usize..)
171            .and_then(|data| data.get(..len as usize))
172            .ok_or(VirtualMemoryError::OutOfBoundsRead { address })
173    }
174
175    fn read_slice_up_to(&self, address: u64, len: u32) -> &[u8] {
176        let offset = address.wrapping_sub(BASE_ADDR);
177
178        if offset > self.data.len() as u64 {
179            cold_path();
180            return &[];
181        }
182
183        let remaining = self.data.get(offset as usize..).unwrap_or_default();
184        remaining.get(..len as usize).unwrap_or(remaining)
185    }
186
187    #[inline(always)]
188    fn write<T>(&mut self, address: u64, value: T) -> Result<(), VirtualMemoryError>
189    where
190        T: BasicInt,
191    {
192        let offset = address.wrapping_sub(BASE_ADDR);
193
194        if offset.saturating_add(size_of::<T>() as u64) > self.data.len() as u64 {
195            cold_path();
196            return Err(VirtualMemoryError::OutOfBoundsWrite { address });
197        }
198
199        // SAFETY: Only writing basic integers to initialized memory
200        unsafe {
201            self.data
202                .as_mut_ptr()
203                .cast::<T>()
204                .byte_add(offset as usize)
205                .write_unaligned(value);
206        }
207
208        Ok(())
209    }
210
211    fn write_slice(&mut self, address: u64, data: &[u8]) -> Result<(), VirtualMemoryError> {
212        let offset = address.wrapping_sub(BASE_ADDR);
213
214        if offset > self.data.len() as u64 {
215            cold_path();
216            return Err(VirtualMemoryError::OutOfBoundsWrite { address });
217        }
218
219        let len = data.len();
220        let Some(target_data) = self
221            .data
222            .get_mut(offset as usize..)
223            .and_then(|data| data.get_mut(..len))
224        else {
225            cold_path();
226            return Err(VirtualMemoryError::OutOfBoundsWrite { address });
227        };
228
229        target_data.copy_from_slice(data);
230
231        Ok(())
232    }
233}

267/// Lazy instruction fetcher implementation
268#[derive(Debug, Copy, Clone)]
269pub struct LazyInstructionFetcher {
270    return_trap_address: u64,
271    pc: u64,
272}
273
274impl<Memory> ProgramCounter<u64, Memory> for LazyInstructionFetcher
275where
276    Memory: VirtualMemory,
277{
278    #[inline(always)]
279    fn get_pc(&self) -> u64 {
280        self.pc
281    }
282
283    #[inline(always)]
284    fn set_pc_relative(
285        &mut self,
286        memory: &Memory,
287        instruction_size: u8,
288        offset: i32,
289    ) -> Result<ControlFlow<()>, ExecutionError<u64>> {
290        let old_pc = <Self as ProgramCounter<_, Memory, _>>::old_pc(self, instruction_size);
291        self.set_pc(memory, old_pc.wrapping_add_signed(i64::from(offset)))
292    }
293
294    #[inline]
295    fn set_pc(&mut self, memory: &Memory, pc: u64) -> Result<ControlFlow<()>, ExecutionError<u64>> {
296        if pc == self.return_trap_address {
297            cold_path();
298            return Ok(ControlFlow::Break(()));
299        }
300
301        if !pc.is_multiple_of(u64::from(
302            ContractInstruction::<ContractRegister>::alignment(),
303        )) {
304            cold_path();
305            return Err(ExecutionError::UnalignedInstruction {
306                address: PackedAddress::new(pc),
307            });
308        }
309
310        // Note: This will not allow reading a 16-bit instruction at the very end of memory range,
311        // but that is going to be the case here anyway since code is followed by read-write memory
312        // anyway
313        if let Err(error) = memory.read::<u32>(pc) {
314            cold_path();
315            return Err(error.into());
316        }
317
318        self.pc = pc;
319
320        Ok(ControlFlow::Continue(()))
321    }
322}

324impl<Memory> InstructionFetcher<ContractInstruction, Memory> for LazyInstructionFetcher
325where
326    Memory: VirtualMemory,
327{
328    #[inline]
329    fn fetch_instruction(
330        &mut self,
331        memory: &Memory,
332    ) -> FetchInstructionResult<ContractInstruction> {
333        // SAFETY: Constructor guarantees that the last instruction is a jump, which means going
334        // through `Self::set_pc()` method does the necessary bounds check and advancing forward by
335        // one instruction can't result in out-of-bounds access.
336        let instruction = unsafe { memory.read_unchecked(self.pc) };
337        // SAFETY: All instructions are valid, according to the constructor contract
338        let instruction =
339            unsafe { ContractInstruction::try_decode(instruction).unwrap_unchecked() };
340
341        self.pc += u64::from(instruction.size());
342
343        FetchInstructionResult::Instruction(instruction)
344    }
345}

361        }

365/// Eager instruction handler eagerly decodes all instructions upfront
366#[derive(Debug, Clone)]
367#[repr(C, align(16))]
368pub struct EagerTestInstructionFetcher {
369    decoded_instruction_byte_offset: usize,
370    // A simple raw pointer separate field helps LLVM with SROA and aliasing analysis, so it can
371    // retain this pointer in the native register
372    instructions: Box<[ContractInstruction]>,
373    base_addr: u64,
374    return_trap_address: u64,
375}
376
377impl<Memory> ProgramCounter<u64, Memory> for EagerTestInstructionFetcher
378where
379    Memory: VirtualMemory,
380{
381    #[inline(always)]
382    fn get_pc(&self) -> u64 {
383        self.base_addr
384            + self.decoded_instruction_byte_offset as u64 * size_of::<u16>() as u64
385                / size_of::<ContractInstruction>() as u64
386    }
387
388    /// Moves within the decoded stream instead of resolving an address and converting it back,
389    /// which is what going through [`Self::set_pc()`] would do.
390    ///
391    /// One comparison and one test are all this needs to hand every case it does not handle itself
392    /// over to [`Self::set_pc()`]: a target past the end of the decoded stream, a backwards branch
393    /// that ran off its start, and an unaligned target. The return trap sits outside the decoded
394    /// stream, so a branch to it fails the bounds check here and is stopped by [`Self::set_pc()`]
395    /// like any other jump to it.
396    #[inline(always)]
397    fn set_pc_relative(
398        &mut self,
399        memory: &Memory,
400        instruction_size: u8,
401        offset: i32,
402    ) -> Result<ControlFlow<()>, ExecutionError<u64>> {
403        // Byte offset from the instruction being executed to the branch target. The program counter
404        // is advanced during instruction fetching, so that instruction starts `instruction_size`
405        // bytes back.
406        let offset = offset as isize - isize::from(instruction_size);
407        // Every `size_of::<u16>()` of guest code owns one decoded instruction, so the target is
408        // reached by moving within the decoded stream
409        let decoded_instruction_byte_offset =
410            self.decoded_instruction_byte_offset.wrapping_add_signed(
411                offset * (size_of::<ContractInstruction>() / size_of::<u16>()).cast_signed(),
412            );
413
414        // A target that does not land on a decoded instruction sits between two guest
415        // instructions, which makes it an unaligned instruction rather than something to round to
416        // the start of one. That rule lives in `set_pc()`, so rather than restating it here, where
417        // it could drift, such a target simply fails to qualify for the fast path, as does one past
418        // the end of the decoded stream, which a backwards branch that ran off its start wraps
419        // around into.
420        if decoded_instruction_byte_offset
421            >= self.instructions.len() * size_of::<ContractInstruction>()
422            || !decoded_instruction_byte_offset.is_multiple_of(size_of::<ContractInstruction>())
423        {
424            cold_path();
425            let pc = <Self as ProgramCounter<_, Memory>>::get_pc(self);
426            return self.set_pc(memory, pc.wrapping_add_signed(offset as i64));
427        }
428
429        self.decoded_instruction_byte_offset = decoded_instruction_byte_offset;
430
431        Ok(ControlFlow::Continue(()))
432    }
433
434    #[inline]
435    fn set_pc(
436        &mut self,
437        _memory: &Memory,
438        pc: u64,
439    ) -> Result<ControlFlow<()>, ExecutionError<u64>> {
440        let address = pc;
441
442        if address == self.return_trap_address {
443            cold_path();
444            return Ok(ControlFlow::Break(()));
445        }
446
447        if !address.is_multiple_of(size_of::<u16>() as u64) {
448            cold_path();
449            return Err(ExecutionError::UnalignedInstruction {
450                address: PackedAddress::new(address),
451            });
452        }
453
454        let Some(offset) = address.checked_sub(self.base_addr) else {
455            cold_path();
456            return Err(ExecutionError::OutOfBoundsRead {
457                address: PackedAddress::new(address),
458            });
459        };
460        let offset = offset as usize;
461        let instruction_offset = offset / size_of::<u16>();
462
463        if instruction_offset >= self.instructions.len() {
464            cold_path();
465            return Err(VirtualMemoryError::OutOfBoundsRead { address }.into());
466        }
467
468        self.decoded_instruction_byte_offset =
469            instruction_offset * size_of::<ContractInstruction>();
470
471        Ok(ControlFlow::Continue(()))
472    }
473}

475impl<Memory> InstructionFetcher<ContractInstruction, Memory> for EagerTestInstructionFetcher
476where
477    Memory: VirtualMemory,
478{
479    #[inline(always)]
480    fn fetch_instruction(
481        &mut self,
482        _memory: &Memory,
483    ) -> FetchInstructionResult<ContractInstruction> {
484        // SAFETY: Constructor guarantees that the last instruction is a jump, which means going
485        // through `Self::set_pc()` method does the necessary bounds check and advancing forward by
486        // one instruction can't result in out-of-bounds access.
487        let instruction = unsafe {
488            // Reading through byte offset rather than index to avoid extra computation (converting
489            // an index to a byte offset) on each fetch
490            self.instructions
491                .as_ptr()
492                .byte_add(self.decoded_instruction_byte_offset)
493                .read()
494        };
495        self.decoded_instruction_byte_offset +=
496            usize::from(instruction.size()) / size_of::<u16>() * size_of::<ContractInstruction>();
497
498        FetchInstructionResult::Instruction(instruction)
499    }
500}

502impl EagerTestInstructionFetcher {
503    /// Create a new instance with the specified instructions and base address.
504    ///
505    /// Instructions are decoded during instantiation of the instruction fetcher, and the base
506    /// address corresponds to the first instruction.
507    ///
508    /// `return_trap_address` is the address at which the interpreter will stop execution
509    /// (gracefully).
510    ///
511    /// # Safety
512    /// The program counter must be valid and aligned, the instructions processed must end with a
513    /// jump instruction, and `return_trap_address` must not fall inside them. Instruction fetching
514    /// does not compare against the return trap, so an address inside the instructions would stop
515    /// execution when jumped to but not when reached by falling through.
516    #[inline(always)]
517    pub unsafe fn new(
518        instructions: &[u8],
519        return_trap_address: u64,
520        base_addr: u64,
521        pc: u64,
522    ) -> Self {
523        let mut decoded_instructions = Vec::with_capacity(instructions.len() / size_of::<u16>());
524
525        let mut offset = 0;
526        while let Some(instruction_bytes) = instructions.get(offset..offset + size_of::<u32>()) {
527            let decoded_instruction = u32::from_le_bytes([
528                instruction_bytes[0],
529                instruction_bytes[1],
530                instruction_bytes[2],
531                instruction_bytes[3],
532            ]);
533            // Use `Unimp` as a fallback, though contract is expected to only contain legal
534            // instructions
535            let decoded_instruction = Instruction::try_decode(decoded_instruction).unwrap_or(
536                ContractInstruction::Unimp {
537                    rs1: Register::ZERO,
538                    rs2: Register::ZERO,
539                },
540            );
541            decoded_instructions.push(decoded_instruction);
542            match decoded_instruction.size() {
543                2 => {
544                    offset += 2;
545                }
546                4 => {
547                    // The second half of a 32-bit instruction is a valid offset and may or may not
548                    // decode to a valid instruction on its own. Try to decode it but ignore
549                    // decoding failures.
550
551                    offset += 2;
552
553                    // Could be both 16-bit and 32-bit instruction, need to handle end of the
554                    // instruction stream
555                    let instruction_word = if let Some(instruction_bytes) =
556                        instructions.get(offset..offset + size_of::<u32>())
557                    {
558                        u32::from_le_bytes([
559                            instruction_bytes[0],
560                            instruction_bytes[1],
561                            instruction_bytes[2],
562                            instruction_bytes[3],
563                        ])
564                    } else {
565                        u32::from_le_bytes([instruction_bytes[2], instruction_bytes[3], 0, 0])
566                    };
567
568                    decoded_instructions.push(Instruction::try_decode(instruction_word).unwrap_or(
569                        ContractInstruction::Unimp {
570                            rs1: Register::ZERO,
571                            rs2: Register::ZERO,
572                        },
573                    ));
574                    offset += 2;
575                }
576                instruction_size => {
577                    unreachable!("Invalid instruction size {instruction_size}, expected 2 or 4");
578                }
579            }
580        }
581
582        let remainder_bytes = instructions.get(offset..).unwrap_or(&[]);
583
584        if remainder_bytes.len() == size_of::<u16>() {
585            let instruction_word =
586                u32::from_le_bytes([remainder_bytes[0], remainder_bytes[1], 0, 0]);
587            decoded_instructions.push(Instruction::try_decode(instruction_word).unwrap_or(
588                ContractInstruction::Unimp {
589                    rs1: Register::ZERO,
590                    rs2: Register::ZERO,
591                },
592            ));
593        }
594
595        let instructions = decoded_instructions.into_boxed_slice();
596        Self {
597            decoded_instruction_byte_offset: (pc - base_addr) as usize / size_of::<u16>()
598                * size_of::<ContractInstruction>(),
599            instructions,
600            base_addr,
601            return_trap_address,
602        }
603    }
